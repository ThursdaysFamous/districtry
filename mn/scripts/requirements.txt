# dependency drift). Used by the weekly roster workflows:
#   .github/workflows/update-ia-congress-roster.yml     (stdlib only)
#   .github/workflows/update-ia-legislature-roster.yml  (+ requests)
#   .github/workflows/ia-validate-sources.yml           (+ requests)
#
# ia/scripts/build_state_counties.py and build_legislative_boundaries.py use
# only curl (via subprocess) and npx mapshaper — no Python HTTP library.
requests==2.33.1

# Conversion 2 (docs/MECHANIZATION_PLAYBOOK.md): worksheet validation in
# scripts/generate_metro_files.py. The generator is otherwise stdlib-only;
# validate_index.py deliberately stays stdlib and does NOT import this.
jsonschema==4.26.0

# PDF text extraction for the two county offices whose only statewide directory
# is a published document rather than a page or an API: the sheriff (ISSDA's
# directory) and the county attorney (the ICAA's roster), both read by
# ia/scripts/ia_county_officer_sources_scraper.py for
# .github/workflows/update-ia-county-officers-roster.yml.
#
# pdfplumber AND NOT pypdf, and the reason is not style. Both documents are
# MULTI-COLUMN. pypdf flattens a page to text, which interleaves the columns --
# it lands one county's sheriff on a line with two other counties' text, and
# glues the ICAA's APPOINTED assistants onto the elected county attorney's
# name, which is the one thing that parse must never do. Both parses therefore
# assign every word to a column by its x-coordinate BEFORE forming a line, and
# extract_words() is what exposes that; pypdf has no equivalent.
pdfplumber==0.11.10

# Geometry for ia/scripts/build_story_supervisor_districts.py, which reads
# Story County's supervisor districts off the county Auditor's printed map.
# An OPERATOR step, not one of the weekly workflows above: these lines move
# when a county redraws them, which Senate File 75 has just done and which
# will not happen again for years. pdfplumber (pinned above for the sheriff
# and county-attorney directories) supplies the vector path objects.
#
# shapely does the point-in-ring and dissolve work; pyproj holds the
# projection. NAD83 / Iowa North is not incidental -- the map's drawn aspect
# ratio matches that projection's to five digits and no other candidate's, so
# reading it needs a real CRS transform rather than a linear degree fit, which
# lands 26 m out.
shapely==2.1.2
pyproj==3.7.2
