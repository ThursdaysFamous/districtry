# Pinned dependencies for this instance's scraper pipeline (playbook R10 —
# heuristic HTML parsers are exactly the code that breaks on silent
# dependency drift). Used by the weekly and monthly workflows:
#   .github/workflows/update-la-congress-roster.yml     (stdlib only)
#   .github/workflows/update-la-legislature-roster.yml  (+ requests)
#   .github/workflows/update-la-parish-officials.yml    (+ requests)
#   .github/workflows/la-validate-sources.yml           (+ requests)
#
# The workflows install with `-c scripts/requirements.txt`, the fleet's one
# set of pins; these lines repeat the versions so a local run can install
# from this file alone.
#
# la/scripts/build_state_counties.py and build_legislative_boundaries.py use
# only curl (via subprocess) and npx mapshaper — no Python HTTP library.
# build_metro_outline.py imports requests inside its one fetching function.
#
# The legislature and parish-officials builders fetch with requests, which
# retries a reset or a timeout the same way on both; the pages and the
# Secretary of State's CSV exports are parsed with the stdlib (re, html, csv).
requests==2.33.1

# la/scripts/build_la_parish_districts.py measures the districts against the
# 2020 census's blocks with shapely when it builds; its --check (in CI) is
# stdlib only.
shapely==2.1.2

# la/scripts/build_la_conservation_districts.py reads the Department of
# Agriculture and Forestry's map and list of district contacts (a PDF) with
# pdfplumber when it builds; its --check (in CI) is stdlib only.
# la/scripts/measure_la_conservation_parishes.py, run by hand when the map
# changes, also needs pyproj to place the map's drawing on the parish
# outlines, and numpy, which shapely already installs.
pdfplumber==0.11.10
pyproj==3.7.2

# Conversion 2 (docs/MECHANIZATION_PLAYBOOK.md): worksheet validation in
# scripts/generate_metro_files.py. The generator is otherwise stdlib-only;
# validate_index.py deliberately stays stdlib and does NOT import this.
jsonschema==4.26.0
