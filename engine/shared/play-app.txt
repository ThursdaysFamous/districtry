/* THE GOOGLE PLAY APP'S MARK, in ONE copy, for the pages that show a
   support link: every app, every sources page, the front page and the
   sponsorship page.

   districtry's Android app on Google Play is this site in a full-screen
   Chrome window (a Trusted Web Activity), so it shows these same pages.
   Google Play's Payments policy does not let an app send a reader to a
   payment method other than Google's own billing, and Google applies that
   to donations: in February 2022 it rejected StreetComplete, an open-source
   map app, for a donation link to its GitHub project page, and accepted it
   once the link was gone. So inside the Play app the links to GitHub
   Sponsors and to the sponsorship page are hidden. The website keeps them,
   and so does a copy installed from the browser's own menu, which no store
   reviews.

   HOW A WINDOW IS KNOWN TO BE THE PLAY APP. The app is built to open
   /app/?ref=play, and the launcher passes ref=play on, and nothing else, to
   the page it opens: a URL is how the Trusted Web Activity documentation
   says an app passes data to its pages. A page whose referrer is the app
   itself (android-app://com.districtry.app, the package name the setup
   guide builds) is marked too. Chrome names the Android app that opened a
   page as its referrer, which is how a link tapped in Gmail arrives as
   android-app://com.google.android.gm, so this should cover a shared link
   opened straight into the app rather than through its icon. That second
   path is not yet measured on a phone; the first is the one the app's own
   launches take, and the one tested here. Either sets a mark in session
   storage, so every later page in the same window (another state, a sources
   page, the front page) hides the links as well, and the mark goes when the
   app is closed. Session storage belongs to one window, so the mark never
   reaches the reader's ordinary browser tabs.

   WHAT IS HIDDEN is matched by address rather than by a class, so a support
   link added later to any page carrying this block is hidden in the Play app
   without anyone remembering to mark it: every link to github.com/sponsors
   or to sponsorship.html, and the paragraph or list item around one, because
   a sentence asking for sponsorship reads wrong with its link taken out.
   The :has() selectors are a rule of their own because a browser that does
   not know :has() drops every selector in a rule that names it, and the
   app panel's sponsorship sentence, .dpf-support, is named by class in the
   first rule for that browser.

   It runs in the head, before first paint, so the links never flash, and an
   ordinary visit adds no style at all. */
(function () {
  var opened = /(?:^|&)ref=play(?:&|$)/.test(location.search.slice(1)) ||
    /^android-app:\/\/com\.districtry\.app(?:\/|$)/.test(document.referrer || "");
  var marked = opened;
  try {
    if (opened) sessionStorage.setItem("districtry-play-app", "1");
    else marked = sessionStorage.getItem("districtry-play-app") === "1";
  } catch (e) { /* blocked storage: only the page the app opened is marked */ }
  if (!marked) return;
  document.documentElement.classList.add("dst-play-app");
  var css = document.createElement("style");
  css.textContent =
    ".dst-play-app a[href*=\"github.com/sponsors\"]," +
    ".dst-play-app a[href*=\"sponsorship.html\"]," +
    ".dst-play-app .dpf-support{display:none!important}" +
    ".dst-play-app p:has(a[href*=\"github.com/sponsors\"])," +
    ".dst-play-app li:has(a[href*=\"github.com/sponsors\"])," +
    ".dst-play-app li:has(a[href*=\"sponsorship.html\"]){display:none!important}";
  document.head.appendChild(css);
})();
