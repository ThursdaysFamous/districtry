  /* =========================================================================
   * GEOLOCATION — "Use my current location" via the browser's
   * navigator.geolocation API. Feeds straight into the same
   * setSelectedPoint() the map click and address search use, so a
   * geolocated point behaves identically everywhere downstream.
   * ======================================================================= */

  var locateBtn = document.getElementById("locate-btn");
  var geoStatusEl = document.getElementById("geo-status");

  function selectLocated(lat, lng) {
    geoStatusEl.textContent = "";
    geoResultsList.innerHTML = "";
    // rounded to ~1km (2 decimal places) — coarse enough that it can't
    // pin an individual's address, fine enough to see which part of the
    // city geolocation gets used from. Path stays fixed so GoatCounter
    // groups every hit under one "geolocate-success" event; the
    // approximate area only shows up in the per-hit title.
    trackEvent("geolocate-success", lat.toFixed(2) + "," + lng.toFixed(2));
    setSelectedPoint(lat, lng);
  }

  locateBtn.addEventListener("click", function () {
    trackEvent("geolocate");
    if (!("geolocation" in navigator)) {
      geoStatusEl.textContent = "Geolocation isn't supported in this browser.";
      geoStatusEl.className = "geo-status err";
      return;
    }

    locateBtn.disabled = true;
    geoStatusEl.textContent = "Locating…";
    geoStatusEl.className = "geo-status";

    navigator.geolocation.getCurrentPosition(
      function (pos) {
        locateBtn.disabled = false;
        var lat = pos.coords.latitude, lng = pos.coords.longitude;
        // Which app answers where the reader is standing is the hand-off's own
        // question (ENGINE metro-portal placeOwner), asked of the same outlines,
        // so a located point and a clicked one can never be routed differently.
        geoStatusEl.textContent = "Locating…";
        placeOwner(lat, lng).then(function (owner) {
          if (owner === "here") { selectLocated(lat, lng); return; }
          if (owner) {
            // standing inside a sibling's area: send them there, deep-linking
            // the located point (the URL fragment stays client-side, same
            // exposure as the sibling's own geolocate → permalink flow)
            geoStatusEl.textContent = "You're outside " + METRO_NAME + " — but " +
              siblingDisplayName(owner) + " covers where you are.";
            geoStatusEl.className = "geo-status";
            offerMetroPortal(owner, { point: { lat: lat, lng: lng }, source: "geolocate" });
            return;
          }
          geoStatusEl.textContent = "Your current location looks like it's outside the " + METRO_NAME + " area this app covers.";
          geoStatusEl.className = "geo-status err";
        });
      },
      function (err) {
        locateBtn.disabled = false;
        var msg = "Couldn't get your location.";
        if (err.code === err.PERMISSION_DENIED) msg = "Location access was denied. Allow it in your browser's site settings to use this.";
        else if (err.code === err.TIMEOUT) msg = "Location request timed out. Try again.";
        else if (err.code === err.POSITION_UNAVAILABLE) msg = "Your location is currently unavailable.";
        geoStatusEl.textContent = msg;
        geoStatusEl.className = "geo-status err";
      },
      { enableHighAccuracy: true, timeout: 10000, maximumAge: 60000 }
    );
  });
