  /* ---------- out-of-scope wash (coverage indicator) ----------
   * The app only fully covers points inside its metro's coverage geometry,
   * but the map pans well past it. So coverage is visible before a wasted
   * click, everything outside gets a subtle gray wash: one world-sized
   * polygon with the coverage outline's rings cut out as holes (Leaflet
   * paths fill even-odd, so the holes read through). Regional layers
   * (county / state / federal) may still resolve under the wash — it marks
   * where the app's full coverage ends; it is not a no-data claim. The wash
   * draws in its own pane between the tile and overlay panes so every
   * district overlay, highlight and marker stays on top, and pointer-events
   * stay off so clicks fall through to the map.
   *
   * drawOutOfScopeMask(loadCoverageGeometry, loadRegionGeometry) is called
   * once from BOOT (fork code): each metro passes an existing cached GeoJSON
   * loader whose features tile the metro exactly, with shared edges
   * topology-preserved — coincident coordinates are what lets the dissolve
   * below cancel interior edges (the playbook's §4 offline anchors qualify).
   * If the fetch fails the wash is skipped silently — it is a hint, not data,
   * and the owning layer's own error/Retry surfaces the failure.
   *
   * THE WASH IS UP TO THREE BANDS, and which ones exist is a fact about the
   * instance rather than a setting. Everywhere outside the coverage geometry
   * is washed; an instance whose full coverage is a proper SUBSET of a wider
   * region its statewide layers still answer in passes that region's geometry
   * as the second argument, and the strip between the two is washed
   * differently — "you still get something here" rather than "nothing here".
   * Illinois is the only such instance today: its county layers cover 89
   * counties while county/township/municipality/school-district/ZIP answer
   * anywhere in the state. Wisconsin's coverage IS its state and San
   * Francisco's and New York City's have no wider region in play, so all
   * three pass one argument and get two bands.
   *
   * AND THE WASH GETS A KEY. It is the app's only claim about where it stops
   * answering, and until the key shipped it made that claim in a colour with
   * nothing to read it against. buildCoverageKey() below names the bands that
   * were ACTUALLY PAINTED — never the ones an instance meant to paint — so a
   * region ring that is declared but fails to load degrades to a two-band
   * wash AND a two-band key together, instead of a key describing a strip
   * that is not on the map. The words come from COVERAGE_KEY in the METRO
   * config (read through a typeof guard: an instance that has not opted in
   * draws the wash and builds no key, exactly as before the key existed). */
  var scopeMaskPane = map.createPane("scope-mask");
  scopeMaskPane.style.zIndex = "350"; // tile pane is 200, overlay pane 400
  scopeMaskPane.style.pointerEvents = "none";

  /* Dissolve the coverage tiling into its outline: a ring segment traversed
   * by two features is an interior border (each neighbor walks it once) and
   * is dropped; the surviving segments are chained back into closed rings.
   * Rendering the outline instead of the raw per-feature rings matters
   * because coincident interior edges do NOT cancel cleanly under
   * antialiased even-odd fill — they leave hairline seams tracing a district
   * layer that isn't even toggled on. Interior holes in the tiling (e.g. an
   * enclave municipality) survive as rings and correctly stay washed.
   * Returns null when a chain fails to close (malformed tiling). */
  function coverageOutlineRings(feats) {
    var counts = {}; // endpoint-sorted "x,y|x,y" -> traversal count
    var segPts = {}; // same key -> [a, b] as [lng,lat] pairs
    function ptKey(pt) { return pt[0] + "," + pt[1]; }
    function eachSeg(ring) {
      for (var i = 0; i < ring.length - 1; i++) {
        var a = ring[i], b = ring[i + 1];
        if (a[0] === b[0] && a[1] === b[1]) continue; // zero-length
        var ka = ptKey(a), kb = ptKey(b);
        var k = ka < kb ? ka + "|" + kb : kb + "|" + ka;
        counts[k] = (counts[k] || 0) + 1;
        segPts[k] = [a, b];
      }
    }
    for (var f = 0; f < feats.length; f++) {
      var geom = feats[f].geometry;
      if (!geom) continue;
      var polys = geom.type === "Polygon" ? [geom.coordinates]
                : geom.type === "MultiPolygon" ? geom.coordinates : [];
      for (var p = 0; p < polys.length; p++) {
        for (var r = 0; r < polys[p].length; r++) eachSeg(polys[p][r]);
      }
    }
    var adj = {}; // "x,y" -> [{key, pt: far endpoint}] over surviving segments
    var kept = 0;
    for (var k in counts) {
      if (counts[k] !== 1) continue;
      kept++;
      var a2 = segPts[k][0], b2 = segPts[k][1];
      (adj[ptKey(a2)] = adj[ptKey(a2)] || []).push({ key: k, pt: b2 });
      (adj[ptKey(b2)] = adj[ptKey(b2)] || []).push({ key: k, pt: a2 });
    }
    var used = {};
    var rings = [];
    for (var seed in counts) {
      if (counts[seed] !== 1 || used[seed]) continue;
      used[seed] = true;
      var start = segPts[seed][0];
      var startKey = ptKey(start);
      var ring = [start, segPts[seed][1]];
      var curKey = ptKey(segPts[seed][1]);
      var guard = kept + 1; // any valid walk closes well before this
      while (curKey !== startKey && guard-- > 0) {
        var options = adj[curKey] || [];
        var next = null;
        for (var o = 0; o < options.length; o++) {
          if (!used[options[o].key]) { next = options[o]; break; }
        }
        if (!next) return null; // open chain — malformed tiling
        used[next.key] = true;
        ring.push(next.pt);
        curKey = ptKey(next.pt);
      }
      if (curKey !== startKey) return null; // guard tripped
      rings.push(ring);
    }
    return rings;
  }

  // The most recent coverage outline in GeoJSON [lng,lat] ring form, retained
  // so other engine surfaces (the gaps panel) can test "is this point inside
  // the covered area" without a second fetch or their own dissolve. null until
  // the mask loads (or if it never does) — consumers treat null as UNKNOWN and
  // fail safe, never as "outside".
  var coverageMaskRings = null;

  // And the REGION's polygons in the same [lng,lat] form, retained for the same
  // reason and with one difference that matters: regionOuterRing below keeps
  // only the largest outer ring, which is right for drawing one boundary line
  // and wrong for a point test, because a region with a detached part would
  // report points on it as outside itself. So this keeps every polygon and
  // every hole. null until the wash actually PAINTS a region band — an
  // instance that declares no region, and one whose region fails to load, both
  // leave it null, and a consumer reads null as "this instance has no middle
  // band" rather than as "outside the region".
  var coverageRegionPolys = null;

  // The wash's own layers land on districtryCoverageLayers, which is declared
  // with the theme controller that restyles them (ENGINE districtry-theme) —
  // the skin darkens the out-of-coverage band on a dark ground where it greys
  // it on a light one. Declared there rather than here because the controller
  // is what reads it; this is where it gets filled.

  // Convert a GeoJSON ring ([lng,lat] pairs) to Leaflet's [lat,lng].
  function ringToLatLng(ring) {
    var out = [];
    for (var i = 0; i < ring.length; i++) out.push([ring[i][1], ring[i][0]]);
    return out;
  }

  // Every polygon of the region geometry as its own ring list — outer ring
  // first, holes after, which is what pointInGeometry wants for a MultiPolygon.
  // Unlike regionOuterRing this discards nothing: it answers "is this point in
  // the region", where dropping an island or filling a hole is a wrong answer
  // rather than a simplified drawing.
  function regionPolygons(geo) {
    var feats = (geo && geo.features) || [];
    var out = [];
    for (var f = 0; f < feats.length; f++) {
      var geom = feats[f].geometry;
      if (!geom) continue;
      var polys = geom.type === "Polygon" ? [geom.coordinates]
                : geom.type === "MultiPolygon" ? geom.coordinates : [];
      for (var p = 0; p < polys.length; p++) {
        if (polys[p] && polys[p].length) out.push(polys[p]);
      }
    }
    return out.length ? out : null;
  }

  // The region's outer boundary as one Leaflet ring, or null. Only the LARGEST
  // outer ring is taken: a region with offshore islands would otherwise draw a
  // glow around each one, and the band is about where the region's answers
  // stop, not about its coastline inventory.
  function regionOuterRing(geo) {
    var feats = (geo && geo.features) || [];
    var biggest = null;
    for (var f = 0; f < feats.length; f++) {
      var geom = feats[f].geometry;
      if (!geom) continue;
      var polys = geom.type === "Polygon" ? [geom.coordinates]
                : geom.type === "MultiPolygon" ? geom.coordinates : [];
      for (var p = 0; p < polys.length; p++) {
        var outer = polys[p][0];
        if (outer && outer.length > 3 && (!biggest || outer.length > biggest.length)) {
          biggest = outer;
        }
      }
    }
    return biggest ? ringToLatLng(biggest) : null;
  }

  /* ---------- the coverage key ----------
   * A <details> rather than a <div>, and that is a layout decision rather
   * than a stylistic one: on desktop it is open and reads as a legend card,
   * but below 768px .map-bottom-left becomes a slim static strip under a
   * short map, where an open key would be pinned to the screen for the whole
   * page. Closed it is a ~32px chip and the bands survive one tap away.
   *
   * Built with textContent throughout — the labels are our own config rather
   * than scraped text, but the app renders every string that way and a key is
   * not the place to make an exception.
   *
   * `bands` is what the draw actually painted, in reading order from the
   * inside out. Only the LAST one ("outside") is guaranteed. */
  function buildCoverageKey(bands) {
    var host = document.querySelector(".map-bottom-left");
    if (!host || !bands.length || document.querySelector(".districtry-map-legend")) return;

    var key = document.createElement("details");
    key.className = "districtry-map-legend";

    var summary = document.createElement("summary");
    summary.className = "dml-kicker";
    summary.appendChild(document.createTextNode("Coverage"));

    // The collapsed chip: the same swatches with no words, so a closed key
    // still says "this is about those colours on the map".
    var mini = document.createElement("span");
    mini.className = "dml-mini";
    mini.setAttribute("aria-hidden", "true");
    var items = document.createElement("div");
    items.className = "dml-items";

    for (var i = 0; i < bands.length; i++) {
      var band = bands[i];
      // A rule before the outside band, once there is something above it to
      // separate: inside the region and beyond it are different claims.
      if (band.rule) {
        var rule = document.createElement("span");
        rule.className = "dml-rule";
        items.appendChild(rule);
      }
      var item = document.createElement("span");
      item.className = "dml-item";
      var sw = document.createElement("span");
      sw.className = band.swatch;
      item.appendChild(sw);
      var label = document.createElement("span");
      label.className = "dml-label";
      label.textContent = band.label;
      item.appendChild(label);
      if (band.sub) {
        var sub = document.createElement("span");
        sub.className = "dml-sub";
        sub.textContent = band.sub;
        item.appendChild(sub);
      }
      items.appendChild(item);
      var chip = document.createElement("span");
      chip.className = band.swatch;
      mini.appendChild(chip);
    }

    summary.appendChild(mini);
    key.appendChild(summary);
    key.appendChild(items);

    /* Open on desktop, closed on a phone — the DEFAULT only. A click on the
       summary is unambiguous intent and, unlike the `toggle` event, fires
       only for the reader; once they have opened or put the key away the
       viewport stops having an opinion, so crossing the breakpoint cannot
       silently reopen it. */
    var phone = window.matchMedia && window.matchMedia("(max-width: 767px)").matches;
    if (!phone) key.setAttribute("open", "");
    summary.addEventListener("click", function () {
      key.setAttribute("data-dst-user-set", "1");
    });

    // First in the bottom-left stack: the key explains the whole map, the
    // controls below it act on one thing each.
    host.insertBefore(key, host.firstChild);
  }

  function drawOutOfScopeMask(loadCoverageGeometry, loadRegionGeometry) {
    if (typeof loadCoverageGeometry !== "function") return;
    // Optional in both senses — an instance may declare no region, and a
    // declared one may fail to load. Either way this resolves null and the
    // wash degrades to two bands rather than failing.
    var regionP = typeof loadRegionGeometry === "function"
      ? loadRegionGeometry().catch(function () { return null; })
      : Promise.resolve(null);
    Promise.all([loadCoverageGeometry(), regionP]).then(function (loaded) {
      var geo = loaded[0], regionGeo = loaded[1];
      var feats = (geo && geo.features) || [];
      if (!feats.length) return;
      var outline = coverageOutlineRings(feats);
      if (!outline || !outline.length) {
        // fall back to the raw rings: hairline seams beat no wash at all
        outline = [];
        for (var i = 0; i < feats.length; i++) {
          var geom = feats[i].geometry;
          if (!geom) continue;
          var polys = geom.type === "Polygon" ? [geom.coordinates]
                    : geom.type === "MultiPolygon" ? geom.coordinates : [];
          for (var p = 0; p < polys.length; p++) {
            for (var r = 0; r < polys[p].length; r++) outline.push(polys[p][r]);
          }
        }
      }
      coverageMaskRings = outline;
      // Outer ring spans several world copies so panning far east/west at
      // low zoom never slips out from under the wash.
      var world = [[-89, -720], [89, -720], [89, 720], [-89, 720]];
      var covLatLng = [];
      for (var j = 0; j < outline.length; j++) covLatLng.push(ringToLatLng(outline[j]));

      var region = regionOuterRing(regionGeo);
      var c = districtryCoverageLayers;
      var ck = typeof COVERAGE_KEY !== "undefined" ? COVERAGE_KEY : null;
      var bands = [];

      /* NOT ONE COLOUR IS NAMED HERE. Every band's fill and stroke comes from
         styleDistrictryCoverage() below, which already owns both themes'
         values and is the only thing that may change them on a toggle. Naming
         the light ones here too would be two copies of one palette, and the
         out-of-coverage band's opacity is not even a constant — it lifts when
         there is no region ring to sit inside, which is a fact this function
         has not finished establishing yet at the point the layers are made.
         So they are built INVISIBLE (fillOpacity/opacity 0) and painted a few
         statements later, in the same synchronous block, before a frame can
         render. The widths and the blur class stay: those are geometry, and
         the skin has no opinion about them. */
      if (region) {
        // Retained for the gaps panel, and set HERE rather than beside the
        // ring above so the panel can only describe a middle band the map
        // actually drew — the same rule buildCoverageKey follows for the key.
        coverageRegionPolys = regionPolygons(regionGeo);
        // Three bands. Leaflet fills even-odd, so each polygon's later rings
        // read as holes: world minus the region is everything outside it, and
        // the region minus the coverage rings is the strip between.
        c.outMask = L.polygon([world, region], {
          pane: "scope-mask", stroke: false, fillOpacity: 0, interactive: false
        }).addTo(map);
        c.pending = L.polygon([region].concat(covLatLng), {
          pane: "scope-mask", stroke: false, fillOpacity: 0, interactive: false
        }).addTo(map);
        // The region edge itself, as a blurred wide line under a hairline —
        // the glow says "this boundary means something" without the weight of
        // a border the app does not own.
        var closed = region.concat([region[0]]);
        c.glow = L.polyline(closed, {
          pane: "scope-mask", className: "dst-glow",
          weight: 9, opacity: 0, interactive: false
        }).addTo(map);
        c.stateLine = L.polyline(closed, {
          pane: "scope-mask", weight: 1.5, opacity: 0.9, interactive: false
        }).addTo(map);
        if (ck && ck.region) {
          bands.push({ swatch: "dml-glow", label: ck.region.edge });
          bands.push({ swatch: "dml-sw dml-pending", label: ck.region.label, sub: ck.region.sub });
        }
      } else {
        c.outMask = L.polygon([world].concat(covLatLng), {
          pane: "scope-mask", stroke: false, fillOpacity: 0, interactive: false
        }).addTo(map);
        // Read by styleDistrictryCoverage: with no region ring to sit inside,
        // one band carries the whole claim and is drawn lighter for it.
        c.fallbackMask = true;
      }
      if (ck) bands.push({ swatch: "dml-sw dml-out", label: ck.outside, rule: bands.length > 0 });

      // Paints every band above, and does it HERE rather than at boot because
      // the wash draws at IDLE: a dark-mode reader would otherwise watch it
      // come up in its light-ground colours and keep them until the next
      // toggle. Runs before the browser can paint the invisible versions.
      styleDistrictryCoverage(districtryTheme() === "dark");
      buildCoverageKey(bands);
    }).catch(function () { /* decorative — skip the wash, never surface an error */ });
  }
