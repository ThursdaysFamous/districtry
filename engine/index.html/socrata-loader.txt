  /* ---------- true if at least one feature has real (non-null) geometry.
   * Some older "legacy" Socrata boundary datasets return HTTP 200 from the
   * primary .geojson route with every feature's geometry stripped to null
   * (confirmed on Boundaries - ZIP Codes / Police Districts) — that's not a
   * fetch failure, so .catch() alone never sees it. ---------- */
  function hasUsableGeometry(geojson) {
    var features = geojson && geojson.features;
    if (!features || !features.length) return false;
    for (var i = 0; i < features.length; i++) {
      if (features[i].geometry) return true;
    }
    return false;
  }

  /* ---------- the primary route's page size, and why a set that fills it is
   * REFUSED rather than drawn.
   *
   * /resource/{id}.geojson returns one page and says nothing about how many
   * rows it left behind, so a set that comes back exactly full is
   * indistinguishable from a complete one. It was 1000 until 2026-09-29, and
   * Chicago's ward-precinct dataset (i8fv-xe4b) has 1,291 rows: the app drew
   * 1,000 of them and, because the portal orders by ward, the 291 it dropped
   * were every precinct of wards 39 to 50 — the whole north side, where a
   * reader who clicked got no precinct once the set had cached.
   *
   * Two changes, and the second is the one that keeps this fixed: the page is
   * now Socrata's own documented maximum, and a response holding exactly that
   * many features fails validation, so the route walk falls through to the
   * export routes, which page the whole dataset themselves. A dataset that
   * outgrows the maximum therefore loads from another route or fails loudly;
   * it never silently loses its tail again. ---------- */
  var SOCRATA_PAGE_MAX = 50000;

  /* ---------- fetch a Socrata dataset as GeoJSON, trying each known route in turn
   * until one returns real geometry. Some datasets 200 with every geometry
   * stripped to null on one route (confirmed on Police Districts /
   * ZIP Codes' /resource/{id}.geojson route) but work fine via another —
   * that's a "succeeded but useless" response, not a fetch failure, so a
   * plain .catch() never sees it. ---------- */
  function socrataRouteUrls(datasetId) {
    return [
      SOCRATA_HOST + "/resource/" + datasetId + ".geojson?$limit=" + SOCRATA_PAGE_MAX,
      SOCRATA_HOST + "/api/v3/views/" + datasetId + "/query.geojson",
      SOCRATA_HOST + "/api/geospatial/" + datasetId + "?method=export&format=GeoJSON"
    ].map(withAppToken);
  }

  /* ---------- usable AND whole: real geometry, and not a full page. Only the
   * first route is paged, so only it can be truncated; the later routes export
   * the whole dataset and a set that happens to hold exactly SOCRATA_PAGE_MAX
   * features would be refused there too, which costs a retry on the next route
   * and never costs a reader an answer. ---------- */
  function hasWholeUsableGeometry(geojson) {
    if (!hasUsableGeometry(geojson)) return false;
    return geojson.features.length < SOCRATA_PAGE_MAX;
  }

  /* ---------- walk a list of fallback routes, returning the first response that
   * passes `validate`. Single catch per route: a fetch failure and a "200 but
   * useless" response both advance to the next route exactly once — the old
   * .then-recursion + .catch-recursion pair re-ran later routes twice, a bug
   * this shared helper keeps fixed for both Socrata callers. Primary route (0)
   * gets 2 retries, fallbacks 1. `label` only names the dataset in the reject. */
  function tryRoutes(urls, validate, label) {
    function tryRoute(i) {
      if (i >= urls.length) return Promise.reject(new Error("no route for " + label + " succeeded"));
      return fetchJSONWithRetry(urls[i], {}, i === 0 ? 2 : 1)
        .then(function (data) {
          if (validate(data)) return data;
          throw new Error("route returned nothing usable");
        })
        .catch(function () { return tryRoute(i + 1); });
    }
    return tryRoute(0);
  }

  function loadSocrataGeoJSON(datasetId) {
    return tryRoutes(socrataRouteUrls(datasetId), hasWholeUsableGeometry, datasetId + " geometry");
  }

  /* ---------- point-first Socrata fetch: only the feature(s) containing one
   * point, via SoQL intersects on the dataset's probed geometry column. A
   * single district's geometry against the full boundary set, so a first
   * click can answer before the whole download lands (see queryFeatureAt).
   * Only the primary /resource/ route takes $where; a legacy dataset that
   * nulls geometry on that route fails hasUsableGeometry here exactly as it
   * does in the route walk, the caller falls back to the full-set load, and
   * the fallback-route machinery keeps owning that failure mode. An empty
   * result rejects too — deliberately: "nothing intersects" from the server
   * is indistinguishable from those null-geometry lies, so the full set
   * (whose routes are validated) stays the only authority for "no district
   * here". ---------- */
  function loadSocrataPointGeoJSON(datasetId, point) {
    return probeGeometryColumn(datasetId).then(function (col) {
      var wkt = "POINT (" + point.lng + " " + point.lat + ")";
      var url = SOCRATA_HOST + "/resource/" + datasetId + ".geojson?$where=" +
        encodeURIComponent("intersects(" + col + ", '" + wkt + "')") + "&$limit=8";
      return fetchJSONWithRetry(withAppToken(url), {}, 1).then(function (geojson) {
        if (!hasUsableGeometry(geojson)) throw new Error("point query for " + datasetId + " returned no usable geometry");
        return geojson;
      });
    });
  }
