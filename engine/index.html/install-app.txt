  /* ==== install as an app: ONE districtry, opened where the reader left off ====
     The apps and the front door link the one manifest at /manifest.webmanifest,
     whose scope is the whole site, so an installed districtry crosses state
     lines without leaving its window, and whose start_url is the launcher at
     /app/. This block is the app's half of that arrangement, in five parts.

     1. WHICH STATE WAS OPEN LAST. Each boot writes this app's tag under
        DST_LAST_APP_KEY, and the launcher reads it to open that state.
        scripts/build_manifests.py --check holds the launcher to this key. On an
        iPhone an installed web app keeps storage apart from Safari's, so its
        first launch has nothing to read and opens the front page; it remembers
        from then on.

     2. THE INVITATION, phones only and once. Android hands the page a
        `beforeinstallprompt` event when Chrome judges the site installable;
        it is held, and an Install button is offered once the reader has an
        answer on screen. Safari has no such event, so an iPhone gets a short
        tip saying where Add to Home Screen is. Never inside an iframe (an
        embedded map is someone else's page), never once installed, and never
        a second time: showing it records DST_INSTALL_INVITE_KEY.

        WHILE THE ANDROID APP IS IN GOOGLE PLAY'S CLOSED TEST (Adam,
        2026-10-20), an Android phone is invited to the test instead of being
        offered the website version to install: Google Play lets a new
        personal account apply to publish its app only after 12 testers have
        stayed in its closed test for 14 days, and the steps page at
        DST_ANDROID_TEST_URL asks testers to remove the website copy. Chrome's
        install event is still held, so its own mini-infobar does not offer
        that copy either, and Chrome's menu still does. The test is offered
        once too, recorded as DST_TEST_INVITE_SEEN under the same key, so a
        reader who was offered the install before the test began is asked
        once more, about the test. Closing the bar or following its link
        records DST_TEST_INVITE_ANSWERED instead, and the front page's own
        invitation to the test reads and writes that word under the same key
        (scripts/build_landing_page.py takes both names from this block), so
        a reader who closed or followed either invitation is not shown the
        other; only seeing this bar leaves the front page's in place (Adam,
        2026-10-20: a "no" stays closed). Not in the Play app, whose window
        carries the mark from engine/shared/play-app.txt. When the app is
        published, set DST_ANDROID_TEST_URL to null and the install offer
        returns; the front page's generator then fails until its notice
        changes too.

     3. THE COUNTS, by state and never by place: `app-install/<tag>` when the
        browser reports an install, `app-launch/<tag>` once per launch of the
        installed app (DST_APP_LAUNCH_KEY in session storage marks the window
        as counted), `install-invite/<what>` for the invitation: shown,
        accepted, declined or closed, tip-shown or tip-closed on an iPhone, and
        test-shown, test-join or test-closed for the Android test,
        `share-in/<tag>` when a place shared into the installed app opened
        this app, and `app-shortcut/<tag>/<action>` when one of the installed
        app's long-press shortcuts did (both below). A launch is counted at
        boot, before GoatCounter's script has always arrived, so these wait for
        it rather than being lost.

     4. A PLACE SHARED INTO THE APP. The manifest's share_target puts the
        installed districtry in Android's Share menu and sends what was shared
        to the launcher at /app/, which hands it here as #q= (the hash, which
        never leaves the browser; the permalink block searches it) and leaves
        DST_SHARE_IN_KEY in session storage. It is counted once and removed, so
        the next page in the window does not count it again. The share arrives
        in the address of a request for /app/, so the launcher's own worker
        must answer it from its copy rather than the host: that worker is
        registered here when districtry is installed and whenever the
        installed app runs, so it is in place before the first share even if
        the installed app has never been opened from its icon.
        scripts/build_manifests.py --check holds the launcher to this key and
        this block to the launcher's worker.

     5. A LONG-PRESS SHORTCUT. The manifest's shortcuts ("Use my location",
        "Report a problem") open the launcher with their action in the hash,
        and the launcher hands the action on to the state it opens, where the
        geolocation and feedback blocks do it, and leaves the action under
        DST_SHORTCUT_KEY in session storage. It is counted once, by state and
        action and never by place, and removed. scripts/build_manifests.py
        --check holds the launcher to this key. */
  var DST_LAST_APP_KEY = "districtry-last-app";
  var DST_INSTALL_INVITE_KEY = "districtry-install-invite";
  var DST_APP_LAUNCH_KEY = "districtry-app-launch";
  var DST_SHARE_IN_KEY = "districtry-share-in";
  var DST_SHORTCUT_KEY = "districtry-shortcut";
  var DST_INVITE_DELAY_MS = 3000;
  var DST_ANDROID_TEST_URL = "/android.html?ref=map";
  var DST_TEST_INVITE_SEEN = "test-shown";
  var DST_TEST_INVITE_ANSWERED = "test-answered";
  var dstFramed = (function () {
    try { return window.top !== window.self; } catch (e) { return true; }
  })();
  function dstCountWhenReady(name, tries) {
    if (window.goatcounter && typeof window.goatcounter.count === "function") {
      trackEvent(name);
      return;
    }
    // up to ten seconds; a blocked or failed counter stays a no-op
    if ((tries || 0) < 40) setTimeout(function () { dstCountWhenReady(name, (tries || 0) + 1); }, 250);
  }
  /* Every invitation event goes through here, so the name the privacy page
     publishes is the one family `install-invite/` and the outcomes are its
     suffixes. */
  function dstInviteEvent(what) {
    dstCountWhenReady("install-invite/" + what);
  }
  function dstInstalledLaunch() {
    return !!((window.matchMedia && window.matchMedia("(display-mode: standalone)").matches) ||
      window.navigator.standalone === true);
  }
  /* The launcher's worker (app/sw.js, scope /app/), so a place shared into
     the installed app is answered from the launcher's copy on the phone. */
  function dstRegisterLauncher() {
    if (dstFramed || !("serviceWorker" in window.navigator)) return;
    try {
      window.navigator.serviceWorker.register("/app/sw.js", { scope: "/app/" })["catch"](function () {});
    } catch (e) { /* the launcher registers it itself on its first launch */ }
  }
  function dstInviteSeen() {
    try { return !!window.localStorage.getItem(DST_INSTALL_INVITE_KEY); } catch (e) { return true; }
  }
  function dstMarkInviteSeen(value) {
    try { window.localStorage.setItem(DST_INSTALL_INVITE_KEY, value); } catch (e) { /* private mode: it may show again */ }
  }
  function dstTestInviteSeen() {
    try {
      var seen = window.localStorage.getItem(DST_INSTALL_INVITE_KEY);
      return seen === DST_TEST_INVITE_SEEN || seen === DST_TEST_INVITE_ANSWERED;
    } catch (e) { return true; }
  }
  /* A browser on Android while the test runs, outside the Play app. */
  function dstOnAndroidTest() {
    return !!DST_ANDROID_TEST_URL &&
      /\bAndroid\b/.test(window.navigator.userAgent || "") &&
      !document.documentElement.classList.contains("dst-play-app");
  }
  /* An iPhone's own Safari. Other iOS browsers and in-app views name
     themselves in the user agent and put Share somewhere else, so the tip's
     directions would be wrong there; an iPad reports itself as a Mac and is
     not a phone. */
  function dstIsIphoneSafari() {
    var ua = window.navigator.userAgent || "";
    if (!/\b(iPhone|iPod)\b/.test(ua)) return false;
    if (/CriOS|FxiOS|EdgiOS|OPiOS|GSA\/|FBAN|FBAV|Instagram|Line\//.test(ua)) return false;
    return /Version\/[\d.]+.*Safari\//.test(ua);
  }
  /* The invitation waits for the reader's first answer: a point selected,
     then a moment to read the cards. */
  function dstAfterFirstAnswer(fn) {
    var poll = setInterval(function () {
      if (!state.selectedPoint) return;
      clearInterval(poll);
      setTimeout(fn, DST_INVITE_DELAY_MS);
    }, 500);
  }
  var dstInviteBar = null;
  function dstHideInvite() {
    if (dstInviteBar && dstInviteBar.parentNode) dstInviteBar.parentNode.removeChild(dstInviteBar);
    dstInviteBar = null;
  }
  /* Built through the DOM with textContent, the way every card is. `go` is
     the bar's one action, a button (onClick) or a link (href), or null for
     the iPhone's tip; `closedAs` is what closing it is counted as, and
     `onClose`, when given, runs after. */
  function dstShowInvite(label, text, go, closedAs, onClose) {
    if (dstInviteBar || dstInstalledLaunch() || !dstIsPhone()) return false;
    var bar = document.createElement("aside");
    bar.className = "install-invite";
    bar.setAttribute("aria-label", label);
    var p = document.createElement("p");
    p.className = "install-invite-text";
    p.textContent = text;
    bar.appendChild(p);
    if (go) {
      var action = document.createElement(go.href ? "a" : "button");
      if (go.href) action.href = go.href;
      else action.type = "button";
      action.className = "install-invite-go";
      action.textContent = go.text;
      action.addEventListener("click", go.onClick);
      bar.appendChild(action);
    }
    var close = document.createElement("button");
    close.type = "button";
    close.className = "install-invite-close";
    close.setAttribute("aria-label", go ? "Not now" : "Close");
    close.textContent = "✕";
    close.addEventListener("click", function () {
      dstHideInvite();
      dstInviteEvent(closedAs);
      if (onClose) onClose();
    });
    bar.appendChild(close);
    document.body.appendChild(bar);
    dstInviteBar = bar;
    return true;
  }

  if (!dstFramed) {
    try { window.localStorage.setItem(DST_LAST_APP_KEY, METRO_BRAND.instanceTag); } catch (e) { /* the launcher then opens the front page */ }
  }

  if (dstInstalledLaunch()) {
    var dstLaunchCounted = false;
    try { dstLaunchCounted = !!window.sessionStorage.getItem(DST_APP_LAUNCH_KEY); } catch (e) { dstLaunchCounted = false; }
    if (!dstLaunchCounted) {
      try { window.sessionStorage.setItem(DST_APP_LAUNCH_KEY, "1"); } catch (e) { /* counted again on the next page in this window */ }
      dstCountWhenReady("app-launch/" + METRO_BRAND.instanceTag);
    }
    dstRegisterLauncher();
  }

  if (!dstFramed) {
    var dstSharedIn = false;
    try {
      dstSharedIn = !!window.sessionStorage.getItem(DST_SHARE_IN_KEY);
      if (dstSharedIn) window.sessionStorage.removeItem(DST_SHARE_IN_KEY);
    } catch (e) { dstSharedIn = false; }
    if (dstSharedIn) dstCountWhenReady("share-in/" + METRO_BRAND.instanceTag);
    var dstShortcut = "";
    try {
      dstShortcut = window.sessionStorage.getItem(DST_SHORTCUT_KEY) || "";
      if (dstShortcut) window.sessionStorage.removeItem(DST_SHORTCUT_KEY);
    } catch (e) { dstShortcut = ""; }
    // A bare word, as the launcher writes it, so the count cannot carry
    // anything else that reached session storage under this key.
    if (/^[a-z]{1,16}$/.test(dstShortcut)) {
      dstCountWhenReady("app-shortcut/" + METRO_BRAND.instanceTag + "/" + dstShortcut);
    }
  }

  window.addEventListener("appinstalled", function () {
    dstHideInvite();
    dstCountWhenReady("app-install/" + METRO_BRAND.instanceTag);
    dstRegisterLauncher();
  });

  var dstHeldPrompt = null;
  var dstInviteQueued = false;
  window.addEventListener("beforeinstallprompt", function (e) {
    /* On a computer, or inside someone else's page, the browser's own
       install control is left exactly as it is. */
    if (dstFramed || !dstIsPhone()) return;
    e.preventDefault();
    dstHeldPrompt = e;
    if (dstInviteQueued || dstInviteSeen() || dstOnAndroidTest()) return;
    dstInviteQueued = true;
    dstAfterFirstAnswer(function () {
      if (!dstHeldPrompt || dstInviteSeen()) return;
      var shown = dstShowInvite("Add districtry to your home screen",
        "Add districtry to your home screen. It opens to the state you used last.", {
          text: "Install",
          onClick: function () {
            var held = dstHeldPrompt;
            dstHeldPrompt = null;
            dstHideInvite();
            if (!held) return;
            held.prompt();
            held.userChoice.then(function (choice) {
              dstInviteEvent(choice && choice.outcome === "accepted" ? "accepted" : "declined");
            }, function () {});
          }
        }, "closed");
      if (shown) {
        dstMarkInviteSeen("shown");
        dstInviteEvent("shown");
      }
    });
  });

  if (!dstFramed && !dstInstalledLaunch() && dstIsIphoneSafari() && !dstInviteSeen()) {
    dstAfterFirstAnswer(function () {
      if (dstInviteSeen()) return;
      var shown = dstShowInvite("Add districtry to your home screen",
        "Add districtry to your Home Screen: tap Share (in the ⋯ menu if you don’t see it), then Add to Home Screen.", null, "tip-closed");
      if (shown) {
        dstMarkInviteSeen("tip-shown");
        dstInviteEvent("tip-shown");
      }
    });
  }

  if (!dstFramed && !dstInstalledLaunch() && dstOnAndroidTest() && !dstTestInviteSeen()) {
    dstAfterFirstAnswer(function () {
      if (dstTestInviteSeen()) return;
      var shown = dstShowInvite("Help test districtry’s Android app",
        "Help test districtry’s Android app. Google Play needs 12 testers.", {
          text: "How to join",
          href: DST_ANDROID_TEST_URL,
          onClick: function () {
            dstMarkInviteSeen(DST_TEST_INVITE_ANSWERED);
            dstInviteEvent("test-join");
            /* after the link is followed, so a page brought back by the
               Back button does not ask again */
            setTimeout(dstHideInvite, 0);
          }
        }, "test-closed", function () { dstMarkInviteSeen(DST_TEST_INVITE_ANSWERED); });
      if (shown) {
        dstMarkInviteSeen(DST_TEST_INVITE_SEEN);
        dstInviteEvent("test-shown");
      }
    });
  }
