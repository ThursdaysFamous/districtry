  function officeFieldHtml(lines) {
    if (!lines || !lines.length) return null;
    return lines.map(sanitize).join("<br>");
  }

  // Extract a geocodable street address from an office's address lines (an array
  // of strings, exactly as scraped): the FIRST street address in the block, its
  // street, any building or suite lines and its city/ZIP line. Used by
  // registerIlgaChamber's pointOfInterest when a roster supplies office lines.
  //
  // A LINE IS DROPPED FOR WHAT IT IS, NEVER FOR A FRAGMENT IT CONTAINS — the
  // splitOfficeLines lesson, one block over (2026-10-20). Until that day this
  // dropped every line matching /phone|fax|tel|\(\d{3}\)|\d{3}[-.\s]\d{4}/i,
  // and both halves misfired on shipped offices. A ZIP+4 satisfies the last
  // alternative ("Fort Smith, AR 72903-5083" contains "903-5083"), so the city
  // line went and the pin was looked up as "6101 Phoenix Ave, Suite 4" with no
  // city or state; and "tel" sits inside street names ("2400 E. Katella Ave.",
  // "1 Kattelville Road"), so the whole address went, nothing was left, and the
  // fallback looked up every line, "Phone: 714-939-8469" included. Measured over
  // the 693 district offices the fleet's rosters shipped that day: 6 U.S. House
  // offices lost their city (AR-3, IN-4, MN-8, TN-2, TN-6, WI-1) and 2 assembly
  // offices were looked up with their telephone numbers (California 68, New
  // York 121). Those 8 are the only shipped addresses this rule changes. It also
  // reads the Louisiana legislature's blocks, which put a physical and a mailing
  // address under headings, sometimes the post office box first, and up to
  // three offices in one block, all of which the old filter joined into one.
  //
  // So a contact line is one LABELLED as a phone, fax or e-mail, or one carrying
  // a whole telephone number (three, three and four digits, which a ZIP+4 is
  // not). A line ending in a ZIP code ends an address when the ZIP follows a
  // two-letter word (the state), a comma, or nothing else on its line, so
  // "P.O. Box 91245" and "Suite 14800" are not ZIP codes; a heading line
  // ("Mailing Address:") ends one too, ZIP or not. The first address that is not
  // only a post office box wins, else the first. Lines after the last address
  // are a note ("*Not ADA accessible…") and count only when they are all there
  // is, so a block with no ZIP and no heading is joined as it always was, and a
  // block with no address line at all gets no pin rather than a lookup of a
  // telephone number.
  function officeAddressForGeocode(lines) {
    if (!lines || !lines.length) return null;
    var CONTACT = /^\s*(?:phone|telephone|tel|fax|cell|mobile|e-?mail)\s*[:.]/i;
    var NUMBER = /(?:\(\d{3}\)|\b\d{3})[\s.\-]*\d{3}[\s.\-]\d{4}\b/;
    var ENDS_IN_ZIP = /(?:^\s*|\b[a-z]{2}\s+|,\s*)\d{5}(?:-\d{4})?[\s.]*$/i;
    var BOX = /^\s*\(?\s*(?:p\.?\s*o\.?|post\s+office)\s*box\b/i;
    var addresses = [];
    var current = [];
    lines.forEach(function (raw) {
      var line = String(raw);
      if (!line.trim() || CONTACT.test(line) || NUMBER.test(line)) return;
      var heading = /:\s*$/.test(line);
      if (!heading) current.push(line);
      if ((heading || ENDS_IN_ZIP.test(line)) && current.length) {
        addresses.push(current);
        current = [];
      }
    });
    if (!addresses.length && current.length) addresses.push(current);
    // only a box: every line but the city line ("P.O. Box 660", "Harrisonburg, LA 71340")
    function boxOnly(address) {
      var head = address.length > 1 ? address.slice(0, -1) : address;
      return head.every(function (l) { return BOX.test(l); });
    }
    var picked = addresses.filter(function (a) { return !boxOnly(a); })[0] || addresses[0];
    if (!picked) return null;
    var addr = picked.join(", ").trim();
    return addr || null;
  }
