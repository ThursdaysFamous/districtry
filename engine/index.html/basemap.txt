  /* ---------- basemap: CARTO vector (MapLibre GL), raster fallback ----------
   * CARTO began watermarking its keyless RASTER tiles "API KEY REQUIRED" in
   * 2026-08; its VECTOR basemaps are the service CARTO itself recommends and
   * are key-free. The map stays a Leaflet map — every layer module, overlay
   * and interaction is Leaflet — and only the basemap is a MapLibre GL
   * canvas, bridged by the vendored maplibre-gl-leaflet plugin (the GL map
   * is non-interactive, in the tile pane, driven by Leaflet). The raster URL
   * survives as the FALLBACK for a browser without WebGL2 (MapLibre GL 5's
   * floor) or a failed GL script load — and with the key below it is no
   * longer watermarked, so the fallback answers civic questions on the same
   * cartography as everyone else. Both paths share one identity — baseTiles
   * is whichever layer is live, baseIsVector which kind — and the theme
   * controller flips styles through the same pair of URL builders. */
  var BASE_ATTRIBUTION = '&copy; <a href="https://www.openstreetmap.org/copyright" target="_blank" rel="noopener">OpenStreetMap</a> contributors &copy; <a href="https://carto.com/attributions" target="_blank" rel="noopener">CARTO</a>';
  // CARTO Basemaps API key, issued 2026-08-26 for districtry.com (the whole
  // fleet is one site, so one key). Client-side by design — CARTO's free tier
  // is enforced by domain + quota (5M tile requests/month across raster and
  // vector), not key secrecy, in exchange for the visible OSM + CARTO
  // attribution above. It de-watermarks the raster fallback today; the
  // vector service does not require it yet, but carries it so the announced
  // extension of the requirement is a change nobody notices.
  var BASE_CARTO_KEY = "cb1_269e_1_18eecdd74aaa4bac77940fa4";
  // kind stays the raster pair ("light_all" | "dark_all") — it is the theme
  // key districtryTileKind already tracks, and the vector styles are the same
  // cartography under CARTO's GL names (Positron / Dark Matter).
  function baseStyleUrl(kind) {
    return "https://basemaps.cartocdn.com/gl/" +
      (kind === "dark_all" ? "dark-matter" : "positron") + "-gl-style/style.json" +
      "?key=" + BASE_CARTO_KEY;
  }
  function baseRasterUrl(kind) {
    return "https://{s}.basemaps.cartocdn.com/" + kind + "/{z}/{x}/{y}{r}.png?key=" + BASE_CARTO_KEY;
  }
  function webgl2Available() {
    // MapLibre GL 5 requires WebGL2. Probe BEFORE constructing: the plugin
    // creates its GL map inside onAdd, and Leaflet attaches a layer's
    // getEvents() handlers before onAdd runs — a constructor throw there
    // would leave move handlers attached that dereference a GL map that
    // never existed, breaking every later pan.
    try {
      var probe = document.createElement("canvas");
      return !!(window.WebGL2RenderingContext && probe.getContext("webgl2"));
    } catch (e) { return false; }
  }
  /* ---------- the basemap's LABELS draw above the districts ----------
   * One GL map in the tile pane drew everything, so every street name sat
   * UNDER the district fills and the selection highlight, and faded with
   * each layer switched on. The style is now split in two: the ground (land,
   * water, roads — every non-symbol layer) stays in the tile pane, and every
   * symbol layer (street, place and water names, house numbers) draws in a
   * second, transparent GL map in its own pane at 440 — above the overlays
   * (400), below the district-name chips (rel-labels, 450) and the markers
   * (600) — so no district colour covers a street name, and no chip, pin or
   * popup is covered by one. ALL the symbols move, not only street names,
   * because MapLibre places labels against the others in the same map, and
   * split across two maps a street name would print over a place name.
   *
   * Street names are also restyled, both themes: CARTO's Positron paints
   * them #838383 at 9-12px, and Dark Matter's major-road names are #383838
   * on #111 — measured 2026-09-25, 1.4:1, all but invisible. They now use
   * the style's own medium weight, a point larger, a dark (light: #3a3f4b)
   * or light (dark: #e6e4ee) ink on a 1.5px halo, and the two smaller road
   * classes are named one zoom level sooner (side streets from 15, secondary
   * roads from 14).
   *
   * The cost is a second GL context and a second pass over the same vector
   * tiles; a failure to build the label map is not a failure of the basemap
   * — the ground map then keeps its own labels, exactly as before. */
  var BASE_LABEL_PANE = "dstBaseLabels";
  var BASE_EMPTY_STYLE = { version: 8, sources: {}, layers: [] };
  var ROADNAME_SOONER = { roadname_minor: 1, roadname_sec: 1 };
  function baseLabelSize(size) {
    if (typeof size === "number") return size + 1;
    if (size && size.stops) {
      return { base: size.base, stops: size.stops.map(function (s) { return [s[0], s[1] + 1]; }) };
    }
    return size;
  }
  // part "ground" keeps every non-symbol layer; "labels" keeps every symbol
  // layer and restyles the street names; "all" keeps both (no label map).
  function splitBaseStyle(style, part, dark) {
    var out = {};
    for (var k in style) if (Object.prototype.hasOwnProperty.call(style, k)) out[k] = style[k];
    out.layers = [];
    (style.layers || []).forEach(function (layer) {
      var isLabel = layer.type === "symbol";
      if (part === "ground" && isLabel) return;
      if (part === "labels" && !isLabel) return;
      if (part === "labels" && /^roadname_/.test(layer.id)) {
        var l = JSON.parse(JSON.stringify(layer));
        if (ROADNAME_SOONER[l.id] && typeof l.minzoom === "number") l.minzoom -= ROADNAME_SOONER[l.id];
        l.layout = l.layout || {};
        if (l.layout["text-size"] !== undefined) l.layout["text-size"] = baseLabelSize(l.layout["text-size"]);
        if (l.layout["text-font"] && l.layout["text-font"].length) {
          l.layout["text-font"] = ["Montserrat Medium"].concat(l.layout["text-font"].slice(1));
        }
        l.paint = l.paint || {};
        l.paint["text-color"] = dark ? "#e6e4ee" : "#3a3f4b";
        l.paint["text-halo-color"] = dark ? "#111111" : "#ffffff";
        l.paint["text-halo-width"] = 1.5;
        out.layers.push(l);
        return;
      }
      out.layers.push(layer);
    });
    return out;
  }
  var baseLabels = null;
  // Both maps follow the theme through here, at boot and on every flip.
  function setBaseStyles(kind) {
    var dark = kind === "dark_all";
    var groundPart = baseLabels ? "ground" : "all";
    baseTiles.getMaplibreMap().setStyle(baseStyleUrl(kind), {
      transformStyle: function (prev, next) { return splitBaseStyle(next, groundPart, dark); }
    });
    if (baseLabels) {
      baseLabels.getMaplibreMap().setStyle(baseStyleUrl(kind), {
        transformStyle: function (prev, next) { return splitBaseStyle(next, "labels", dark); }
      });
    }
  }

  var baseIsVector = !!(window.maplibregl && L.maplibreGL && webgl2Available());
  var baseTiles;
  // A GL layer that throws mid-addTo (a WebGL context the probe granted but
  // the real canvas refused) is left half-registered — see the probe
  // comment above. Unwind it by hand.
  function unwindGLLayer(layer) {
    try { map.off(layer.getEvents(), layer); } catch (e2) {}
    try {
      var glContainer = layer.getContainer && layer.getContainer();
      if (glContainer && glContainer.parentNode) glContainer.parentNode.removeChild(glContainer);
    } catch (e2) {}
    try { delete map._layers[L.Util.stamp(layer)]; } catch (e2) {}
    layer._map = null;
  }
  if (baseIsVector) {
    try {
      // the style arrives through setBaseStyles, which splits it; the empty
      // style is only what the canvas holds for the moment before that
      baseTiles = L.maplibreGL({
        style: BASE_EMPTY_STYLE,
        attributionControl: { customAttribution: BASE_ATTRIBUTION }
      });
      baseTiles.addTo(map);
    } catch (e) {
      if (baseTiles) unwindGLLayer(baseTiles);
      baseTiles = null;
      baseIsVector = false;
    }
  }
  if (baseIsVector) {
    try {
      var labelPane = map.createPane(BASE_LABEL_PANE);
      labelPane.style.zIndex = "440";
      labelPane.style.pointerEvents = "none";
      baseLabels = L.maplibreGL({ style: BASE_EMPTY_STYLE, pane: BASE_LABEL_PANE, attributionControl: false });
      baseLabels.addTo(map);
    } catch (e) {
      if (baseLabels) unwindGLLayer(baseLabels);
      baseLabels = null;
    }
    setBaseStyles("light_all");
  }
  if (!baseIsVector) {
    baseTiles = L.tileLayer(baseRasterUrl("light_all"), {
      maxZoom: 19,
      subdomains: "abcd",
      attribution: BASE_ATTRIBUTION
    }).addTo(map);
  }

  /* ---------- base-map tile-failure notice (R6) ----------
   * The app boots and stays usable offline (SW shell + same-origin data/app),
   * but the basemap comes from a CDN — an outage otherwise leaves a silent
   * gray map with no explanation. Failure signals are debounced: only surface
   * the banner after several failures in a short window (a single transient
   * hiccup shouldn't flash it), and hide it the moment the basemap loads
   * again (the CDN recovered). Manual dismiss latches for the session — the
   * user has acknowledged the base map is down. Vector failures arrive as
   * MapLibre "error" events (style, tile, glyph and sprite fetches alike —
   * each one is the basemap failing to arrive), recovery as a "sourcedata"
   * event CARRYING A TILE — a tile that actually arrived. NOT "idle": idle
   * fires when the map merely SETTLES, and a map that settled into every
   * tile having errored would hide the banner it just earned. One vector
   * case skips the debounce entirely: an error while a REQUESTED STYLE has
   * not yet arrived is a single failed style.json, not one tile of many —
   * the whole basemap is down and there will never be four signals. That
   * window is gated on "styledata" (the style document arrived),
   * deliberately NOT on "load": load waits for every initial tile, glyph
   * and sprite besides, so keying on it would hand a lone flaky tile during
   * first paint the full basemap-down banner. And a theme flip re-opens the
   * window (baseStyleRequested below, called by the theme controller before
   * setStyle), because a failed theme style is the boot case over again —
   * one fetch, no second signal, while the canvas quietly keeps the old
   * cartography. Raster failures arrive per-tile as Leaflet
   * "tileerror"/"tileload" exactly as they always did. */
  var tileBanner = document.getElementById("tile-banner");
  var tileErrorCount = 0;
  var tileErrorResetTimer = null;
  var tileBannerDismissed = false;
  var TILE_ERROR_THRESHOLD = 4;

  function baseTileTrouble() {
    if (tileBannerDismissed) return;
    tileErrorCount += 1;
    if (tileErrorResetTimer) clearTimeout(tileErrorResetTimer);
    // if errors stop arriving, let the running count decay so a slow trickle of
    // one-off failures over minutes never accumulates into a false alarm
    tileErrorResetTimer = setTimeout(function () { tileErrorCount = 0; }, 4000);
    if (tileErrorCount >= TILE_ERROR_THRESHOLD && tileBanner) tileBanner.hidden = false;
  }
  function baseTileRecovered() {
    tileErrorCount = 0;
    if (tileBanner && !tileBanner.hidden) tileBanner.hidden = true; // tiles are flowing again
  }
  var glStyleArrived = false;
  function baseStyleRequested() { glStyleArrived = false; }
  if (baseIsVector) {
    var glBase = baseTiles.getMaplibreMap();
    glBase.on("styledata", function () {
      // the empty placeholder style has no layers; only the real one counts
      var st = glBase.getStyle();
      if (st && st.layers && st.layers.length) glStyleArrived = true;
    });
    glBase.on("error", function () {
      if (!glStyleArrived) {
        if (!tileBannerDismissed && tileBanner) tileBanner.hidden = false;
        return;
      }
      baseTileTrouble();
    });
    glBase.on("sourcedata", function (e) { if (e && e.tile) baseTileRecovered(); });
  } else {
    baseTiles.on("tileerror", baseTileTrouble);
    baseTiles.on("tileload", baseTileRecovered);
  }

  var tileBannerDismiss = document.getElementById("tile-banner-dismiss");
  if (tileBannerDismiss) {
    tileBannerDismiss.addEventListener("click", function () {
      tileBannerDismissed = true;
      if (tileBanner) tileBanner.hidden = true;
    });
  }
