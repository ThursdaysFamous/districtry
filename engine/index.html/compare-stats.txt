  /* =========================================================================
   * COMPARISON STATS — the numbers behind the compared outlines. With a
   * district pinned ("★ Comparing"), its card carries a "Stats" button that
   * opens this screen: every district of every other switched-on polygon
   * layer that overlaps the pinned one, grouped by layer, each with
   *   - whether it lies inside the pinned district or crosses its edge;
   *   - how much of IT is inside the pinned district, and how much of the
   *     pinned district it makes up — by people where the app ships Census
   *     block populations (POPULATION_INDEX_URL), and by area everywhere;
   *   - its own card, run on demand when the row is opened.
   * and, per layer, how much of the pinned district no district of that
   * layer covers at all.
   *
   * THE SAME DISTRICTS AS THE OUTLINES. The rows are chosen by the tests the
   * map's outlines use (polygonFullyInside / polygonsOverlap) over the same
   * layers, so the screen never lists a district the map does not outline.
   * Names use the outline labels' wording (relLabelText), which is the card
   * pill's wording where that can be proven.
   *
   * PEOPLE ARE COUNTED, AREA IS MEASURED (engine block geometry-measure).
   * People are the Census 2020 blocks whose internal point falls inside a
   * district, so a map drawn on block lines (congress, the legislature,
   * wards) gets an exact count, and one that is not gets a close one — the
   * footnote says so. The block files are fetched per county, only for the
   * counties the districts touch, and only when this screen opens.
   *
   * A ROW'S CARD RUNS THE LAYER'S OWN QUERY AT A POINT INSIDE THE DISTRICT
   * (the outline label's spot, which lies in the part inside the pinned
   * district), and renders it with the layer's own render — so it reads
   * exactly as a click there would. That query is the same request a click
   * makes, which for a layer that asks its server about a point sends that
   * point; the privacy page says so. It runs only when a reader opens the
   * row, or prints the report.
   *
   * THE PINNED DISTRICT'S OWN CARD heads the report, run by the same query
   * its sidebar card ran at the selected point, so the report names who
   * represents it as well as how it divides.
   *
   * PRINT OR SHARE opens every row, waits for each row's card (up to 15 s),
   * opens every section inside the cards, and calls the browser's print
   * dialog, where a phone also offers to share or save a PDF. While printing,
   * body.stats-printing hides everything but the report and prints the page's
   * address and the date under the pinned district.
   *
   * A LINK REOPENS THE REPORT. While the report is open the permalink carries
   * `stats=1` beside the pin (syncUrlHash), and "Copy link" hands that URL
   * out — on a touch device through the platform's share sheet. A reader
   * who opens it gets the point, the layers and the pin from the permalink,
   * and then the report, once the pinned district and the other switched-on
   * layers' boundaries have loaded — or after 20 s with whatever has loaded,
   * because a layer whose source is down must not keep the report shut. A
   * link whose pin cannot be restored opens no report, the same rule the
   * pin follows. ---------- */
  // the print styles apply only while the report is being printed
  window.addEventListener("afterprint", function () { document.body.classList.remove("stats-printing"); });
  var statsModal = null;
  var statsOpenSeq = 0;
  var statsPopIndex = null;
  var statsPopFiles = {};

  function statsLoadPopulationIndex() {
    if (typeof POPULATION_INDEX_URL === "undefined") return Promise.resolve(null);
    if (!statsPopIndex) {
      statsPopIndex = fetchJSONWithRetry(POPULATION_INDEX_URL, {}, 2)["catch"](function (err) {
        console.warn("population index unavailable; comparing by area only:", err);
        statsPopIndex = null;
        return null;
      });
    }
    return statsPopIndex;
  }
  // The block points of every county whose extent meets `box`, as a list of
  // arrays (one per county file, never concatenated: Cook alone is ~60,000).
  function statsLoadPopulation(box) {
    return statsLoadPopulationIndex().then(function (index) {
      if (!index || !index.counties) return null;
      var base = POPULATION_INDEX_URL.replace(/[^\/]*$/, "");
      var wanted = [];
      Object.keys(index.counties).forEach(function (fips) {
        var c = index.counties[fips], b = c.bbox;
        if (!b || b[0] > box[2] || b[2] < box[0] || b[1] > box[3] || b[3] < box[1]) return;
        if (!statsPopFiles[c.file]) {
          statsPopFiles[c.file] = fetchJSONWithRetry(base + c.file, {}, 2)["catch"](function (err) {
            delete statsPopFiles[c.file];
            throw err;
          });
        }
        wanted.push(statsPopFiles[c.file]);
      });
      return Promise.all(wanted).then(function (docs) {
        return docs.map(function (d) { return d.points || []; });
      });
    });
  }
  function statsPointsIn(geometry, sets) {
    return sets.map(function (pts) { return measurePointsIn(geometry, pts); });
  }
  function statsWeight(geometry, sets) {
    var sum = 0;
    for (var i = 0; i < sets.length; i++) sum += measurePointWeight(geometry, sets[i]);
    return sum;
  }
  function statsSetsWeight(sets) {
    var sum = 0;
    for (var i = 0; i < sets.length; i++) for (var j = 0; j < sets[i].length; j++) sum += sets[i][j][2] || 0;
    return sum;
  }

  // A share as a reader should see it: never "0%" for something that is
  // there, never "100%" for something that is not all of it.
  function statsPct(part, whole) {
    if (!(whole > 0)) return "—";
    var x = part / whole;
    if (x <= 0) return "none";
    if (x >= 1) return "all";
    if (x < 0.01) return "under 1%";
    if (x > 0.99) return "over 99%";
    return Math.round(x * 100) + "%";
  }
  function statsNum(n) {
    return Math.round(n).toString().replace(/\B(?=(\d{3})+(?!\d))/g, ",");
  }
  function statsSqMi(km2) {
    var mi = km2 * 0.386102;
    return (mi >= 10 ? statsNum(mi) : mi >= 1 ? mi.toFixed(1) : mi.toFixed(2)) + " sq mi";
  }

  // The districts the outlines would draw, per layer, with their relation.
  function statsCollect(parentFeature) {
    var groups = [], drawn = 0, capped = false;
    for (var li = 0; li < layers.length && !capped; li++) {
      var mod = layers[li];
      if (mod.id === pinnedParentId || !state.layersOn[mod.id]) continue;
      if (!mod.overlay || !mod.overlay.style) continue;
      var rt = layerRuntime[mod.id];
      if (!rt || !rt.geojson || rt.relevant === false) continue;
      var wrapper = relLabelWrapper(mod), rows = [];
      var feats = rt.geojson.features || [];
      for (var j = 0; j < feats.length; j++) {
        var f = feats[j];
        var inside = polygonFullyInside(f, parentFeature);
        if (!inside && !polygonsOverlap(f, parentFeature)) continue;
        if (drawn >= REL_MAX_OUTLINES) { capped = true; break; }
        drawn++;
        rows.push({ mod: mod, feature: f, inside: inside,
          name: relLabelText(mod, f, wrapper) || (mod.label + " (unnamed district)") });
      }
      if (rows.length) groups.push({ mod: mod, rows: rows });
    }
    // the districts making up most of the compared one first
    var parentGeom = parentFeature.geometry;
    groups.forEach(function (g) {
      g.rows.forEach(function (r) { r.ov = measureOverlap(r.feature.geometry, parentGeom); });
      g.rows.sort(function (a, b) { return b.ov.overlap - a.ov.overlap; });
    });
    return { groups: groups, capped: capped };
  }

  function statsBuildModal() {
    if (statsModal) return statsModal;
    var modal = document.createElement("div");
    modal.className = "feedback-modal stats-modal";
    modal.id = "stats-modal";
    modal.hidden = true;
    modal.setAttribute("role", "dialog");
    modal.setAttribute("aria-modal", "true");
    modal.setAttribute("aria-labelledby", "stats-title");
    var backdrop = cardEl("div", "feedback-backdrop");
    var panel = cardEl("div", "feedback-panel stats-panel");
    panel.setAttribute("role", "document");
    var close = document.createElement("button");
    close.type = "button";
    close.className = "feedback-close";
    close.setAttribute("aria-label", "Close");
    close.textContent = "×";
    var title = cardEl("h2", null, "Comparison stats");
    title.id = "stats-title";
    var body = cardEl("div", "stats-body");
    body.id = "stats-body";
    panel.appendChild(close);
    panel.appendChild(title);
    panel.appendChild(body);
    modal.appendChild(backdrop);
    modal.appendChild(panel);
    document.body.appendChild(modal);
    var lastFocus = null;
    function onKey(e) { if (e.key === "Escape") statsClose(); }
    statsModal = {
      el: modal, body: body, close: close,
      open: function () {
        lastFocus = document.activeElement;
        modal.hidden = false;
        close.focus();
        document.addEventListener("keydown", onKey);
      },
      shut: function () {
        modal.hidden = true;
        document.removeEventListener("keydown", onKey);
        if (lastFocus && lastFocus.focus && document.contains(lastFocus)) lastFocus.focus();
      }
    };
    close.addEventListener("click", statsClose);
    backdrop.addEventListener("click", statsClose);
    return statsModal;
  }
  function statsIsOpen() {
    return !!(statsModal && !statsModal.el.hidden);
  }
  function statsClose() {
    statsOpenSeq++; // anything still computing for the closed screen stops
    if (statsIsOpen()) {
      statsModal.shut();
      syncUrlHash(); // the permalink stops carrying stats=1
    }
  }

  // REOPENING FROM A LINK (`stats=1`). Requested by applyStateFromHash after
  // it restores the pin; tried again every time the outlines refresh, which
  // is every time a layer's query or boundaries arrive.
  var STATS_RESTORE_WAIT_MS = 20000;
  var statsRestore = null;
  function statsRequestRestore() {
    statsRestore = { seq: state.sequence, deadline: Date.now() + STATS_RESTORE_WAIT_MS };
    setTimeout(statsTryRestore, STATS_RESTORE_WAIT_MS + 50);
    statsTryRestore();
  }
  // Every other switched-on polygon layer has its boundaries, is out of
  // coverage here, or has failed to load them (overlayLoaded goes back to
  // false) — so the report opens with every row it will ever have.
  function statsLayersSettled() {
    for (var i = 0; i < layers.length; i++) {
      var mod = layers[i];
      if (mod.id === pinnedParentId || !state.layersOn[mod.id]) continue;
      if (!mod.overlay || !mod.overlay.style) continue;
      var rt = layerRuntime[mod.id];
      if (!rt || rt.relevant === false || rt.geojson) continue;
      if (rt.overlayLoaded) return false; // still downloading
    }
    return true;
  }
  function statsTryRestore() {
    var r = statsRestore;
    if (!r) return;
    // a new point, or the pin let go, before the report could open: the
    // link's report no longer describes what is on screen
    if (r.seq !== state.sequence || !pinnedParentId) { statsRestore = null; return; }
    var late = Date.now() >= r.deadline;
    if (!relationshipParentFeature()) { if (late) statsRestore = null; return; }
    if (!late && !statsLayersSettled()) return;
    statsRestore = null;
    if (!statsIsOpen()) openCompareStats({ fromLink: true });
  }

  // `opts` is { fromLink: true } when a link reopens the report; the Stats
  // button's click passes its event, which carries no such field.
  function openCompareStats(opts) {
    var parentFeature = relationshipParentFeature();
    var parentMod = pinnedParentId ? getLayerById(pinnedParentId) : null;
    if (!parentFeature || !parentMod) return;
    statsRestore = null; // opened, whether by a link or by hand
    // Counted by the PINNED layer, the shape compare/<id> already uses, and
    // never by district: which ward a reader compares says where they live.
    // A report reopened by a shared link is counted apart from one a reader
    // opened, so the count says how often a shared report is followed.
    // Two calls, not a ternary: build_privacy_page.py reads the event names
    // off literal first arguments.
    if (opts && opts.fromLink === true) trackEvent("compare-stats-reopen/" + pinnedParentId);
    else trackEvent("compare-stats/" + pinnedParentId);
    var m = statsBuildModal();
    var token = ++statsOpenSeq, seq = state.sequence;
    var live = function () { return token === statsOpenSeq && seq === state.sequence; };
    var body = m.body;
    body.innerHTML = "";
    var parentName = relLabelText(parentMod, parentFeature, relLabelWrapper(parentMod)) || parentMod.label;
    var head = cardEl("div", "stats-head");
    head.appendChild(cardEl("div", "stats-layer", parentMod.label));
    head.appendChild(cardEl("div", "stats-name", parentName));
    var parentArea = measureArea(parentFeature.geometry);
    var totals = cardEl("div", "stats-totals", statsSqMi(parentArea));
    head.appendChild(totals);
    // the pinned district's own card, from the same query its sidebar card ran
    // at the selected point — so the report names who represents it too
    var parentCard = cardEl("div", "stats-card stats-parent-card");
    head.appendChild(parentCard);
    var parentRun = statsShowCard(parentMod, state.selectedPoint, parentCard, live);
    var actions = cardEl("div", "stats-actions");
    var printBtn = document.createElement("button");
    printBtn.type = "button";
    printBtn.className = "stats-print-btn";
    printBtn.textContent = "Print or share";
    actions.appendChild(printBtn);
    actions.appendChild(statsLinkButton(parentMod, parentName));
    head.appendChild(actions);
    // what a printout carries that the screen does not: where it came from.
    // Written when printing, from the permalink as it is then, so the address
    // on paper reopens this report.
    var printSource = cardEl("p", "stats-print-source");
    head.appendChild(printSource);
    body.appendChild(head);

    var found = statsCollect(parentFeature);
    var cells = []; // {row, peopleEl, area overlap} filled once people load
    if (!found.groups.length) {
      body.appendChild(cardEl("p", "stats-note",
        "Turn on another district layer to see how its districts fall inside this one."));
    }
    found.groups.forEach(function (g) {
      var sec = cardEl("section", "stats-group");
      sec.appendChild(cardEl("h3", "stats-group-title", g.mod.label));
      var sumArea = 0;
      g.rows.forEach(function (r) {
        var ov = r.ov;
        sumArea += ov.overlap;
        var det = document.createElement("details");
        det.className = "stats-row";
        var sum = document.createElement("summary");
        var top = cardEl("span", "stats-row-top");
        top.appendChild(cardEl("span", "stats-row-name", r.name));
        top.appendChild(cardEl("span", "stats-rel " + (r.inside ? "is-inside" : "is-cross"),
          r.inside ? "inside" : "crosses the edge"));
        sum.appendChild(top);
        var people = cardEl("span", "stats-line stats-people");
        sum.appendChild(people);
        sum.appendChild(cardEl("span", "stats-line stats-area",
          "Area: " + statsPct(ov.overlap, ov.area) + " of it is in " + parentName + ", making up " +
          statsPct(ov.overlap, parentArea) + " of " + parentName));
        det.appendChild(sum);
        var card = cardEl("div", "stats-card");
        det.appendChild(card);
        var run = function () {
          if (!card.__ran) card.__ran = statsRunCard(r, parentFeature, card, live);
          return card.__ran;
        };
        det.addEventListener("toggle", function () {
          if (!det.open) return;
          // A reader opening a row, counted once per row by the ROW's layer
          // (which layer's cards readers want), never its district. Printing
          // runs every row before its toggle event fires, so card.__ran is
          // already set and a print counts nothing here.
          if (!card.__ran) trackEvent("compare-stats-row/" + r.mod.id);
          run();
        });
        sec.appendChild(det);
        cells.push({ row: r, el: people, det: det, run: run });
      });
      // What no district of this layer covers. Only stated for a layer whose
      // districts do not overlap each other — summed shares of an overlapping
      // layer would count some ground twice and say nothing true.
      var restArea = parentArea - sumArea;
      g.restEl = cardEl("p", "stats-rest");
      if (sumArea <= parentArea * 1.005) {
        g.restEl.textContent = restArea > parentArea * 0.005
          ? statsPct(restArea, parentArea) + " of " + parentName + "'s area is in no " + g.mod.label + " district."
          : "Every part of " + parentName + " is in one of these districts.";
        sec.appendChild(g.restEl);
      }
      g.sumArea = sumArea;
      body.appendChild(sec);
    });
    if (found.capped) body.appendChild(cardEl("p", "stats-note", "Showing the first " + REL_MAX_OUTLINES + " districts."));
    var foot = cardEl("p", "stats-foot");
    body.appendChild(foot);
    m.open();
    syncUrlHash(); // the permalink now reopens this report (stats=1)

    // People, where this app ships them: every county the districts touch.
    var box = featureBBox(parentFeature).slice();
    found.groups.forEach(function (g) {
      g.rows.forEach(function (r) {
        var b = featureBBox(r.feature);
        box = [Math.min(box[0], b[0]), Math.min(box[1], b[1]), Math.max(box[2], b[2]), Math.max(box[3], b[3])];
      });
    });
    // PRINT OR SHARE: every row opens and its card loads before the browser's
    // print dialog appears, because a printout cannot be clicked open. On a
    // phone that dialog is also where the report is shared or saved as a PDF.
    // Each row's card waits up to 15 s; one that has not answered prints as
    // "Loading…" rather than holding the whole report back.
    var peopleDone = Promise.resolve();
    printBtn.addEventListener("click", function () {
      trackEvent("compare-stats-print/" + parentMod.id);
      printBtn.disabled = true;
      printBtn.textContent = "Opening every row\u2026";
      var waits = [parentRun, peopleDone].concat(cells.map(function (c) { c.det.open = true; return c.run(); }));
      var settled = Promise.all(waits.map(function (w) { return Promise.resolve(w)["catch"](function () {}); }));
      var timeout = new Promise(function (res) { setTimeout(res, 15000); });
      Promise.race([settled, timeout]).then(function () {
        // and every section inside the cards (a card's "Offices"), which
        // paper cannot open either
        [].forEach.call(m.body.querySelectorAll("details"), function (d) { d.open = true; });
        printBtn.disabled = false;
        printBtn.textContent = "Print or share";
        if (!live() || m.el.hidden) return;
        syncUrlHash();
        printSource.textContent = "From " + String(window.location.href) + " \u00b7 " + new Date().toLocaleDateString();
        document.body.classList.add("stats-printing");
        window.print();
      });
    });
    var pending = typeof POPULATION_INDEX_URL !== "undefined";
    cells.forEach(function (c) { c.el.textContent = pending ? "People: counting…" : ""; c.el.hidden = !pending; });
    foot.textContent = pending
      ? "People are Census 2020 counts by block. A block on a district's edge counts wholly on the side its centre falls in, so a district not drawn on block lines gets a close figure, not an exact one. Areas are measured from the map's simplified boundaries."
      : "Areas are measured from the map's simplified boundaries. This app has no population figures yet.";
    if (!pending) return;
    peopleDone = statsLoadPopulation(box).then(function (sets) {
      if (!live()) return;
      if (!sets) { cells.forEach(function (c) { c.el.hidden = true; }); return; }
      var inParent = statsPointsIn(parentFeature.geometry, sets);
      var parentPop = statsSetsWeight(inParent);
      totals.textContent = statsNum(parentPop) + " people · " + statsSqMi(parentArea);
      var sharedByLayer = {};
      cells.forEach(function (c) {
        var all = statsWeight(c.row.feature.geometry, sets);
        var shared = statsWeight(c.row.feature.geometry, inParent);
        sharedByLayer[c.row.mod.id] = (sharedByLayer[c.row.mod.id] || 0) + shared;
        c.el.textContent = "People: " + statsPct(shared, all) + " of its " + statsNum(all) + " are in " +
          parentName + ", making up " + statsPct(shared, parentPop) + " of " + parentName;
        c.shared = shared;
      });
      // with people counted, the districts holding most of the compared
      // one's people lead each layer's list (area ordered it until now)
      found.groups.forEach(function (g) {
        var mine = cells.filter(function (c) { return c.row.mod === g.mod; });
        if (!mine.length) return;
        var parent = mine[0].det.parentNode, after = g.restEl && g.restEl.parentNode === parent ? g.restEl : null;
        mine.sort(function (a, b) { return b.shared - a.shared; });
        mine.forEach(function (c) { parent.insertBefore(c.det, after); });
      });
      found.groups.forEach(function (g) {
        if (!g.restEl || !g.restEl.parentNode) return;
        var rest = parentPop - (sharedByLayer[g.mod.id] || 0);
        if (rest > 0) {
          g.restEl.textContent = statsNum(rest) + " people (" + statsPct(rest, parentPop) + ") and " +
            statsPct(Math.max(0, parentArea - g.sumArea), parentArea) + " of the area of " + parentName +
            " are in no " + g.mod.label + " district.";
        }
      });
    })["catch"](function (err) {
      if (!live()) return;
      console.warn("population files unavailable:", err);
      cells.forEach(function (c) { c.el.textContent = "People: the population files did not load."; });
    });
  }

  // "Copy link": the permalink with stats=1, so the reader it is sent to
  // lands on this report. On a touch device the platform's share sheet, as
  // the point chip's Share does. Counted by the pinned layer, never the
  // district.
  function statsLinkButton(parentMod, parentName) {
    var btn = document.createElement("button");
    btn.type = "button";
    btn.className = "stats-link-btn";
    var native = !hoverCapable && !!navigator.share;
    var label = native ? "Share link" : "Copy link";
    btn.textContent = label;
    btn.title = "A link that opens this report";
    btn.addEventListener("click", function () {
      trackEvent("compare-stats-link/" + parentMod.id);
      var url = shareLinkUrl();
      if (native) {
        navigator.share({ title: parentName + " \u00b7 " + appDisplayName(), url: url })
          .catch(function () { /* dismissed, or the platform refused — nothing to undo */ });
        return;
      }
      copyTextToClipboard(url).then(function (ok) {
        btn.textContent = ok ? "Copied!" : "Copy failed";
        setTimeout(function () { btn.textContent = label; }, 1600);
      });
    });
    return btn;
  }

  // One row's card: the layer's own query at a point inside the district, and
  // its own render. Isolated like every card: a failure stays in this row.
  function statsRunCard(r, parentFeature, el, live) {
    var spots = relLabelPoint(r.feature, parentFeature, r.inside);
    var point = spots.length ? { lat: spots[0].lat, lng: spots[0].lng } : statsPointInside(r.feature);
    if (!point) {
      el.appendChild(cardEl("p", "stats-note", "This district is too thin to look up a point inside it."));
      return Promise.resolve();
    }
    return statsShowCard(r.mod, point, el, live);
  }
  // Any point inside a district, for one whose share of the compared district
  // is a sliver the label search cannot land in: a finer grid over the
  // district's own extent, the point nearest its middle. Its card is the same
  // wherever inside the district it is asked.
  function statsPointInside(feature) {
    var b = featureBBox(feature), best = null, bestD = Infinity, N = 24;
    var cx = (b[0] + b[2]) / 2, cy = (b[1] + b[3]) / 2;
    for (var i = 0; i < N; i++) {
      for (var j = 0; j < N; j++) {
        var x = b[0] + (b[2] - b[0]) * (i + 0.5) / N, y = b[1] + (b[3] - b[1]) * (j + 0.5) / N;
        var d = (x - cx) * (x - cx) + (y - cy) * (y - cy);
        if (d < bestD && pointInGeometry({ lat: y, lng: x }, feature.geometry)) { best = { lat: y, lng: x }; bestD = d; }
      }
    }
    return best;
  }
  // A layer's card for one point: its own query and its own render, plus its
  // footer link. Resolves when the card is on the page (or has failed).
  function statsShowCard(mod, point, el, live) {
    var r = { mod: mod };
    el.appendChild(cardEl("p", "stats-note", "Loading…"));
    return Promise.resolve()
      .then(function () { return r.mod.query(point, state.sequence); })
      .then(function (result) {
        if (!live()) return;
        el.innerHTML = "";
        if (!result) { el.appendChild(cardEl("p", "stats-note", "This layer has nothing more to show here.")); return; }
        if (!r.mod.compact && r.mod.render) el.appendChild(r.mod.render(result));
        var link = r.mod.primaryLink ? r.mod.primaryLink(result) : null;
        var a = link ? cardExternalLink(link.label || "More", link.url) : null;
        // the card's footer link, unless its body already carries that address
        var dup = a && [].some.call(el.querySelectorAll("a[href]"), function (x) { return x.href === a.href; });
        if (a && !dup) { a.classList.add("stats-link"); el.appendChild(a); }
        if (!el.firstChild) el.appendChild(cardEl("p", "stats-note", "This layer's card names only the district."));
      })["catch"](function (err) {
        if (!live()) return;
        console.error("[" + r.mod.id + "] stats card failed:", err);
        el.innerHTML = "";
        el.appendChild(cardEl("p", "stats-note", "This layer's data source didn't respond. It won't affect the other rows."));
      });
  }
