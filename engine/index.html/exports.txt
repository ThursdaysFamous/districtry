  // A feature's geometry as a short key: its coordinate count and its first
  // coordinate to six decimals. What layerSources and loadCountyEntry report,
  // and what scripts/probe_layer_sources.mjs computes for a file's features.
  function geometryFingerprint(f) {
    var n = 0, first = null;
    (function walk(c) {
      if (typeof c[0] === "number") { n++; if (!first) first = c; return; }
      for (var i = 0; i < c.length; i++) walk(c[i]);
    })((f && f.geometry && f.geometry.coordinates) || []);
    return n + ":" + (first ? first[0].toFixed(6) + "," + first[1].toFixed(6) : "");
  }

  var EXPLORER_EXPORTS = {
    state: state,
    map: map, // debug/programmatic map control (lets tests drive panning, e.g. the sibling-metro handoff)

    // Every district of a county-dispatched layer, fetched through that
    // layer's OWN per-county loaders and named by its own hover identity —
    // what scripts/build_district_search.py reads to index "Lake County Board
    // District 3" without re-implementing ~60 county fetches. Settles every
    // county, failed ones included (ok:false), so a build can refuse a partial
    // index rather than silently ship one.
    layerDistricts: function (id) {
      var mod = getLayerById(id);
      if (!mod || !mod.countyEntries) return Promise.reject(new Error(id + " is not a county-dispatched layer"));
      return Promise.all(mod.countyEntries.map(function (entry) {
        return Promise.resolve().then(function () { return entry.loadGeometry(); }).then(function (geojson) {
          return {
            key: entry.key,
            ok: true,
            features: (geojson.features || []).map(function (f) {
              var wrapped = { type: "Feature", geometry: f.geometry, properties: f.properties, dxCounty: entry.key };
              return { value: hoverValue(mod, wrapped), geometry: f.geometry };
            })
          };
        }, function (err) {
          return { key: entry.key, ok: false, error: String(err && err.message || err) };
        });
      }));
    },
    // What scripts/probe_layer_sources.mjs asks of a layer once it is on: the
    // geometry types its own loader returned (so a point dataset is not counted
    // as shapes) and, for a county-dispatched layer, its entry keys — each
    // loaded one at a time through loadCountyEntry so its requests can be
    // attributed to that county.
    layerSources: function (id) {
      var mod = getLayerById(id), rt = layerRuntime[id];
      if (!mod) return null;
      var types = {};
      var features = (rt && rt.geojson && rt.geojson.features) || [];
      features.forEach(function (f) {
        if (f && f.geometry && f.geometry.type) types[f.geometry.type] = true;
      });
      var parent = mod.subOf ? getLayerById(mod.subOf) : null;
      return {
        loaded: !!(rt && rt.geojson),
        geometryTypes: Object.keys(types).sort(),
        countyKeys: mod.countyEntries ? mod.countyEntries.map(function (e) { return e.key; }) : null,
        // a sub-layer switches its parent on too, so the probe subtracts the
        // parent's requests; one drawing the SAME loaded set as its parent
        // makes no request of its own, and says so here
        subOf: mod.subOf || null,
        sharesParentLoader: !!(parent && parent.overlay && mod.overlay && parent.overlay.load === mod.overlay.load),
        // drawn from a vector-tile archive: its whole file is fetched only when
        // asked (loadWholeFile below), so the probe asks before reading
        // from the registration, not the overlay: a layer hidden at this
        // point (mod.coverage) detaches its overlay and is still tiled
        tiles: tileDrawAvailable(mod) ? mod.tiles : null,
        // one short key per loaded feature, so the probe can tell which of
        // this site's files the layer DRAWS: a file the app also loads at boot
        // is subtracted with the boot requests, and matching features finds it
        fingerprints: features.map(geometryFingerprint)
      };
    },
    // What scripts/probe_layer_load.mjs polls while one layer loads: whether
    // its card has an answer yet, whether its boundaries are on the map, and
    // whether the selected district is lit — the three moments a reader sees.
    layerLoadState: function (id) {
      var rt = layerRuntime[id];
      if (!rt) return null;
      var body = rt.cardBody, cls = (body && body.className) || "";
      var card = !body ? "none"
        : /state-off/.test(cls) ? "off"
        : body.querySelector(".loading-row") ? "loading"
        : /state-error/.test(cls) ? "error"
        : /state-empty/.test(cls) ? "empty"
        : "result";
      return {
        card: card,
        relevant: rt.relevant !== false,
        overlay: !!(rt.overlayLayer && map.hasLayer(rt.overlayLayer)),
        highlight: !!rt.highlightApplied || !!rt.provisionalLayer,
        features: (rt.geojson && rt.geojson.features && rt.geojson.features.length) || 0
      };
    },
    // A tile-drawn layer's whole GeoJSON file, which it otherwise fetches
    // only for a comparison; what scripts/probe_layer_sources.mjs asks for
    // so it can still say which file the layer's districts come from.
    // A tile-drawn layer's state: whether it is on the tile canvas, the lit
    // district's tile feature id, and how many runs of real edge the
    // boundary-street names have to follow (tileEdgesForStreets)
    tileState: function (id) {
      var mod = getLayerById(id), rt = layerRuntime[id];
      if (!mod || !rt) return null;
      var g = rt.highlightedLayer && rt.highlightedLayer.feature && rt.highlightedLayer.feature.geometry;
      return { drawn: isTileDrawn(mod), selected: rt.tileSelectedId,
        edgeRuns: g && g.type === "MultiPolygon" && g.coordinates[0] ? g.coordinates[0].length : 0 };
    },
    // A tile-drawn layer's own query at each point, answered first from its
    // tiles and then from its whole file, as JSON with every `seq` and
    // `geometry` dropped (a tile answer carries the piece of a district
    // inside one tile, which is not what a card reads). What
    // scripts/probe_tile_cards.mjs compares: the card a reader sees must be
    // the same either way. The tile answers run first, while the whole file
    // is not yet in memory, or the loader would answer them from it.
    //
    // A county-dispatched layer turns its tiles off for the second run
    // (mod.tilesOff), and each answer also says whether the county that
    // answered is in the archive at all (`inArchive`): a live county, or a
    // point just over a county line, is answered from its own source by
    // design, and the probe does not count it against the tiles. It also
    // compares the HOVER name: the canvas names a district from the tile's
    // own properties, with nothing a loader stamps after its fetch, so a
    // hover that reads a stamped property would differ from the file's.
    tileCardCheck: function (id, points) {
      var mod = getLayerById(id), load = mod && mod.overlay && mod.overlay.load;
      if (!load || !load.tileMod) return Promise.reject(new Error(id + " is not drawn from tiles"));
      var rt = layerRuntime[id], viaTiles = [];
      if (mod.countyEntries) {
        var countyResults = [], hovers = [];
        var byKey = {};
        mod.countyEntries.forEach(function (e) { byKey[e.key] = e; });
        function hoverPair(p) {
          return tileFeaturesAt(tileArchiveFor(mod), p).then(function (fc) {
            var tf = findFeatureContaining(fc, p);
            var key = tf && tf.properties._c, entry = key != null && byKey[key];
            if (!entry || !entry.loadGeometry) return null;
            var props = {};
            for (var k in tf.properties) if (tf.properties.hasOwnProperty(k) && k !== "_c") props[k] = tf.properties[k];
            var tiles = mod.hoverName({ type: "Feature", properties: props, dxCounty: key });
            return Promise.resolve(entry.loadGeometry()).then(function (g) {
              var ff = findFeatureContaining(g, p);
              return { tiles: tiles, file: ff ? mod.hoverName({ type: "Feature", properties: ff.properties, dxCounty: key }) : null };
            });
          }).then(null, function (e) { return { tiles: "error: " + String(e && e.message || e), file: null }; });
        }
        return countyTileKeys(mod).then(function (keys) {
          function runCounty(out, record) {
            return points.reduce(function (chain, p, i) {
              return chain.then(function () {
                rt.tileAnswer = null;
                return Promise.resolve().then(function () { return mod.query(p, -1 - i); }).then(function (r) {
                  out.push(r == null ? null : JSON.stringify(r, function (k, v) {
                    return k === "seq" || k === "geometry" ? undefined : v;
                  }));
                  if (record) {
                    countyResults.push({ via: !!(rt.tileAnswer && rt.tileAnswer.point === p),
                      county: r ? r.dxCounty : null, inArchive: !!(r && keys[r.dxCounty]) });
                    return hoverPair(p).then(function (h) { hovers.push(h); });
                  }
                }, function (e) {
                  out.push("error: " + String(e && e.message || e));
                  if (record) { countyResults.push({ via: false, county: null, inArchive: false }); hovers.push(null); }
                });
              });
            }, Promise.resolve());
          }
          var tiled = [], whole = [];
          return runCounty(tiled, true).then(function () {
            mod.tilesOff = true;
            return runCounty(whole);
          }).then(function () {
            mod.tilesOff = false;
            return points.map(function (p, i) {
              var c = countyResults[i];
              var h = hovers[i];
              return { point: p, tiles: tiled[i], file: whole[i], viaTiles: c.via, county: c.county, inArchive: c.inArchive,
                hoverTiles: h ? h.tiles : null, hoverFile: h ? h.file : null };
            });
          }, function (e) { mod.tilesOff = false; throw e; });
        });
      }
      function run(out, record) {
        return points.reduce(function (chain, p, i) {
          return chain.then(function () {
            if (record) rt.tileAnswer = null;
            return Promise.resolve().then(function () { return mod.query(p, -1 - i); }).then(function (r) {
              out.push(r == null ? null : JSON.stringify(r, function (k, v) {
                return k === "seq" || k === "geometry" ? undefined : v;
              }));
              // whether the tile path really answered: a query that reaches
              // its boundaries some other way passes the comparison while
              // downloading the whole file and lighting nothing
              if (record) viaTiles.push(!!(rt.tileAnswer && rt.tileAnswer.point === p));
            }, function (e) { out.push("error: " + String(e && e.message || e)); if (record) viaTiles.push(false); });
          });
        }, Promise.resolve());
      }
      var tiled = [], whole = [], keep = load.tileMod;
      return run(tiled, true).then(function () {
        load.tileMod = null;
        return run(whole);
      }).then(function () {
        load.tileMod = keep;
        return points.map(function (p, i) { return { point: p, tiles: tiled[i], file: whole[i], viaTiles: viaTiles[i] }; });
      }, function (e) { load.tileMod = keep; throw e; });
    },
    // The colour a layer's outline is painted in right now, whichever way it
    // is drawn: the tile canvas's paint, or a Leaflet path's stroke. What the
    // smoke tests read to see the dark theme repaint a live overlay.
    /* Every county a layer's archive holds, checked whole rather than at
     * points: the county's own files, run through its loader's `decorate` (a
     * tile carries the file's properties and nothing the app adds after the
     * fetch), must give exactly the features the loader itself returns. A
     * loader that stamps, filters or merges outside `withStamp` fails here the
     * day its county enters the archive, where a point probe finds it only if
     * a point lands there — which is how Boone's library officials were missed.
     * `filesByKey` is layer-sources.json's `counties.files`, the list the
     * archive was built from. */
    countyStampCheck: function (id, filesByKey) {
      var mod = getLayerById(id);
      if (!mod || !mod.countyEntries || !mod.tiles) return Promise.reject(new Error(id + " is not a tiled county layer"));
      var byKey = {};
      mod.countyEntries.forEach(function (e) { byKey[e.key] = e; });
      function canon(v) {
        if (v === null || typeof v !== "object") return JSON.stringify(v);
        if (Array.isArray(v)) return "[" + v.map(canon).join(",") + "]";
        return "{" + Object.keys(v).sort().map(function (k) { return JSON.stringify(k) + ":" + canon(v[k]); }).join(",") + "}";
      }
      function signatures(geojson) {
        return ((geojson && geojson.features) || []).map(function (f) {
          return canon(f.properties || {}) + "|" + geometryFingerprint(f);
        }).sort();
      }
      return countyTileKeys(mod).then(function (keys) {
        return Object.keys(keys).sort().reduce(function (chain, key) {
          return chain.then(function (out) {
            var entry = byKey[key], files = (filesByKey && filesByKey[key]) || [];
            var rec = { key: key };
            out.push(rec);
            if (!entry || !entry.loadGeometry) { rec.problem = "the archive holds a county the layer has no entry for"; return out; }
            if (!files.length) { rec.problem = "layer-sources.json names no file for this county"; return out; }
            var load = entry.loadGeometry;
            return Promise.all([
              Promise.resolve().then(function () { return load(); }),
              Promise.all(files.map(function (u) {
                return fetch(u, { cache: "no-store" }).then(function (r) {
                  if (!r.ok) throw new Error(u + " answered " + r.status);
                  return r.json();
                });
              })).then(function (docs) {
                var raw = { type: "FeatureCollection", features: [] };
                docs.forEach(function (d) { raw.features = raw.features.concat((d && d.features) || []); });
                return load.decorate ? load.decorate(raw) : raw;
              })
            ]).then(function (both) {
              var a = signatures(both[0]), b = signatures(both[1]);
              rec.features = a.length;
              rec.decorate = !!load.decorate;
              var i = 0;
              while (i < a.length && i < b.length && a[i] === b[i]) i++;
              if (a.length !== b.length || i < a.length) {
                rec.problem = "the loader returns " + a.length + " feature(s) and its files through decorate give " + b.length +
                  (i < Math.max(a.length, b.length) ? "; first difference: loader " + String(a[i]).slice(0, 200) + " | files " + String(b[i]).slice(0, 200) : "");
              }
              return out;
            }, function (e) { rec.problem = "error: " + String(e && e.message || e); return out; });
          });
        }, Promise.resolve([]));
      });
    },
    overlayColor: function (id) {
      var mod = getLayerById(id), rt = layerRuntime[id];
      if (!mod || !rt || !rt.overlayLayer) return null;
      if (rt.overlayLayer.dxTiles) {
        var gl = tileGLLayer && tileGLLayer.getMaplibreMap(), ids = tileIds(mod);
        return gl && gl.getLayer(ids.line) ? gl.getPaintProperty(ids.line, "line-color") : null;
      }
      var color = null;
      rt.overlayLayer.eachLayer(function (l) { if (!color && l.options && l.options.color) color = l.options.color; });
      return color;
    },
    loadWholeFile: function (id) {
      var mod = getLayerById(id);
      if (!mod || !tileDrawAvailable(mod)) return Promise.resolve(false);
      return ensureTileGeojson(mod).then(function () { return true; });
    },
    loadCountyEntry: function (id, key) {
      var mod = getLayerById(id);
      var entry = mod && mod.countyEntries && mod.countyEntries.filter(function (e) { return e.key === key; })[0];
      if (!entry) return Promise.reject(new Error(id + " has no county entry " + key));
      return Promise.resolve().then(function () { return entry.loadGeometry(); }).then(function (geojson) {
        var types = {};
        ((geojson && geojson.features) || []).forEach(function (f) {
          if (f && f.geometry && f.geometry.type) types[f.geometry.type] = true;
        });
        return {
          features: ((geojson && geojson.features) || []).length,
          geometryTypes: Object.keys(types).sort(),
          fingerprints: ((geojson && geojson.features) || []).map(geometryFingerprint)
        };
      });
    },

    splitBaseStyle: splitBaseStyle, // exposed so the smoke test can hold the ground/labels split and the street-name restyle to a fixture, with no CARTO fetch

    boundaryStreets: function () { return { on: boundaryStreetsOn, available: !!baseLabels, names: boundaryStreetNames }; }, // which street names the boundary-street layer is showing, and their colours
    boundaryStreetLabels: function () { return measureBoundaryStreetLabels(); }, // per boundary street: its coloured labels on screen, and any ordinary copy of its name printed inside the stretch
    basemap: function () { return { kind: districtryTileKind, vector: baseIsVector, labels: !!baseLabels }; } // which basemap kind is live + which theme style it shows — the vector canvas has no tile <img> whose src a test could read
