  // expose a couple of things for later module threads / debugging, without
  // polluting global scope beyond one namespace. The object is built here so
  // the member list stays engine; the window property it lands on is fork-
  // branded, so the assignment below stays outside the fence (twinned with
  // each fork's smoke_test.mjs).
  var EXPLORER_EXPORTS = {
    state: state,
    map: map, // debug/programmatic map control (lets tests drive panning, e.g. the sibling-metro handoff)
    registerLayer: registerLayer,
    setSelectedPoint: setSelectedPoint, // debug/programmatic point selection (also used by the smoke test)
    sanitize: sanitize,
    pointInGeometry: pointInGeometry,
    fetchJSONWithRetry: fetchJSONWithRetry,
    probeGeometryColumn: probeGeometryColumn,
    // Every district of a county-dispatched layer, fetched through that
    // layer's OWN per-county loaders and named by its own hover identity —
    // what scripts/build_district_search.py reads to index "Lake County Board
    // District 3" without re-implementing ~60 county fetches. Settles every
    // county, failed ones included (ok:false), so a build can refuse a partial
    // index rather than silently ship one.
    layerDistricts: function (id) {
      var mod = getLayerById(id);
      if (!mod || !mod.countyEntries) return Promise.reject(new Error(id + " is not a county-dispatched layer"));
      return Promise.all(mod.countyEntries.map(function (entry) {
        return Promise.resolve().then(function () { return entry.loadGeometry(); }).then(function (geojson) {
          return {
            key: entry.key,
            ok: true,
            features: (geojson.features || []).map(function (f) {
              var wrapped = { type: "Feature", geometry: f.geometry, properties: f.properties, dxCounty: entry.key };
              return { value: hoverValue(mod, wrapped), geometry: f.geometry };
            })
          };
        }, function (err) {
          return { key: entry.key, ok: false, error: String(err && err.message || err) };
        });
      }));
    },
    cleanPoiAddress: cleanPoiAddress, // exposed so the smoke test can regression-guard the office-pin address cleaner (a recurring break point) without any network call
    splitBaseStyle: splitBaseStyle, // exposed so the smoke test can hold the ground/labels split and the street-name restyle to a fixture, with no CARTO fetch
    basemap: function () { return { kind: districtryTileKind, vector: baseIsVector, labels: !!baseLabels }; } // which basemap kind is live + which theme style it shows — the vector canvas has no tile <img> whose src a test could read
  };
