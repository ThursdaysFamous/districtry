  /* =========================================================================
   * PIN-PARENT RELATIONSHIPS — pick a district to compare against from its
   * sidebar card, then see how every OTHER active layer's districts relate to
   * it:
   *   • solid outline = that district falls fully INSIDE the pinned one
   *   • dashed outline = it CROSSES the pinned one's edge (partly in, partly out)
   * Pinning lives on the result cards (not a map click) so click-to-select a
   * point still works. The parent feature is always "the pinned layer's district
   * at the currently selected point", so it tracks as you pick new points.
   * Outlines are drawn into their own featureGroup, never by restyling the real
   * overlays, so the click-selection highlight is untouched. Reuses the shared
   * polygonFullyInside / polygonsOverlap geometry (each candidate bbox-gated).
   * ======================================================================= */
  var pinnedParentId = null;      // mod.id of the pinned parent layer, or null
  var relationshipGroup = null;   // L.featureGroup holding the relationship outlines
  var relLabelGroup = null;       // L.layerGroup holding each outlined district's map label
  var relLabels = [];             // [{marker, el, spots, clearance, here}] for the layout pass
  var REL_MAX_OUTLINES = 250;     // perf backstop for a huge parent over a dense layer
  var relLegendEl = document.getElementById("rel-legend");

  /* ---- district labels on the compared outlines ----
   * Each outlined district carries a label naming it, so a reader comparing
   * wards against a school-board district can read "Ward 47" off the map
   * instead of hovering every shape.
   *
   * THE TEXT IS THE CARD PILL'S WORDING, AND IT IS LEARNED RATHER THAN WRITTEN.
   * A card pill is computed from a query RESULT for one point, so there is no
   * per-feature pill to call for the other forty wards. What is known is the
   * pill on screen for the feature at the selected point ("Ward 47") and that
   * same feature's hover identity ("47"). From those two:
   *   1. a property whose value IS the pill names every feature exactly, by
   *      each feature's own value for that property;
   *   2. otherwise, when the pill is the identity inside a wrapper ("Ward ",
   *      "IL-"), the wrapper is reused — but only if none of its words occurs
   *      in the feature's own properties (bar a value that carries the
   *      identity itself — the feature's own formatted name), and only on
   *      features with the same property schema. Text read from a property
   *      varies by feature: one layer holds Lockport's wards and Joliet's
   *      council DISTRICTS, the word coming from each feature's dxSeatLabel,
   *      and a township precinct pill starts with the township's name. Text
   *      found nowhere in the feature is a literal in the card's code and
   *      holds for every feature it built.
   * Anything unproven falls back to the bare identity, which is always true of
   * the feature it is printed on. */
  function relPropSignature(feature) {
    return Object.keys(feature.properties || {}).sort().join("|");
  }
  function relLabelWrapper(mod) {
    var rt = layerRuntime[mod.id];
    if (!rt || rt.resultSeq !== state.sequence || !state.selectedPoint) return null;
    var el = rt.idPill && !rt.idPill.hidden ? rt.idPill
      : (rt.compactValue && !rt.compactValue.hidden ? rt.compactValue : null);
    if (!el) return null;
    var pill = String(el.textContent || "").trim();
    var here = findFeatureContaining(rt.geojson, state.selectedPoint);
    if (!pill || !here) return null;
    var props = here.properties || {};
    for (var key in props) {
      if (props.hasOwnProperty(key) && props[key] !== null && String(props[key]).trim() === pill) return { key: key };
    }
    var value = hoverValue(mod, here);
    if (value === "—") return null;
    var sig = relPropSignature(here);
    if (pill === value) return { pre: "", post: "", sig: sig };
    var at = pill.indexOf(value);
    if (at < 0 || pill.indexOf(value, at + 1) >= 0) return null; // absent, or ambiguous
    var pre = pill.slice(0, at), post = pill.slice(at + value.length);
    // the identity must be a whole token of the pill: "4" inside "IL-14" is not "14"
    if (/[A-Za-z0-9]$/.test(pre) || /^[A-Za-z0-9]/.test(post)) return null;
    if ((pre + post).length > 24) return null;
    var words = (pre + " " + post).match(/[A-Za-z]+/g) || [];
    var ident = " " + value.toLowerCase().replace(/[^a-z0-9]+/g, " ") + " ";
    for (var k in props) {
      if (!props.hasOwnProperty(k) || typeof props[k] !== "string") continue;
      var text = " " + props[k].toLowerCase().replace(/[^a-z0-9]+/g, " ") + " ";
      // a value carrying the identity itself ("State House District 13") is
      // the feature's own name for itself, not wording that varies around it
      if (text.indexOf(ident) >= 0) continue;
      for (var i = 0; i < words.length; i++) {
        if (text.indexOf(" " + words[i].toLowerCase() + " ") >= 0) return null; // feature-specific text
      }
    }
    return { pre: pre, post: post, sig: sig };
  }
  function relLabelText(mod, feature, wrapper) {
    if (wrapper && wrapper.key) {
      var own = (feature.properties || {})[wrapper.key];
      if (own !== null && own !== undefined && String(own).trim() !== "") return String(own).trim();
    }
    var value = hoverValue(mod, feature);
    if (value === "—") return null;
    if (wrapper && wrapper.sig !== undefined && relPropSignature(feature) === wrapper.sig) {
      return wrapper.pre + value + wrapper.post;
    }
    return value;
  }

  // Planar distance in LATITUDE-degree units, with longitude scaled by cos(lat)
  // so a label's clearance means the same distance in every direction.
  function relSegDist(px, py, a, b, kx) {
    return distPointToSegment(px * kx, py, a[0] * kx, a[1], b[0] * kx, b[1]);
  }
  function relRingsNear(geometry, box, pad) {
    // the boundary segments that could be the nearest edge to any point in box
    var out = [];
    eachRing(geometry, function (ring) {
      for (var i = 0; i < ring.length - 1; i++) {
        var a = ring[i], b = ring[i + 1];
        if (Math.max(a[0], b[0]) < box[0] - pad || Math.min(a[0], b[0]) > box[2] + pad ||
            Math.max(a[1], b[1]) < box[1] - pad || Math.min(a[1], b[1]) > box[3] + pad) continue;
        out.push(a, b);
      }
    });
    return out;
  }

  // Where to put a district's label: the points of (feature ∩ parent) farthest
  // from both boundaries, found on a coarse grid with the best refined once.
  // For a district fully inside the parent that is simply its own widest
  // spot; for one crossing the edge it lands in the part that falls
  // INSIDE the compared district, so every label sits within that shape.
  // Memoized per (feature, parent) pair: outlines redraw each time any layer's
  // query resolves, and the geometry has not changed between those redraws.
  function relLabelPoint(feature, parent, inside) {
    var memo = feature.__relLabel;
    if (memo && memo.parent === parent) return memo.spots;
    var fb = featureBBox(feature), pb = featureBBox(parent);
    var box = inside ? fb : [Math.max(fb[0], pb[0]), Math.max(fb[1], pb[1]), Math.min(fb[2], pb[2]), Math.min(fb[3], pb[3])];
    var spots = [];
    if (box[0] < box[2] && box[1] < box[3]) {
      var kx = Math.cos(((box[1] + box[3]) / 2) * Math.PI / 180);
      var pad = Math.max((box[2] - box[0]) * kx, box[3] - box[1]);
      var segs = relRingsNear(feature.geometry, box, pad);
      // a district that is fully inside its parent is nearer its own edge than
      // the parent's at every interior point, so the parent's edge is skipped
      if (!inside) segs = segs.concat(relRingsNear(parent.geometry, box, pad));
      var clearance = function (x, y) {
        if (!pointInGeometry({ lat: y, lng: x }, feature.geometry)) return -1;
        if (!inside && !pointInGeometry({ lat: y, lng: x }, parent.geometry)) return -1;
        var best = Infinity;
        for (var s = 0; s < segs.length; s += 2) {
          var d = relSegDist(x, y, segs[s], segs[s + 1], kx);
          if (d < best) best = d;
        }
        return best;
      };
      var scan = function (x0, y0, x1, y1, n, into) {
        for (var i = 0; i < n; i++) {
          for (var j = 0; j < n; j++) {
            var x = x0 + (x1 - x0) * (i + 0.5) / n, y = y0 + (y1 - y0) * (j + 0.5) / n;
            var c = clearance(x, y);
            if (c > 0) into.push({ lat: y, lng: x, clearance: c });
          }
        }
      };
      var N = 10, grid = [];
      scan(box[0], box[1], box[2], box[3], N, grid);
      grid.sort(function (a, b) { return b.clearance - a.clearance; });
      if (grid.length) {
        var dx = (box[2] - box[0]) / N, dy = (box[3] - box[1]) / N, fine = [];
        scan(grid[0].lng - dx, grid[0].lat - dy, grid[0].lng + dx, grid[0].lat + dy, 5, fine);
        fine.sort(function (a, b) { return b.clearance - a.clearance; });
        if (fine.length && fine[0].clearance > grid[0].clearance) grid[0] = fine[0];
      }
      // up to eight spots, each clear of the ones before it by its own
      // clearance, so a label that would cover the selected point or a
      // neighbour's label has somewhere else in its district to go
      for (var g = 0; g < grid.length && spots.length < 8; g++) {
        var cand = grid[g];
        var apart = spots.every(function (sp) {
          return Math.hypot((sp.lng - cand.lng) * kx, sp.lat - cand.lat) >= Math.max(sp.clearance, cand.clearance);
        });
        if (apart) spots.push(cand);
      }
    }
    feature.__relLabel = { parent: parent, spots: spots };
    return spots;
  }

  // Show a label wherever it collides with nothing already placed; the
  // district holding the selected point claims its spot first and then the
  // widest, so where labels compete for space those keep theirs and zooming
  // in reveals the rest. Runs on every
  // zoom, since pixel positions are the test.
  // A CHIP MAY CROSS ITS OWN DISTRICT'S BORDER. The first rule required the
  // chip to (nearly) fit inside its district, and on a phone that asked the
  // reader to zoom in until the comparison no longer fit on screen — you could
  // see the districts or read their names, not both. Overlap with another
  // label or a map marker is still refused, because a chip over a chip is
  // unreadable and a chip over the star hides the point the reader chose.
  // The one size floor is a district under ~8px across, a speck that would
  // otherwise claim a chip ahead of a neighbour the reader can actually see.
  var REL_LABEL_MIN_ROOM = 4; // px from the label spot to its nearest edge
  function layoutRelLabels() {
    if (!relLabels.length) return;
    // every marker on the map (the selected point, office pins) is occupied
    // ground: a label drawn over the star hides the point the reader chose
    var placed = [];
    var box = map.getContainer().getBoundingClientRect();
    map.eachLayer(function (layer) {
      if (!(layer instanceof L.Marker) || layer.options.pane === "rel-labels") return;
      var icon = layer.getElement && layer.getElement();
      if (!icon) return;
      var r = icon.getBoundingClientRect();
      if (!r.width && !r.height) return;
      placed.push({ x0: r.left - box.left, y0: r.top - box.top, x1: r.right - box.left, y1: r.bottom - box.top });
    });
    relLabels.forEach(function (l) {
      l.el.classList.remove("rel-map-label-hidden");
      var w = l.el.offsetWidth, h = l.el.offsetHeight;
      for (var i = 0; i < l.spots.length; i++) {
        var spot = l.spots[i];
        var p = map.latLngToContainerPoint([spot.lat, spot.lng]);
        var edge = map.latLngToContainerPoint([spot.lat + spot.clearance, spot.lng]);
        var room = Math.abs(p.y - edge.y); // pixels from this spot to the nearest edge
        if (room < REL_LABEL_MIN_ROOM) continue;
        var rect = { x0: p.x - w / 2 - 3, y0: p.y - h / 2 - 3, x1: p.x + w / 2 + 3, y1: p.y + h / 2 + 3 };
        var hits = placed.some(function (r) { return rect.x0 < r.x1 && rect.x1 > r.x0 && rect.y0 < r.y1 && rect.y1 > r.y0; });
        if (hits) continue;
        l.marker.setLatLng([spot.lat, spot.lng]);
        placed.push(rect);
        return;
      }
      l.el.classList.add("rel-map-label-hidden");
    });
  }

  function relationshipParentFeature() {
    if (!pinnedParentId || !state.selectedPoint || !state.layersOn[pinnedParentId]) return null;
    var rt = layerRuntime[pinnedParentId];
    if (!rt || !rt.geojson) return null;
    // a relevance-hidden parent (mod.coverage) contributes no outlines while
    // out of coverage; pinnedParentId is kept so the pin resurrects with it
    if (rt.relevant === false) return null;
    return findFeatureContaining(rt.geojson, state.selectedPoint);
  }

  function drawRelationshipOutlines(parentFeature) {
    // the pinned parent itself — bold dark frame so it reads as the reference
    L.geoJSON(parentFeature, {
      style: { color: "#14181C", weight: 4, opacity: 1, fill: false, className: "rel-parent-outline" },
      interactive: false
    }).addTo(relationshipGroup);

    var nIn = 0, nCross = 0, drawn = 0, capped = false;
    for (var li = 0; li < layers.length && !capped; li++) {
      var mod = layers[li];
      if (mod.id === pinnedParentId) continue;          // don't relate a layer to itself
      if (!state.layersOn[mod.id]) continue;
      if (!mod.overlay || !mod.overlay.style) continue; // polygon layers only
      var rt = layerRuntime[mod.id];
      if (!rt || !rt.geojson) continue;
      if (rt.relevant === false) continue; // relevance-hidden here (mod.coverage) — draw nothing for it
      var color = darkenHexColor(moduleColor(mod), 0.15);
      var wrapper = relLabelWrapper(mod);
      var feats = rt.geojson.features || [];
      for (var j = 0; j < feats.length; j++) {
        var f = feats[j];
        var inside = polygonFullyInside(f, parentFeature);
        var cross = !inside && polygonsOverlap(f, parentFeature);
        if (!inside && !cross) continue;
        if (drawn >= REL_MAX_OUTLINES) { capped = true; break; }
        L.geoJSON(f, {
          // fill:false → outline only (no added fill-murk over the base layers);
          // interactive:false → never intercepts the map click/mousemove beneath
          style: {
            color: color, weight: 2.5, opacity: 0.95, fill: false,
            dashArray: inside ? null : "5 4",
            className: inside ? "rel-inside-outline" : "rel-cross-outline"
          },
          interactive: false
        }).addTo(relationshipGroup);
        drawn++;
        if (inside) nIn++; else nCross++;
        addRelLabel(mod, f, parentFeature, inside, wrapper, color);
      }
    }
    if (capped) console.warn("pin relationships: capped at " + REL_MAX_OUTLINES + " outlines");
    relationshipGroup.bringToFront();
    // widest first, so a big district keeps its label when a small one collides
    // the district holding the selected point first — it is the one the
    // reader is standing in, and its spots crowd against the point's own
    // marker, so ranked by width alone it lost to its neighbours (measured
    // 2026-09-22: Ward 47 was the one ward of six unlabelled on a phone)
    relLabels.sort(function (a, b) {
      if (a.here !== b.here) return a.here ? -1 : 1;
      return b.clearance - a.clearance;
    });
    layoutRelLabels();
    return { inside: nIn, cross: nCross, capped: capped };
  }

  function addRelLabel(mod, feature, parentFeature, inside, wrapper, color) {
    var text = relLabelText(mod, feature, wrapper);
    if (!text) return; // no identity to print — an unlabelled outline says nothing false
    var spots = relLabelPoint(feature, parentFeature, inside);
    if (!spots.length) return; // a sliver too thin to hold a point: nothing to label
    var el = document.createElement("span");
    el.className = "rel-map-label";
    el.textContent = text; // textContent: feature text is external
    el.style.borderColor = color;
    el.style.setProperty("--rel-label-ink", hoverPillStyle(color).color);
    var marker = L.marker([spots[0].lat, spots[0].lng], {
      icon: L.divIcon({ className: "rel-map-label-anchor", html: el, iconSize: null }),
      pane: "rel-labels",
      interactive: false,
      keyboard: false
    }).addTo(relLabelGroup);
    var here = !!state.selectedPoint && pointInGeometry(state.selectedPoint, feature.geometry);
    relLabels.push({ marker: marker, el: el, spots: spots, clearance: spots[0].clearance, here: here });
  }

  var relCounts = null; // last computed {inside, cross, capped}, or null

  function refreshRelationshipOutlines() {
    if (!relationshipGroup) {
      relationshipGroup = L.featureGroup().addTo(map);
      // labels sit above every overlay (400) and below the selected-point
      // marker (600); pointer-events off so they never swallow a map click
      var labelPane = map.createPane("rel-labels");
      labelPane.style.zIndex = "450";
      labelPane.style.pointerEvents = "none";
      relLabelGroup = L.layerGroup().addTo(map);
      map.on("zoomend", layoutRelLabels);
    }
    relationshipGroup.clearLayers();
    relLabelGroup.clearLayers();
    relLabels = [];
    var parentFeature = relationshipParentFeature();
    relCounts = parentFeature ? drawRelationshipOutlines(parentFeature) : null;
    updateRelLegend();
  }

  // The legend lives above the Hover toggle and, per the toggle's role as the
  // explorer's on/off switch, is shown only while hover is on AND a parent is
  // pinned. (Outlines themselves stay pinned regardless; unpin is still
  // reachable from the parent's sidebar card when the legend is hidden.)
  function updateRelLegend() {
    if (!relLegendEl) return;
    if (!pinnedParentId || !hoverActive) { relLegendEl.hidden = true; relLegendEl.innerHTML = ""; return; }
    var counts = relCounts;
    var mod = getLayerById(pinnedParentId);
    var parentName = mod ? mod.label : "district";
    var summary;
    if (!counts) {
      summary = '<div class="rel-note">Pick a point inside this district to see what relates to it.</div>';
    } else {
      summary = '<div class="rel-key"><span class="rel-sw rel-sw-in"></span>' + counts.inside + ' fully inside</div>' +
        '<div class="rel-key"><span class="rel-sw rel-sw-cross"></span>' + counts.cross + ' crossing the edge</div>' +
        (counts.capped ? '<div class="rel-note">(showing first ' + REL_MAX_OUTLINES + ')</div>' : "");
    }
    relLegendEl.hidden = false;
    relLegendEl.innerHTML =
      '<div class="rel-legend-head"><span class="rel-sw rel-sw-parent"></span><b>Comparing to:</b>&nbsp;' + sanitize(parentName) + "</div>" +
      summary +
      '<button type="button" class="rel-unpin-btn" id="rel-unpin">Stop comparing</button>';
    var unpin = document.getElementById("rel-unpin");
    if (unpin) unpin.addEventListener("click", function () { setPinnedParent(null); });
  }

  function setPinnedParent(id) {
    pinnedParentId = (pinnedParentId === id) ? null : id;
    // NOTHING COUNTED THIS UNTIL 2026-09-18, so "nobody uses the comparison"
    // was an inference and never a measurement: the app has instrumented
    // eleven actions since it shipped and this was not one of them, which
    // left no way to tell a control readers never find from one they find
    // and do not want. Those have opposite fixes. Same vocabulary as the
    // layer toggles (layer/<id>), and the privacy page's event list is
    // MEASURED off these call sites rather than hand-kept, so it picks this
    // up on its own next build.
    // WRITTEN AS TWO CALLS, NOT A TERNARY, AND THAT IS NOT A STYLE CHOICE.
    // build_privacy_page.py MEASURES the published event list off these call
    // sites, with a regex anchored on a quoted literal immediately inside the
    // parentheses. A ternary argument is invisible to it, so the privacy page
    // would have gone on naming eight events while the app sent ten, and on
    // that page an undercount is a false statement exactly as an overcount is.
    // The same regex reads COMMENTS as code, so this one deliberately does not
    // spell the function name next to a quote.
    if (pinnedParentId) trackEvent("compare/" + pinnedParentId);
    else trackEvent("compare-stop");
    refreshAllPinButtons();
    refreshRelationshipOutlines();
    // The pin is part of the shared view, so pinning and unpinning both move
    // the permalink (see syncUrlHash). This call also carries the RESTORE
    // path, which is easy to miss: applyStateFromHash selects the point
    // before it reads `pin=`, and that selection's own syncUrlHash rewrites
    // the hash while pinnedParentId is still null, so the incoming pin is
    // gone by the time the restore runs. Without this line a shared link
    // restores the pin on screen and hands the next reader a pin-less URL.
    // The other path that clears the pin — toggling the parent layer OFF —
    // needs no call here: onLayerToggled clears pinnedParentId directly and
    // its own change handler runs syncUrlHash immediately afterwards.
    syncUrlHash();
  }

  // Update every card's pin button to reflect the current pinned layer.
  function refreshAllPinButtons() {
    layers.forEach(function (mod) {
      var rt = layerRuntime[mod.id];
      if (rt && rt.pinBtn) stylePinButton(rt.pinBtn, mod.id);
    });
  }
  function stylePinButton(btn, id) {
    var pinned = pinnedParentId === id;
    // THE LABEL IS THE FEATURE'S DISCOVERABILITY, and it used to name the data
    // model rather than the question. "Pin as parent" is the internal shape of
    // this code — a parent feature and its children — and a reader who wants to
    // know whether their ward sits inside one congressional district would not
    // read it as the control that answers that. The internal names stay
    // (pinnedParentId, rel-sw-parent, .pin-parent-btn); only what a reader sees
    // changed, on 2026-09-18.
    btn.textContent = pinned ? "★ Comparing — click to stop" : "☆ Compare to this district";
    btn.setAttribute("aria-pressed", pinned ? "true" : "false");
    btn.classList.toggle("is-pinned", pinned);
  }

  // Append the "Compare to this district" control to a layer's card — only for
  // polygon layers
  // that actually have a district at the selected point. Called after each card
  // renders its result.
  function appendPinControl(mod) {
    if (!mod.overlay || !mod.overlay.style) return; // polygon layers only
    var rt = layerRuntime[mod.id];
    if (!rt || !rt.geojson || !state.selectedPoint) return;
    if (!findFeatureContaining(rt.geojson, state.selectedPoint)) return;
    var btn = document.createElement("button");
    btn.type = "button";
    btn.className = "pin-parent-btn";
    stylePinButton(btn, mod.id);
    btn.addEventListener("click", function () { setPinnedParent(mod.id); });
    rt.pinBtn = btn;
    if (!rt.cardFooter && rt.cardBody.querySelector(".card-flush")) {
      // a migrated (flush) card whose module declares no primaryLink still
      // needs the footer — the flush body has no padding for a loose button
      var footer = document.createElement("div");
      footer.className = "card-footer";
      rt.block.insertBefore(footer, rt.cardBody.nextSibling);
      rt.cardFooter = footer;
    }
    if (rt.cardFooter) {
      // migrated cards: the pin sits left in the shell-owned footer, before
      // the right-aligned primary link (docs/CARD_RENDER_API.md)
      rt.cardFooter.insertBefore(btn, rt.cardFooter.firstChild);
    } else {
      rt.cardBody.appendChild(btn);
    }
  }
