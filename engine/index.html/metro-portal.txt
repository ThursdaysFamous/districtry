  /* =========================================================================
   * SIBLING-METRO HANDOFF — every instance knows where its siblings live
   * (fleet-outlines.json; METRO_EXPLORERS[].bbox for the search rows' coarse
   * pass). When the map centre, a selected point, a searched address or a
   * geolocation fix lands inside a sibling's area, the reader goes STRAIGHT
   * there, carrying the zoom they were at (and the point, when there is one)
   * so the map they arrive on is framed like the one they left.
   *
   * This was an easter egg until 2026-08-25: a card, a five-second countdown
   * and a "Stay" button. Retired at the operator's direction — it asked a
   * question whose answer was always yes, and made every reader who crossed a
   * state line watch a timer to get where they were already going.
   * ======================================================================= */

  /* A metro without a (well-formed) bbox simply opts out of the handoff —
   * it still gets its footer link, so new METRO_EXPLORERS entries can land
   * before their bbox is decided. */
  function validMetroBbox(b) {
    return !!b && isFinite(b.minLat) && isFinite(b.maxLat) &&
      isFinite(b.minLng) && isFinite(b.maxLng) &&
      b.minLat < b.maxLat && b.minLng < b.maxLng;
  }

  function pointInMetroBbox(lat, lng, b) {
    return lat >= b.minLat && lat <= b.maxLat && lng >= b.minLng && lng <= b.maxLng;
  }

  function siblingMetroAt(lat, lng) {
    // greater-metro bboxes may overlap once enough metros exist (two nearby
    // cities) — the nearest bbox center wins, not list order. Squared-degree
    // distance is plenty for a tie-break between neighbors.
    var best = null, bestDist = Infinity;
    for (var i = 0; i < METRO_EXPLORERS.length; i++) {
      var m = METRO_EXPLORERS[i];
      if (m.id === THIS_METRO || !validMetroBbox(m.bbox)) continue;
      if (!pointInMetroBbox(lat, lng, m.bbox)) continue;
      var dLat = lat - (m.bbox.minLat + m.bbox.maxLat) / 2;
      var dLng = lng - (m.bbox.minLng + m.bbox.maxLng) / 2;
      var dist = dLat * dLat + dLng * dLng;
      if (dist < bestDist) { best = m; bestDist = dist; }
    }
    return best;
  }

  /* The sibling's URL, framed like the view being left: the point when there
   * is one (a click, a search hit, a geolocation fix) and the zoom always. A
   * pan-triggered handoff carries zoom alone — the reader selected nothing, so
   * arriving with a point selected for them would answer a question they never
   * asked. */
  function metroPortalUrl(metro, point, zoom) {
    var parts = [];
    if (point) parts.push("point=" + point.lat.toFixed(5) + "," + point.lng.toFixed(5));
    if (isFinite(zoom)) parts.push("zoom=" + zoom);
    return parts.length ? metro.url + "#" + parts.join("&") : metro.url;
  }

  // One handoff per page load. Belt and braces beside the coverage guard
  // below: whatever happens, a reader can never be bounced twice by one view.
  var metroHandedOff = false;

  /* Send the reader to a sibling instance. opts: {point, source} — point
   * deep-links via #point=, source labels the trackEvent. (title/note were the
   * retired card's copy; callers may still pass them and they are ignored.) */
  function offerMetroPortal(metro, opts) {
    opts = opts || {};
    if (!metro || metroHandedOff) return false;
    var here = opts.point || null;
    // NEVER hand off a place this instance also answers for. Outlines meet at
    // shared borders, and a silent redirect on both sides of that line would
    // bounce a reader back and forth with nothing on screen to stop it. Every
    // caller has already asked placeOwner(); this is the last word on it.
    if (here && servedHereNow(here.lat, here.lng) === true) return false;
    metroHandedOff = true;
    trackEvent("metro-portal-go/" + metro.id, opts.source || "wander");
    window.location.href = metroPortalUrl(metro, here, map.getZoom());
    return true;
  }

  /* ---------- which app a place belongs to: outlines, not rectangles ----------
   * Until 2026-09-27 every decision below was made with bounding boxes: stay
   * if the point is inside THIS app's box, else go to the sibling whose box
   * holds it, nearest box centre winning. Simulated over 35 real places in all
   * six apps, 79 selections went to the wrong place, because a box is not a
   * state. Illinois's box reaches into Iowa, so Davenport did nothing from
   * Illinois; Michigan's runs west across the lake and held Chicago, Milwaukee
   * and Madison; overlapping sibling boxes sent Marquette to Wisconsin; and the
   * boxes claim places nobody serves, so Gary went to Illinois and Hartford to
   * New York.
   *
   * Two outlines decide it now. THIS app's own, at full detail, is the one the
   * coverage wash already retained (the region band where there is one, else
   * the coverage rings), so a selection inside it is answered here at once with
   * no fetch. Anything else is settled against ../fleet-outlines.json, every
   * live instance's widest outline simplified to about 200 m and generated by
   * scripts/build_fleet_outlines.py, fetched once and only when a selection
   * first leaves this app's own outline. Its own entry is asked first, so a
   * point on a border this app also draws stays here, and a point no live
   * instance covers stays here too, where the app says so, rather than being
   * sent to an app that has nothing to show. */
  var FLEET_OUTLINES_URL = "../fleet-outlines.json";
  var fleetOutlines = null;
  var fleetOutlinesLoading = null;

  function loadFleetOutlines() {
    if (fleetOutlines) return Promise.resolve(fleetOutlines);
    if (!fleetOutlinesLoading) {
      fleetOutlinesLoading = fetchJSONWithRetry(FLEET_OUTLINES_URL, {}, 1)
        .then(function (data) {
          if (!data || !data.outlines) throw new Error("fleet-outlines.json has no outlines");
          fleetOutlines = data.outlines;
          return fleetOutlines;
        })
        ["catch"](function (err) {
          fleetOutlinesLoading = null;  // a later selection may try again
          throw err;
        });
    }
    return fleetOutlinesLoading;
  }

  /* true / false from the outline this app drew its wash with; null while
   * that has not loaded, which is not an answer either way. */
  function servedHereNow(lat, lng) {
    var pt = { lat: lat, lng: lng };
    if (typeof coverageRegionPolys !== "undefined" && coverageRegionPolys && coverageRegionPolys.length) {
      return pointInGeometry(pt, { type: "MultiPolygon", coordinates: coverageRegionPolys });
    }
    if (typeof coverageMaskRings !== "undefined" && coverageMaskRings && coverageMaskRings.length) {
      return pointInGeometry(pt, { type: "Polygon", coordinates: coverageMaskRings });
    }
    return null;
  }

  /* The sibling whose fleet outline holds the point, or null. Two can only
   * both hold it along a shared border, where the simplified outlines overlap
   * by a few hundred metres; the nearer box centre settles that. */
  function fleetSiblingAt(lat, lng) {
    var pt = { lat: lat, lng: lng };
    var best = null, bestDist = Infinity;
    for (var i = 0; i < METRO_EXPLORERS.length; i++) {
      var m = METRO_EXPLORERS[i];
      if (m.id === THIS_METRO || !fleetOutlines[m.id]) continue;
      if (!pointInGeometry(pt, fleetOutlines[m.id])) continue;
      var dist = 0;
      if (validMetroBbox(m.bbox)) {
        var dLat = lat - (m.bbox.minLat + m.bbox.maxLat) / 2;
        var dLng = lng - (m.bbox.minLng + m.bbox.maxLng) / 2;
        dist = dLat * dLat + dLng * dLng;
      }
      if (dist < bestDist) { best = m; bestDist = dist; }
    }
    return best;
  }

  /* Resolves to "here" (this app answers there), a METRO_EXPLORERS entry (a
   * sibling does), or null (no live instance does). If the fleet file cannot
   * be read, only this app's own outline is left to go on: inside it is
   * "here", outside it is null, and with neither loaded this app's box is the
   * last resort. Nobody is sent to a sibling on a guess. */
  function placeOwner(lat, lng) {
    var own = servedHereNow(lat, lng);
    if (own === true) return Promise.resolve("here");
    return loadFleetOutlines().then(function () {
      if (fleetOutlines[THIS_METRO] &&
          pointInGeometry({ lat: lat, lng: lng }, fleetOutlines[THIS_METRO])) return "here";
      return fleetSiblingAt(lat, lng);
    }, function () {
      if (own === false) return null;
      return pointInMetroBbox(lat, lng, METRO_BBOX) ? "here" : null;
    });
  }

  /* A SELECTION: a map click, or the keyboard "select the centre" button.
   * `proceed` selects the point here, and runs unless the reader is sent on.
   * Returns true when it has taken over (the caller stops and `proceed` runs
   * later, if at all), false when the point is plainly this app's and the
   * caller should select it now.
   *
   * Panning already hands off on `moveend`, but only when the CENTRE crosses.
   * A reader looking at Illinois can click Wisconsin near the top of the map
   * without the centre ever leaving Illinois, which is exactly the case this
   * covers: the click carries the point, so they arrive with it selected
   * rather than merely somewhere in the neighbourhood. */
  var handoffSeq = 0;
  function handoffSelection(lat, lng, source, proceed) {
    if (metroHandedOff) return false;
    if (servedHereNow(lat, lng) === true) return false;       // ours; no fetch, no wait
    var seq = ++handoffSeq;
    placeOwner(lat, lng).then(function (owner) {
      if (seq !== handoffSeq) return;                         // a newer selection took over
      if (owner && owner !== "here" &&
          offerMetroPortal(owner, { point: { lat: lat, lng: lng }, source: source || "select" })) return;
      if (typeof proceed === "function") proceed();           // ours, or nobody's: the app says which
    });
    return true;
  }

  var wanderSeq = 0;
  map.on("moveend", function () {
    if (metroHandedOff) return;
    var c = map.getCenter();
    // Only once this app's own outline has loaded and says the centre has left
    // it. Before then, or while it is inside, the reader is looking at their
    // own map near its edge.
    if (servedHereNow(c.lat, c.lng) !== false) return;
    var seq = ++wanderSeq;
    placeOwner(c.lat, c.lng).then(function (owner) {
      if (seq !== wanderSeq || !owner || owner === "here") return;
      offerMetroPortal(owner, { source: "wander" });
    });
  });
