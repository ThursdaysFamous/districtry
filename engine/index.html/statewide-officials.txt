  /* ---------- Statewide officials (Adam's choice of 2026-11-10: name the
   * officers a state elects statewide, in every app).
   *
   * NOT A LAYER (Adam, 2026-11-10: "3 that can be expanded into 2"). The answer
   * is the same anywhere in the state, so a tick box and a shaded state said
   * nothing a reader needed. Instead, whenever a point inside the state is
   * selected, one line sits at the top of the results naming the governor and
   * how many more there are; pressing it opens the full list in place, and it
   * stays open for later points until it is closed again (kept in memory only,
   * so nothing is stored in the browser; Adam, 2026-11-10: "forget"). Opening
   * it is counted as statewide-officials/<tag>, by the app and never by place
   * (Adam, the same day: "count in traffic"); closing it is not counted.
   *
   * One call per app: registerStatewideOfficials({ outline }) where outline is
   * the app's widest outline file, the state's. The list is
   * data/app/statewide-officials.json, built by
   * scripts/build_statewide_officials.py, which reads each state officer's own
   * government page every run and refuses to write when the name is gone; it
   * keeps the date each name was last checked, which the list shows. An app
   * that makes no call (a state not live yet) shows nothing.
   *
   * runAllActiveLayerQueries (engine overlay-cards) calls syncStatewideOfficials
   * on every selection, so a new point updates the line and a point outside the
   * state hides it.
   * ---------- */
  var statewideOfficials = null; // { opts, loadOutline, loadRoster, open, el } once registered

  function registerStatewideOfficials(opts) {
    statewideOfficials = {
      opts: opts,
      open: false,
      el: null,
      loadOutline: makeCached(function () {
        return fetchJSONWithRetry(opts.outline, {}, 2);
      }),
      loadRoster: makeCached(function () {
        return fetchJSONWithRetry(opts.roster || "data/app/statewide-officials.json", {}, 2);
      })
    };
  }

  function statewidePointInside(geojson, point) {
    var feats = (geojson && geojson.features) || (geojson && geojson.geometry ? [geojson] : []);
    for (var i = 0; i < feats.length; i++) {
      if (feats[i] && feats[i].geometry && pointInGeometry(point, feats[i].geometry)) return true;
    }
    return false;
  }

  function statewidePersonRow(p, badge) {
    var row = { name: String(p.name), badge: badge };
    if (p.phone) row.phone = String(p.phone);
    if (p.email) row.email = String(p.email);
    if (p.url) row.links = [{ label: "Website", url: String(p.url) }];
    return row;
  }

  // The closed line: "Governor JB Pritzker and 7 more". Names the first
  // officer the list names, and counts everyone else on it, senators included.
  function statewideSummary(r) {
    var officers = r.officers || [], senators = r.usSenate || [];
    var total = officers.length + senators.length;
    var lead = null;
    for (var i = 0; i < officers.length && !lead; i++) {
      if (!officers[i].unnamed && officers[i].name) lead = officers[i];
    }
    if (!lead) return total + " officials elected by the whole state";
    var rest = total - 1;
    return String(lead.office) + " " + String(lead.name) +
      (rest > 0 ? " and " + rest + " more" : "");
  }

  function statewideBody(r) {
    var wrap = cardEl("div", "card-flush statewide-strip-body");
    var officers = r.officers || [];
    if (officers.length) {
      wrap.appendChild(renderSectionLabel("Elected by the whole state"));
      wrap.appendChild(renderPersonRows(officers.map(function (o) {
        // an office whose own website this app cannot read is listed and
        // linked, and named nowhere rather than from memory
        if (o.unnamed) return statewidePersonRow({ name: "Not listed", url: o.url }, String(o.office));
        return statewidePersonRow(o, String(o.office));
      })));
    }
    var senators = r.usSenate || [];
    if (senators.length) {
      wrap.appendChild(renderSectionLabel("U.S. Senate"));
      wrap.appendChild(renderPersonRows(senators.map(function (s) {
        return statewidePersonRow(s, s.party ? "U.S. Senator, " + String(s.party) : "U.S. Senator");
      })));
    }
    var fields = [];
    if (r.officersNote) fields.push({ label: "Offices", value: String(r.officersNote) });
    var why = [];
    officers.forEach(function (o) {
      var line = o.unnamed ? String(o.office) + ": " + String(o.unnamed) : null;
      if (line && why.indexOf(line) < 0) why.push(line);
    });
    if (why.length) fields.push({ label: "Not listed", value: why.join(" ") });
    var oldest = null;
    officers.forEach(function (o) {
      if (o.verified && (!oldest || String(o.verified) < oldest)) oldest = String(o.verified);
    });
    var read = [];
    if (oldest) {
      read.push("Each state officer's name was last checked against that " +
        "officer's own government website on " + statewideDate(oldest) + " or later.");
    }
    if (senators.length && r.usSenateRead) {
      read.push("The senators are from the U.S. Senate's list of senators, read " +
        statewideDate(String(r.usSenateRead)) + ".");
    }
    if (read.length) fields.push({ label: "Names read", value: read.join(" ") });
    if (fields.length) wrap.appendChild(renderFieldStack(fields));
    var src = r.usSenateSource ? cardExternalLink("U.S. Senate — senators by state", r.usSenateSource) : null;
    if (src) {
      var foot = cardEl("p", "statewide-strip-source");
      foot.appendChild(src);
      wrap.appendChild(foot);
    }
    return wrap;
  }

  // "2026-11-10" -> "November 10, 2026", the way the rest of the site writes a date
  function statewideDate(iso) {
    var m = /^(\d{4})-(\d{2})-(\d{2})$/.exec(String(iso));
    if (!m) return String(iso);
    var months = ["January", "February", "March", "April", "May", "June", "July",
      "August", "September", "October", "November", "December"];
    return months[parseInt(m[2], 10) - 1] + " " + parseInt(m[3], 10) + ", " + m[1];
  }

  function statewideBuild(r) {
    var sw = statewideOfficials;
    var el = sw.el;
    if (!el) {
      el = document.createElement("section");
      el.className = "statewide-strip";
      el.id = "statewide-strip";
      el.setAttribute("aria-label", "Statewide officials");
      sw.el = el;
    }
    el.textContent = "";
    var btn = document.createElement("button");
    btn.type = "button";
    btn.className = "statewide-strip-toggle";
    btn.setAttribute("aria-expanded", sw.open ? "true" : "false");
    btn.setAttribute("aria-controls", "statewide-strip-panel");
    btn.appendChild(cardEl("span", "statewide-strip-label",
      (r.state != null ? String(r.state) : "Statewide") + " officials"));
    btn.appendChild(cardEl("span", "statewide-strip-summary", statewideSummary(r)));
    btn.appendChild(cardEl("span", "statewide-strip-more", sw.open ? "Hide" : "Show all"));
    var panel = document.createElement("div");
    panel.id = "statewide-strip-panel";
    panel.className = "statewide-strip-panel";
    panel.hidden = !sw.open;
    if (sw.open) panel.appendChild(statewideBody(r));
    btn.addEventListener("click", function () {
      sw.open = !sw.open;
      if (sw.open) trackEvent("statewide-officials/" + METRO_BRAND.instanceTag);
      btn.setAttribute("aria-expanded", sw.open ? "true" : "false");
      btn.lastChild.textContent = sw.open ? "Hide" : "Show all";
      if (sw.open && !panel.firstChild) panel.appendChild(statewideBody(r));
      panel.hidden = !sw.open;
    });
    el.appendChild(btn);
    el.appendChild(panel);
    return el;
  }

  function statewidePlace(el) {
    var root = document.getElementById("groups-root");
    if (!root || !root.parentNode) return;
    var hint = document.getElementById("no-layer-hint");
    var before = hint && hint.parentNode === root.parentNode ? hint : root;
    if (el.parentNode !== root.parentNode || el.nextSibling !== before) {
      root.parentNode.insertBefore(el, before);
    }
  }

  function syncStatewideOfficials() {
    var sw = statewideOfficials;
    if (!sw) return;
    var point = state.selectedPoint;
    var seq = state.sequence;
    if (!point) {
      if (sw.el) sw.el.hidden = true;
      return;
    }
    Promise.all([sw.loadOutline(), sw.loadRoster()]).then(function (got) {
      if (seq !== state.sequence) return;
      if (!statewidePointInside(got[0], point) || !got[1]) {
        if (sw.el) sw.el.hidden = true;
        return;
      }
      var el = statewideBuild(got[1]);
      statewidePlace(el);
      el.hidden = false;
    }, function (err) {
      // the line is an extra: if either file will not load it stays away, and
      // every card below answers exactly as it would without it
      if (seq !== state.sequence) return;
      console.error("[statewide-officials] could not load:", err);
      if (sw.el) sw.el.hidden = true;
    });
  }
