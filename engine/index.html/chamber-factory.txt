  function registerIlgaChamber(opts) {
    // A TIGERweb legislative chamber (Congress / State Senate / State Assembly)
    // joined to a same-origin roster keyed by district number. opts: { id, label,
    // layerIndex, districtFields, loadRoster, memberLabel, style, directoryUrl,
    // profileLabel?, directoryLabel?, capitolLabel?, districtPrefix?,
    // seatsLabel?, seatsNote? }. Member rows are { name, party?, url?, phone?,
    // phonesExtra?, email?, districtOffice?, capitolOffice? }; office fields
    // only render when the roster supplies them (empty placeholder -> just the
    // link). A chamber that elects several members from each district keys
    // each district to { seats, members: [member rows] } instead (seatMembers
    // below), and seatsLabel/seatsNote word the line that says so.
    // Metro-provided geometry loader wins; forks that haven't pre-built their
    // chamber geometry fall back to the live TIGERweb fetch by layer index,
    // unchanged. (opts.loadDistricts is a metro-specific value passed by the
    // registration below — the engine only supplies the default.) R2-2.
    var loadChamber = opts.loadDistricts || makeCached(function () { return loadTigerLayer(opts.layerIndex); });
    // "IL‑14" where the concept is state-qualified (districtPrefix), plain
    // "District 14" otherwise — one string shared by the header pill and the
    // roster-less card body so the two can never disagree.
    function districtLabel(result) {
      if (!result || !result.district) return null;
      return opts.districtPrefix
        ? opts.districtPrefix + String(result.district)
        : "District " + String(result.district);
    }
    // Split a roster office's raw lines (address lines + phone/fax lines
    // mixed, exactly as scraped) into the renderOfficeGroup shape: the first
    // phone-looking non-fax line becomes the tel:-linked phone (a leading
    // "Phone:" prefix dropped); fax and every other line stay as text lines
    // in scraped order, so no scraped detail is silently discarded.
    // A LABELLED LINE WINS OVER A LINE THAT MERELY CONTAINS DIGITS, and that
    // ordering is the whole of the 2026-09-19 fix. The previous single pass
    // tested `\d{3}[-.\s]\d{4}`, which a ZIP+4 satisfies — "245 Cannon House
    // Office Building Washington DC 20515-3210" contains "515-3210" — and the
    // first match won, so the ADDRESS became the phone, got a tel: link built
    // from its own digits (tel:245205153210), and the "Phone: 202-225-7944"
    // line below it was demoted to a plain text line. Measured across the
    // fleet's shipped rosters that day: 84 of 1,454 office blocks rendered an
    // address as a dialable number — every New York congressional D.C. office
    // (30 blocks), 53 in San Francisco, 1 in Wisconsin.
    //
    // The two passes are ordered by how much the line PROVES, not by where it
    // sits, because the defect was entirely one of position. Measured on the
    // same 1,454 blocks: 1,198 lines carry an explicit label and ALL 1,198
    // have a bare number after it; 354 more are a bare number with no label.
    // So a labelled line is required to leave a number behind, and a line that
    // is a number and nothing else is the only unlabelled candidate — a street
    // address satisfies neither.
    function splitOfficeLines(lines) {
      if (!lines || !lines.length) return null;
      var LABEL = /^\s*(?:phone|telephone|tel)\b[:.\s-]*/i;
      var FAX = /fax/i;
      var NUMBER_ONLY = /^\+?1?[\s.\-()]*\d{3}[\s.\-()]*\d{3}[\s.\-]*\d{4}(?:\s*(?:x|ext\.?|extension)[\s.]*\d+)?$/i;
      var office = { lines: [], phone: null };
      var phoneIndex = -1;
      // Pass 1: an explicitly labelled line, wherever in the block it sits.
      lines.forEach(function (raw, i) {
        if (phoneIndex >= 0) return;
        var line = String(raw);
        if (FAX.test(line) || !LABEL.test(line)) return;
        var rest = line.replace(LABEL, "").trim();
        if (!NUMBER_ONLY.test(rest)) return;
        office.phone = rest;
        phoneIndex = i;
      });
      // Pass 2: only when nothing was labelled, a line that is a number and
      // nothing else. Never a line that merely contains one.
      if (phoneIndex < 0) {
        lines.forEach(function (raw, i) {
          if (phoneIndex >= 0) return;
          var line = String(raw).trim();
          if (FAX.test(line) || !NUMBER_ONLY.test(line)) return;
          office.phone = line;
          phoneIndex = i;
        });
      }
      lines.forEach(function (raw, i) {
        if (i !== phoneIndex) office.lines.push(String(raw));
      });
      if (!office.lines.length && !office.phone) return null;
      return office;
    }
    function memberProfileUrl(member) {
      return member && member.url ? safeHttpUrl(member.url) : null;
    }
    // A SEAT THE CHAMBER ITSELF CALLS EMPTY IS A RECORD, NOT AN ABSENCE. Until
    // 2026-10-01 a roster built from a bulk export simply had no row for a
    // vacant district, so three things went wrong at once and all three were
    // reader-facing: the card fell through to the no-roster path and said only
    // the district number, the chamber's own seat count was taken from the
    // roster's length and read one short (the generated table on each
    // legislature page said "All 79 seats on the California State Assembly"
    // for a chamber of 80), and nothing anywhere said why. The roster carries a
    // record for every district the map draws now, and the record says which of
    // two different things is true.
    //
    // THE TWO ARE KEPT APART BECAUSE ONE IS A STRONGER STATEMENT THAN THE
    // OTHER. `vacant: true` means the chamber's own members page says the seat
    // is empty — California's prints "Member Vacant" — and that is the only way
    // a reader is ever told "Vacant". A record with no name and no `vacant`
    // means our source named nobody, which is a fact about our source, so it
    // reads "No member listed". Reporting the second as a vacancy would be
    // guessing an officeholder out of existence, which is the same honesty rule
    // that forbids guessing one into it.
    function isMember(member) {
      return !!(member && member.name);
    }
    // A DISTRICT THAT ELECTS SEVERAL MEMBERS. Arizona's House of
    // Representatives and New Jersey's General Assembly elect two members from
    // each district, so a roster record for such a chamber is
    // { seats: N, members: [member rows] } rather than one member row.
    // scripts/validate_chamber_rosters.py holds that shape: `seats` equals the
    // builder's per_district, every listed member carries a name, and a
    // district listing fewer members than it elects carries asOf, because the
    // card then says how many the roster listed and on what day. A one-member
    // record in such a chamber returns null here and reads exactly as before.
    // West Virginia's Senate, two senators per district, predates this branch
    // and keeps a layer block of its own.
    function seatMembers(seat) {
      if (!seat || !Array.isArray(seat.members)) return null;
      return seat.members.filter(isMember);
    }
    // "A", "A and B", "A, B and C": the hover names everyone the card names.
    function joinNames(names) {
      if (names.length < 2) return names.join("");
      return names.slice(0, -1).join(", ") + " and " + names[names.length - 1];
    }
    // THE RECORD STATES THE CLAIM AND THIS ONLY PRINTS IT. Where the authority
    // does not use the word, the record carries its own one-sentence reason:
    // Iowa's chamber roster ends a member's service with a date and names
    // nobody after them, so a hardcoded "its members page says this seat is
    // vacant" would be false of the very page cited. The default sentence
    // serves the chambers that do print the word.
    function vacancyNote(member) {
      // `asOf` is the day the chamber said it, carried so a reader can see how
      // old the statement is. A record with no date says the thing without
      // dating it rather than inventing a day.
      var when = member && member.asOf ? String(member.asOf) : null;
      var why = member && member.vacantWhy
        ? String(member.vacantWhy)
        : "The chamber's own members page says nobody holds this seat.";
      return "Vacant — " + why + (when ? " Read on " + when + "." : "");
    }
    // One member's row: name, party badge, own telephone and e-mail, and
    // their page. Shared by the one-member card and every row of a
    // multi-member district, so the two cannot word a member differently.
    function personRow(member) {
      var person = { name: String(member.name) };
      if (member.party) person.badge = String(member.party);
      // THE MEMBER'S OWN TELEPHONE, where the roster carries one: a number
      // the chamber prints for the member apart from the office blocks
      // below, such as a home or cell line. Published since Adam's ruling of
      // 2026-10-17 that a number a government source prints beside an
      // official is shown even where it may be the official's own; a label
      // the source prints ("home 803-555-0100") rides inside the string.
      if (member.phone) person.phone = String(member.phone);
      if (member.phonesExtra && member.phonesExtra.length) {
        person.phonesExtra = member.phonesExtra.map(String);
      }
      // THE MEMBER'S E-MAIL ADDRESS, where the roster carries one. Before
      // 2026-10-17 this card showed no legislator's address at all, though
      // eight states' rosters carried one for nearly every seat. Measured
      // that day, every one of them is the chamber's own mailbox except
      // fifteen Arkansas members' personal addresses, which the
      // legislature's own member pages print and which Adam chose to show
      // ("E-mails too"). A roster reads it from the legislature's pages or
      // from Open States' compilation of them.
      if (member.email) person.email = String(member.email);
      // Label follows the actual href: only a roster-supplied member URL
      // gets the per-member label — the directory fallback lives in the
      // footer (primaryLink) and must not masquerade as the member's page.
      var memberUrl = memberProfileUrl(member);
      if (memberUrl) person.links = [{ label: opts.profileLabel || "Member profile", url: memberUrl }];
      return person;
    }
    // One member's offices as office blocks, each kicker led by `prefix`
    // (empty on a one-member card, the member's name on a multi-member one).
    function memberOffices(member, prefix) {
      // District office first (it's the local, on-map one this card pins),
      // Springfield/D.C. second — same order the old key/value card used.
      var offices = [];
      var districtOffice = splitOfficeLines(member.districtOffice);
      if (districtOffice) { districtOffice.label = prefix + "District Office"; offices.push(districtOffice); }
      var capitolOffice = splitOfficeLines(member.capitolOffice || member.springfieldOffice);
      if (capitolOffice) { capitolOffice.label = prefix + (opts.capitolLabel || "Capitol Office"); offices.push(capitolOffice); }
      return offices;
    }
    registerLayer({
      id: opts.id,
      group: "political",
      label: opts.label,
      // hover popup: same district extraction the card's query uses (chamber-
      // specific TIGERweb fields + the name-field regex fallback).
      hoverName: function (feature) {
        return extractDistrictNumber(feature.properties || {}, opts.districtFields);
      },
      overlay: { load: loadChamber, style: opts.style },
      tiles: opts.tiles, // optional vector-tile archive (tile-overlay block)
      // boundary resolution is point-first by default (queryFeatureAt); the
      // roster stays a whole-file fetch — it's the small half of the join
      query: function (point, seq) {
        return Promise.all([
          queryFeatureAt(loadChamber, point),
          opts.loadRoster().catch(function () { return {}; })
        ]).then(function (results) {
          var feature = results[0], roster = results[1];
          if (!feature) return null;
          var props = feature.properties || {};
          var district = extractDistrictNumber(props, opts.districtFields);
          var member = district != null ? roster[String(district)] : undefined;
          var members = seatMembers(member);
          // A nameless record is NOT a member and must not reach the person
          // row, which would render the string "undefined" as somebody's name.
          // It is carried separately so the card can say which kind it is.
          return { seq: seq, district: district,
                   member: !members && isMember(member) ? member : null,
                   members: members,
                   seat: member || null };
        });
      },
      render: function (result) {
        // 4a representative card (docs/CARD_RENDER_API.md): person row with a
        // party badge, then the offices behind one <details> group. District
        // identity lives in the header pill (cardIdentifier below).
        var wrap = cardEl("div", "card-flush");
        var member = result.member;
        if (result.members) {
          // Every member the roster lists, each with the same row a one-member
          // district gets, their offices in one group (each kicker naming its
          // member), and a line saying how many the district elects, so a
          // reader shown two names knows both represent the point clicked.
          var listed = result.members;
          if (listed.length) wrap.appendChild(renderPersonRows(listed.map(personRow)));
          var groupOffices = [];
          listed.forEach(function (m) {
            groupOffices = groupOffices.concat(memberOffices(m, String(m.name) + " · "));
          });
          if (groupOffices.length) wrap.appendChild(renderOfficeGroup({ offices: groupOffices }));
          var record = result.seat || {};
          var seats = Number(record.seats) || listed.length;
          var shortNote = "";
          if (listed.length < seats) {
            shortNote = "The roster lists " +
              (listed.length ? String(listed.length) + " of the " : "none of the ") +
              String(seats) + " members this district elects" +
              (record.asOf ? " (read on " + String(record.asOf) + ")" : "") + ", so " +
              (listed.length === 1 ? "only that one is" : listed.length ? "only those are" : "no one is") +
              " named here. ";
          }
          wrap.appendChild(renderFieldStack([{
            label: opts.seatsLabel || "Members",
            value: shortNote + (opts.seatsNote ||
              "This district elects " + String(seats) + " members, and each represents every point in it.")
          }]));
          return wrap;
        }
        if (!member) {
          // Three states, told apart. A seat the chamber calls vacant says so;
          // a record naming nobody says that instead; and no record at all is
          // either a roster gap or a roster-fetch failure, which is OUR silence
          // rather than the chamber's, so it claims neither and shows district
          // identity plus the official directory (primaryLink in the footer) —
          // never an invented name (honesty rules).
          var seat = result.seat;
          var note = null;
          if (seat && seat.vacant) note = vacancyNote(seat);
          else if (seat) note = "No member listed for this seat.";
          wrap.appendChild(renderBodyIntro({
            title: districtLabel(result) || "Unknown district",
            note: note
          }));
          return wrap;
        }
        wrap.appendChild(renderPersonRows([personRow(member)]));
        var offices = memberOffices(member, "");
        if (offices.length) wrap.appendChild(renderOfficeGroup({ offices: offices }));
        return wrap;
      },
      cardIdentifier: districtLabel,
      primaryLink: function (result) {
        // member.url is this factory's one external-URL sink; scheme-check it
        // through safeHttpUrl like every other href, falling back to the
        // trusted (hardcoded) directory URL when the roster supplies no
        // usable http(s) link.
        var memberUrl = memberProfileUrl(result.member);
        var url = memberUrl || opts.directoryUrl || opts.ilgaUrl;
        if (!url) return null;
        return {
          label: memberUrl ? (opts.profileLabel || "Member profile") : (opts.directoryLabel || "Member directory"),
          url: url
        };
      },
      pointOfInterest: function (result) {
        var member = result.member;
        if (!member) return null;
        var address = officeAddressForGeocode(member.districtOffice);
        if (!address) return null;
        return { label: member.name, address: address };
      },
      // hover popup: name the senator / state representative from the prefetched
      // roster (keyed by district number, same as the query).
      hoverOfficial: {
        load: opts.loadRoster,
        name: function (feature, roster) {
          if (!roster) return null;
          var d = extractDistrictNumber(feature.properties || {}, opts.districtFields);
          var member = d != null ? roster[String(d)] : null;
          var members = seatMembers(member);
          if (members) {
            return members.length ? joinNames(members.map(function (m) { return String(m.name); })) : null;
          }
          if (isMember(member)) return member.name;
          // The hover names who holds the seat, so a seat the chamber calls
          // empty is named "Vacant" — the hover and the card cannot disagree.
          // A record naming nobody hovers as nothing, exactly as a missing
          // record always has, because "no member listed" is a sentence about
          // our source and does not belong in a one-word popup.
          return member && member.vacant ? "Vacant" : null;
        }
      }
    });
  }
