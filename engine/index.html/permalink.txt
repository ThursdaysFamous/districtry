  /* =========================================================================
   * SHAREABLE URL — #p=lat,lng&l=code1,code2&c=code&s=1&z=N mirrors the
   * current selection so any view can be copied out of the URL bar (or via the
   * point chip's Copy-link button) and restored on load. replaceState, not
   * location.hash assignment, so clicking around never pollutes the back
   * button. Parsed once at boot by applyStateFromHash().
   *
   * THE SHORT FORM SINCE 2026-10-09. Every link until then spelled the view
   * out — #point=41.88250,-87.62850&layers=ward,congress&pin=…&stats=1&zoom=N
   * — and a typical shared link ran 96 to 140 characters, 652 with every
   * Illinois layer on. Layers are now written as their permanent short codes
   * (LAYER_CODES, generated from the fleet-wide layer-codes.json), the keys
   * are one letter, and a coordinate drops its trailing zeros while keeping
   * all five decimals, because a shared link has to land on the same side of
   * a boundary as the click that made it. THE LONG FORM IS STILL READ, and
   * must be forever: those links sit in messages, posts and print nobody can
   * recall. readPermalink() is the one reading of both.
   * ======================================================================= */

  // Layer id <-> link code. Guarded, so an instance whose metro config carries
  // no LAYER_CODES writes full ids exactly as before.
  var layerCodeOf = {}, layerIdOfCode = {};
  if (typeof LAYER_CODES === "object" && LAYER_CODES) {
    Object.keys(LAYER_CODES).forEach(function (id) {
      layerCodeOf[id] = LAYER_CODES[id];
      layerIdOfCode[LAYER_CODES[id]] = id;
    });
  }
  function layerLinkCode(id) { return layerCodeOf[id] || id; }
  // A token from `l=` or `c=`: a code, or — hand-typed, or from an instance
  // with no table — a full id. No code spells an id (generate_metro_files.py
  // refuses one), so the two cannot be confused.
  function layerFromLinkCode(token) { return layerIdOfCode[token] || token; }

  // Five decimals (about a metre), trailing zeros dropped: 41.88250 -> 41.8825.
  function linkCoord(n) { return String(+n.toFixed(5)); }

  // ONE reading of a map link's hash (without the "#"), in either spelling,
  // for the boot parse and for the smoke tests. Long keys are read verbatim
  // as ids; short keys go through the code table. A key in both spellings —
  // nothing writes that — takes the short one, and `layers=` with `l=` is
  // their union, so nothing a link names is dropped.
  function readPermalink(raw) {
    var params = {};
    (raw || "").split("&").forEach(function (kv) {
      var eq = kv.indexOf("=");
      if (eq <= 0) return;
      try {
        params[kv.slice(0, eq)] = decodeURIComponent(kv.slice(eq + 1));
      } catch (e) {
        // a malformed %-escape drops that one parameter, not the whole link
      }
    });
    function pick(shortKey, longKey) {
      return params[shortKey] !== undefined ? params[shortKey] : params[longKey];
    }
    var layers = [], seen = {};
    function addLayer(id) {
      if (id && !seen[id]) { seen[id] = true; layers.push(id); }
    }
    if (params.layers) params.layers.split(",").forEach(addLayer);
    if (params.l) params.l.split(",").forEach(function (t) { if (t) addLayer(layerFromLinkCode(t)); });
    return {
      point: pick("p", "point"),
      layers: layers,
      pin: params.c !== undefined ? layerFromLinkCode(params.c) : params.pin,
      stats: pick("s", "stats"),
      zoom: pick("z", "zoom"),
      q: params.q
    };
  }

  // The zoom this fork boots at, read from the map rather than configured, so
  // the engine needs no per-metro constant and every fork gets its own default.
  // Safe to capture here: nothing between map creation and this line changes the
  // zoom, and applyStateFromHash() runs at the end of boot.
  var bootZoom = map.getZoom();

  function syncUrlHash() {
    var parts = [];
    if (state.selectedPoint) {
      parts.push("p=" + linkCoord(state.selectedPoint.lat) + "," + linkCoord(state.selectedPoint.lng));
    }
    var on = layers.filter(function (l) { return state.layersOn[l.id]; })
      .map(function (l) { return layerLinkCode(l.id); });
    if (on.length) parts.push("l=" + on.join(","));
    // The PINNED PARENT rides the permalink too, because the relationship
    // outlines ARE part of the view being shared: a link that restores the
    // point and the layers and drops the pin hands the reader the same map
    // without the comparison the sender was looking at, and nothing on screen
    // says a pin was ever set. Emitted only while its layer is on, which is
    // also why the pin (`c=`, for compare) can never be the ONLY parameter in
    // a hash — the same condition puts its layer in `on` above, so `l=` is
    // always emitted beside it.
    if (pinnedParentId && state.layersOn[pinnedParentId]) {
      parts.push("c=" + layerLinkCode(pinnedParentId));
      // The comparison stats report, while it is open (compare-stats). Only
      // beside a pin, because the report is OF the pinned district.
      if (statsIsOpen()) parts.push("s=1");
    }
    // Zoom rides the permalink because it is part of what was shared: an embed of
    // a whole county answers a different question from an embed of one block, and
    // a refresh should not throw the reader back to the default view. Emitted
    // ONLY when the user has moved off the boot zoom, so an untouched map still
    // produces a clean, hash-free URL.
    if (map.getZoom() !== bootZoom) parts.push("z=" + map.getZoom());
    var hash = parts.length ? "#" + parts.join("&") : "";
    try {
      history.replaceState(null, "", hash || location.pathname + location.search);
    } catch (e) {
      // some file:// contexts refuse replaceState — losing the permalink
      // there is fine, everything else keeps working
    }
  }

  // Point and layer changes call syncUrlHash themselves; zoom has no such hook,
  // because a zoom changes no app state — only the view. zoomend fires once per
  // gesture (not per animation frame), so this is a handful of replaceState calls.
  map.on("zoomend", syncUrlHash);

  function applyStateFromHash() {
    var raw = location.hash.replace(/^#/, "");
    if (!raw) return;
    var params = readPermalink(raw);
    params.layers.forEach(function (id) {
      var box = document.getElementById("toggle-" + id);
      if (box && !box.checked) box.click(); // a real click, so the change handler runs
    });
    // Parsed BEFORE the point, because the point's pan has to know about it.
    // An unparseable or out-of-range zoom is ignored rather than clamped: a link
    // from a fork with different limits should fall back to this fork's default
    // view, not to its nearest edge.
    var zoom = null;
    if (params.zoom !== undefined) {
      // parseFloat, not parseInt: zoomSnap 0.5 makes half steps legal on the
      // statewide instances, and a sibling handoff carries the zoom the reader
      // was at. parseInt("7.5") is 7, which silently dropped half a level on
      // every hop between two apps.
      var wanted = parseFloat(params.zoom);
      if (isFinite(wanted) && wanted >= map.getMinZoom() && wanted <= map.getMaxZoom()) zoom = wanted;
    }
    var framed = false;
    if (params.point) {
      var pieces = params.point.split(",");
      var lat = parseFloat(pieces[0]), lng = parseFloat(pieces[1]);
      // sanity-bound to the greater metro area this app serves (PERMALINK_GATE)
      if (isFinite(lat) && isFinite(lng) &&
          lat > PERMALINK_GATE.minLat && lat < PERMALINK_GATE.maxLat &&
          lng > PERMALINK_GATE.minLng && lng < PERMALINK_GATE.maxLng) {
        // skipPan when the link carries its own zoom, then frame it here.
        // setSelectedPoint's pan is ANIMATED and animates to at least zoom 15;
        // an animation in flight settles AFTER anything applied synchronously
        // afterwards, so a shared zoom set post-hoc is silently overwritten by
        // the pan that was already running. Measured, not theorised — that is
        // exactly what #point=…&zoom=9 did before this branch existed.
        setSelectedPoint(lat, lng, zoom === null ? null : { skipPan: true });
        if (zoom !== null) {
          map.setView([lat, lng], zoom, { animate: false });
          framed = true;
        }
      }
    }
    if (zoom !== null && !framed) map.setView(map.getCenter(), zoom, { animate: false });
    // Parsed LAST, after the layers are on and the point is set, because a pin
    // is a reference to one of those layers AT that point. A pin that cannot
    // be honoured is DROPPED SILENTLY rather than half-applied — a legend
    // naming a parent whose outlines can never be drawn is worse than no pin,
    // because it reads as present and does nothing. Three ways that happens,
    // and all three are ordinary rather than exceptional: the id names no
    // layer this instance registers (a link from a sibling that has it, or
    // from before a rename), its layer is not on (the `layers=` list disagreed
    // with the pin, or a toggle refused), and it is not a polygon layer — the
    // same test appendPinControl applies before it offers the control at all,
    // so a hand-typed `pin=fire-station` restores nothing.
    //
    // Setting the pin here with no geometry loaded yet is correct and not a
    // race: refreshRelationshipOutlines runs again as each layer's query
    // resolves, so the outlines appear when the parent's boundaries arrive,
    // and the card's own button picks up the pinned styling from
    // pinnedParentId whenever that card next renders.
    if (params.pin) {
      var parent = getLayerById(params.pin);
      if (parent && state.layersOn[parent.id] &&
          parent.overlay && parent.overlay.style) {
        setPinnedParent(parent.id);
        // `s=1` (`stats=1` in the long form): reopen the comparison report once the pinned district
        // and the other layers' boundaries have loaded (statsRequestRestore).
        // Only when the pin itself was restored — a report of nothing is
        // not opened.
        if (params.stats === "1") statsRequestRestore();
      }
    }
    // #q=<address> — an address handed over by a question page's lookup
    // form; prefillGeocodeSearch runs the app's own bounded search on it.
    // A link carrying BOTH a point and a query keeps the point: it is
    // already an exact answer, and re-geocoding could move it. It sits after
    // the pin because it is not a state parse — it starts a search from the
    // state the parses above have already applied.
    if (params.q && !params.point) prefillGeocodeSearch(params.q);
  }

  /* ---------- share controls — the point chip's single "Share" surface -----
   * One button replaces the old Copy-link / Embed pill pair: "Share" names
   * the INTENT, and the popover behind it names each mechanism's OUTCOME
   * ("the link opens the map at this exact point, with the layers you have
   * on") — which two words on a pill next to a lat,lng readout never could;
   * the old label read as "copy these coordinates" as easily as "share this
   * view". Touch-first devices get the platform's native share sheet when
   * the browser offers one; hover-capable devices get the popover with the
   * permalink, the <iframe> snippet (visible BEFORE it is copied — no more
   * blind clipboard write), and the coordinates as their own copyable row.
   *
   * Every URL handed out carries a SOURCE TAG before the #hash, where the
   * hash router never sees it: ?ref=share for a copied or sent link, ?ref=qr
   * for the QR code, ?ref=embed for the iframe. GoatCounter's count.js sends
   * location.search with every hit and its server takes the first of
   * utm_source, ref, src and source as the referrer, while its canonical-link
   * path normalization keeps the recorded path clean.
   *
   * IT WAS ?utm_source=share&utm_medium=link|qr UNTIL 2026-10-09, and the
   * medium half never reached the counter: GoatCounter keeps no utm_medium,
   * so a scan and a pasted link both read "share" there, and only Google
   * Analytics — on the two apps that run it — could tell them apart. One
   * short tag per source is 25 characters shorter AND splits them in the
   * counter every app runs. The cost is on those two apps: gtag reads only
   * utm_* tags, so a shared arrival no longer carries a share label in Google
   * Analytics. Organic traffic stays untagged, and a re-share from a tagged
   * page rebuilds from origin+pathname, so an inbound tag is never copied
   * forward as the next reader's source.
   *
   * Everything interpolated is our own trusted content — canonical fork URL
   * (METRO_EXPLORERS), numeric coords, registry layer ids, configured
   * METRO_NAME — so the strings are safe verbatim, the same argument the
   * embed snippet carried as fork code. Values are built at OPEN/COPY time
   * from live state, never cached in the DOM, so the popover cannot hand
   * out a stale view. Forks wire it with one line in their setSelectedPoint:
   * pointChip.appendChild(buildShareControl()). ---------- */

  function shareCanonicalBase() {
    for (var i = 0; i < METRO_EXPLORERS.length; i++) {
      if (METRO_EXPLORERS[i].id === THIS_METRO) return METRO_EXPLORERS[i].url;
    }
    return location.origin + location.pathname; // preview/localhost fallback
  }

  // The shareable link: CURRENT origin (a preview shares the preview, exactly
  // like the old Copy-link did) + the source tag + the live hash. Any inbound
  // search — including someone else's tags — is deliberately dropped.
  function shareLinkUrl() {
    syncUrlHash(); // reflect the very latest point/layers/zoom first
    return location.origin + location.pathname + "?ref=share" + location.hash;
  }

  // The QR's own URL. Same view, but `ref=qr`, because a scan and a pasted
  // link are different sources and a feature nobody can measure is one nobody
  // can evaluate.
  function shareQrUrl() {
    syncUrlHash();
    return location.origin + location.pathname + "?ref=qr" + location.hash;
  }

  function buildEmbedSnippet() {
    syncUrlHash();
    // Canonical base, not location: a snippet grabbed on a preview/localhost
    // must still embed production. width:100% + max-width keeps it
    // responsive; loading=lazy so it never blocks the host page;
    // allow=geolocation keeps the "locate me" button working inside.
    var url = shareCanonicalBase().replace(/[?#].*$/, "") +
      "?ref=embed" + location.hash;
    return '<iframe src="' + url + '" width="100%" height="600" ' +
      'style="border:0;max-width:100%" loading="lazy" ' +
      'title="' + appDisplayName() + '" ' +
      'allow="geolocation" referrerpolicy="no-referrer-when-downgrade"></iframe>';
  }

  function copyTextToClipboard(text) {
    if (navigator.clipboard && navigator.clipboard.writeText) {
      return navigator.clipboard.writeText(text).then(function () { return true; }, function () { return false; });
    }
    return Promise.resolve(false); // clipboard API unavailable (e.g. some file:// contexts)
  }

  // A copy button that reports its outcome in place ("Copied!" / "Copy
  // failed") and reverts — the same feedback contract the old pills had.
  function shareCopyButton(label, getText, eventName) {
    var btn = document.createElement("button");
    btn.type = "button";
    btn.className = "share-copy-btn";
    btn.textContent = label;
    btn.addEventListener("click", function () {
      trackEvent(eventName);
      copyTextToClipboard(getText()).then(function (ok) {
        btn.textContent = ok ? "Copied!" : "Copy failed";
        setTimeout(function () { btn.textContent = label; }, 1600);
      });
    });
    return btn;
  }

  function shareCoordsText() {
    if (!state.selectedPoint) return "";
    return state.selectedPoint.lat.toFixed(5) + ", " + state.selectedPoint.lng.toFixed(5);
  }

  // The chip's share control (button + popover). Rebuilt with the rest of
  // the chip on every selection, so an open popover never outlives the
  // point it described; the document-level listeners self-clean on the next
  // keydown/pointerdown even when the chip is torn down mid-open.
  function buildShareControl() {
    var wrap = document.createElement("span");
    wrap.className = "share-control";

    var btn = document.createElement("button");
    btn.type = "button";
    btn.className = "copy-link-btn share-btn";
    btn.textContent = "Share";
    btn.title = "Share this view — link, embed code, or coordinates";
    btn.setAttribute("aria-haspopup", "dialog");
    btn.setAttribute("aria-expanded", "false");
    wrap.appendChild(btn);

    var popover = null;
    function closePopover(refocus) {
      if (!popover) return;
      document.removeEventListener("keydown", onKeydown, true);
      document.removeEventListener("pointerdown", onOutside, true);
      if (popover.parentNode) popover.parentNode.removeChild(popover);
      popover = null;
      btn.setAttribute("aria-expanded", "false");
      if (refocus) btn.focus();
    }
    function onKeydown(e) {
      if (e.key === "Escape") closePopover(true);
    }
    function onOutside(e) {
      if (popover && !popover.contains(e.target) && !wrap.contains(e.target)) closePopover(false);
    }
    function selectOnFocus() { this.select(); }

    function openPopover() {
      popover = document.createElement("div");
      popover.className = "share-popover";
      popover.setAttribute("role", "dialog");
      popover.setAttribute("aria-label", "Share this view");

      var head = document.createElement("div");
      head.className = "share-popover-head";
      var title = document.createElement("strong");
      title.className = "share-popover-title";
      title.textContent = "Share this view";
      var close = document.createElement("button");
      close.type = "button";
      close.className = "share-popover-close";
      close.textContent = "✕";
      close.setAttribute("aria-label", "Close share panel");
      close.addEventListener("click", function () { closePopover(true); });
      head.appendChild(title);
      head.appendChild(close);
      popover.appendChild(head);

      var note = document.createElement("p");
      note.className = "share-popover-note";
      note.textContent = "The link opens the map at this exact point, with the layers you have on.";
      popover.appendChild(note);

      // THE QR IS NOT THEMED: dark modules on white with a white quiet zone in
      // both tiers, because an inverted QR fails on many scanners. It is built
      // through the DOM rather than as markup, and encoded here — no request is
      // made and nothing is sent, which is the whole reason it is not an image
      // from a QR service (see engine/index.html/qr-code.txt).
      // A touch device keeps its platform share sheet, as the first thing in the
      // popover rather than instead of it: the popover is now the one share
      // surface everywhere, because the QR has to be reachable on a phone too
      // and the OS sheet cannot host it. One extra tap for the native path is
      // the cost, and it is the reason this row is first.
      if (!hoverCapable && navigator.share) {
        var nativeRow = document.createElement("div");
        nativeRow.className = "share-popover-row share-popover-row-end";
        var nativeBtn = document.createElement("button");
        nativeBtn.type = "button";
        nativeBtn.className = "share-popover-copy";
        nativeBtn.textContent = "Share\u2026";
        nativeBtn.addEventListener("click", function () {
          trackEvent("share-native");
          navigator.share({ title: appDisplayName(), url: shareLinkUrl() })
            .catch(function () { /* dismissed, or the platform refused — nothing to undo */ });
        });
        nativeRow.appendChild(nativeBtn);
        popover.appendChild(nativeRow);
      }

      var qrLabel = document.createElement("div");
      qrLabel.className = "share-popover-label";
      qrLabel.textContent = "Scan with a phone";
      popover.appendChild(qrLabel);
      var qrRow = document.createElement("div");
      qrRow.className = "share-popover-qr";
      var drawn = qrSvgElement(shareQrUrl(), { level: "M" });
      if (drawn) {
        drawn.element.setAttribute("aria-label", "QR code for this view");
        qrRow.appendChild(drawn.element);
        popover.appendChild(qrRow);
        trackEvent("share-qr");
      } else {
        // Unreachable for any view this app can produce — the longest share URL
        // measured 199 bytes on 2026-10-09 (all 41 Illinois layers, a pin, the
        // stats flag) against version 40's 2,953 — but a reader is told rather
        // than shown an empty box if it ever happens.
        var qrNote = document.createElement("p");
        qrNote.className = "share-popover-note";
        qrNote.textContent = "This view's link is too long for a QR code. Copy it instead.";
        popover.appendChild(qrNote);
      }

      var linkRow = document.createElement("div");
      linkRow.className = "share-popover-row";
      var linkInput = document.createElement("input");
      linkInput.type = "text";
      linkInput.readOnly = true;
      linkInput.className = "share-popover-url";
      linkInput.value = shareLinkUrl();
      linkInput.setAttribute("aria-label", "Shareable link");
      linkInput.addEventListener("focus", selectOnFocus);
      linkRow.appendChild(linkInput);
      linkRow.appendChild(shareCopyButton("Copy link", shareLinkUrl, "share-permalink"));
      popover.appendChild(linkRow);

      var embedLabel = document.createElement("div");
      embedLabel.className = "share-popover-label";
      embedLabel.textContent = "Embed on your site";
      popover.appendChild(embedLabel);
      var embedBox = document.createElement("textarea");
      embedBox.readOnly = true;
      embedBox.rows = 3;
      embedBox.className = "share-popover-embed";
      embedBox.value = buildEmbedSnippet();
      embedBox.setAttribute("aria-label", "Embed code");
      embedBox.addEventListener("focus", selectOnFocus);
      popover.appendChild(embedBox);
      var embedRow = document.createElement("div");
      embedRow.className = "share-popover-row share-popover-row-end";
      embedRow.appendChild(shareCopyButton("Copy embed code", buildEmbedSnippet, "embed-iframe"));
      popover.appendChild(embedRow);

      var coordsLabel = document.createElement("div");
      coordsLabel.className = "share-popover-label";
      coordsLabel.textContent = "Coordinates";
      popover.appendChild(coordsLabel);
      var coordsRow = document.createElement("div");
      coordsRow.className = "share-popover-row";
      var coordsVal = document.createElement("code");
      coordsVal.className = "share-popover-coords";
      coordsVal.textContent = shareCoordsText();
      coordsRow.appendChild(coordsVal);
      coordsRow.appendChild(shareCopyButton("Copy", shareCoordsText, "copy-coordinates"));
      popover.appendChild(coordsRow);

      wrap.appendChild(popover);
      btn.setAttribute("aria-expanded", "true");
      document.addEventListener("keydown", onKeydown, true);
      document.addEventListener("pointerdown", onOutside, true);
      /* Anchored at the top of a scrolling results panel the card can open
         below the fold — on a 390x844 phone it opens 295px tall against about
         255px of room. `nearest` scrolls the least that makes it whole and
         does nothing at all when it already fits, so the desktop case is
         untouched. An engine that predates the options object coerces it to
         truthy and aligns to the top of the scroll box, which is a worse
         choice than `nearest` and still a correct one. */
      try { popover.scrollIntoView({ block: "nearest" }); } catch (e) {}
      linkInput.focus();
      trackEvent("share-open");
    }

    btn.addEventListener("click", function () {
      if (popover) closePopover(true); else openPopover();
    });

    return wrap;
  }
