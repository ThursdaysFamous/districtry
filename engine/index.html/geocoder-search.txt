  // `rows`, when given, can name a class for a row (rows.mark) and take a
  // row's click instead of selecting its point, by returning true
  // (rows.pick): listHeldAnswers uses both
  function renderGeocodeResults(results, fromLink, rows) {
    // the answers to a ZIP code no list holds are each one spot (zipAnswerRows)
    if (!rows && zipAnswers.seq === districtSearchSeq) rows = zipAnswerRows(zipAnswers.zip);
    // the district section (district-result / district-hint rows) is owned by
    // renderDistrictSection and stays on top; only address rows are replaced
    clearResultRows(false);
    if (!results || results.length === 0) {
      if (geoResultsList.querySelector("li.district-result")) { syncResultCount(); return; }
      // "keep typing" is wrong for words the reader did not type: a link that
      // names a place without saying where it is says so instead
      geoStatus.textContent = fromLink ? mapLinkTextMessage(fromLink) :
        "No addresses matched inside " + appPlaceName() + " yet. Keep typing, or check the spelling.";
      geoStatus.className = "geo-status err";
      return;
    }
    results.forEach(function (r) {
      var li = document.createElement("li");
      if (rows && rows.mark) li.className = rows.mark(r);
      var btn = document.createElement("button");
      btn.type = "button";
      btn.textContent = r.display_name; // textContent — never innerHTML — for untrusted geocoder output
      btn.addEventListener("click", function () {
        // cancel any pending debounce fire and in-flight search, so a late
        // response can't re-open the list over the "Selected:" status
        clearTimeout(debounceTimer);
        if (inFlightController) inFlightController.abort();
        if (rows && rows.pick && rows.pick(r)) return;
        trackEvent("address-search");
        setSelectedPoint(parseFloat(r.lat), parseFloat(r.lon));
        geoResultsList.innerHTML = "";
        geoStatus.textContent = "Selected: " + r.display_name;
        // a point is now chosen — collapse the shell back to just the input row
        // so the confirmation + type-ahead controls stop covering the map. The
        // "Selected: …" text stays in the DOM, so it reappears on hover/focus.
        collapseAfterSelect = true;
        refreshSearchShellExpanded();
      });
      li.appendChild(btn);
      geoResultsList.appendChild(li);
    });
    syncResultCount(); // every row shown, the district section's included
  }

  /* ---------- sibling-metro search fallback (easter egg) ----------
   * When a query matches nothing inside the home metro, quietly retry it
   * against each sibling District Explorer's metro bbox (METRO_EXPLORERS —
   * see the ENGINE metro-portal block). A hit renders as a special result
   * row that hands off to that fork with the address pre-selected via its
   * #point= permalink. Only runs after a zero-result home search, so the
   * extra request never fires for normal in-city typing. */
  function siblingRow(metro, label, go) {
    var li = document.createElement("li");
    li.className = "sibling-result";
    var btn = document.createElement("button");
    btn.type = "button";
    var addr = document.createElement("span");
    addr.className = "sibling-result-addr";
    // textContent — never innerHTML — for untrusted geocoder output
    addr.textContent = (metro.emoji ? metro.emoji + " " : "") + label;
    var hint = document.createElement("span");
    hint.className = "sibling-result-hint";
    hint.textContent = "Open in " + siblingDisplayName(metro) + " →";
    btn.appendChild(addr);
    btn.appendChild(hint);
    btn.addEventListener("click", function () {
      clearTimeout(debounceTimer);
      if (inFlightController) inFlightController.abort();
      go();
    });
    li.appendChild(btn);
    return li;
  }
  function buildSiblingResultRow(metro, r) {
    return siblingRow(metro, r.display_name, function () {
      trackEvent("metro-portal-go/" + metro.id, "search");
      window.location.href = metroPortalUrl(metro, { lat: parseFloat(r.lat), lng: parseFloat(r.lon) }, map.getZoom());
    });
  }
  // An answer that is not the house an address names (isHouse): that app is
  // opened with the address, as #q=, and searches it there, where such an
  // answer is never selected (listHeldAnswers); the answer's own point is a
  // spot somewhere along a street, or another street's house, not the
  // address
  function handOnSiblingRow(metro, r, address) {
    var li = siblingRow(metro, r.display_name, function () {
      trackEvent("metro-portal-go/" + metro.id, "search");
      var url = metroPortalUrl(metro);
      window.location.href = url + (url.indexOf("#") < 0 ? "#" : "&") + "q=" + encodeURIComponent(address);

  }

  /* ---------- a link to a place in another state ----------
   * A link whose text names another state in its address (ENGINE map-link
   * addressStateTag) is that state's to search: this app's search is bounded
   * to its own state, so it would find nothing there, or a street of the same
   * name here. It draws one row, which opens that state's app with the link
   * as #q=, where the search reads it as a link and tries the address alone,
   * the street and the name as it would here; #q= follows the row
   * (prefillGeocodeSearch), so a place shared into the installed app opens in
   * its own state whichever state the launcher chose. A state with no app is
   * named, and nothing is searched.
   *
   * ONLY A WHOLE STATE'S APP. San Francisco's app carries California's tag
   * and answers for one city, so an address in California does not say the
   * place is in it: such a link is searched here as before, and the sibling
   * lookup decides by the place's position. It returns false then, and true
   * when it has answered. */
  function metroForTag(tag) {
    for (var i = 0; i < METRO_EXPLORERS.length; i++) {
      var m = METRO_EXPLORERS[i], path = /\/([a-z]{2})\/?$/.exec(m.url || "");
      if (m.id !== THIS_METRO && path && path[1] === tag) return m;
    }
    return null;
  }
  function showLinkElsewhere(link, tag) {
    var metro = metroForTag(tag);
    if (metro && metro.label !== US_STATE_NAMES[tag]) return false;
    var maker = link.source === "apple" ? "Apple Maps" : "Google Maps";
    if (inFlightController) inFlightController.abort();
    clearResultRows(true);
    clearResultRows(false);
    if (!metro) {
      geoStatus.textContent = "That " + maker + " link is to a place in " + stateInSentence(tag) +
        ", which districtry doesn't cover yet.";
      geoStatus.className = "geo-status err";
      countLinkRead("uncovered");
      return true;
    }
    var row = siblingRow(metro, link.text, function () {
      if (metroHandedOff) return;
      metroHandedOff = true;
      geoStatus.textContent = "Opening " + siblingDisplayName(metro) + "\u2026";
      geoStatus.className = "geo-status";
      trackEvent("metro-portal-go/" + metro.id, "map-link");
      var url = metroPortalUrl(metro);
      window.location.href = url + (url.indexOf("#") < 0 ? "#" : "&") + "q=" + encodeURIComponent(mapLinkForText(link));
    });
    row.className += " map-link-away";
    geoResultsList.appendChild(row);
    geoStatus.textContent = "That " + maker + " link is to a place in " + stateInSentence(tag) + ".";
    geoStatus.className = "geo-status";
    countLinkRead("other-state");
    return true;
  }

  var SIBLING_ROWS_PER_METRO = 2; // keeps a many-metro future from flooding the result list
  // `handOn` is an address held to its house number (runAddressSearch): when
  // no answer is the house (isHouse), the answers are listed as rows that
  // open their app with the address (handOnSiblingRow) instead of being
  // dropped
  function maybeRenderSiblingMatches(query, mustCarry, seen, handOn) {
    var anySibling = METRO_EXPLORERS.some(function (m) { return m.id !== THIS_METRO && m.bbox; });
    if (!anySibling) return;
    // share the geocoder's abort slot so a new keystroke/search cancels this
    // lookup exactly like it cancels an in-city one
    inFlightController = new AbortController();
    var signal = inFlightController.signal;
    // one UNBOUNDED whole-coverage query, classified into metros locally
    // (fleetSiblingAt) — exactly one extra request per home-metro miss no
    // matter how many sibling metros exist (plus fleet-outlines.json, once
    // per page load); hits outside every sibling metro
    // are dropped. geocodeUnbounded is the fork's provider (the home search
    // box may be scoped to the city, so this lookup needs its own wider call).
    // Each hit is classified by the same fleet outlines a click is
    // (ENGINE metro-portal), so a searched address and a clicked one go to
    // the same app; only if that file cannot be read do the boxes decide.
    return geocodeUnbounded(query, signal)
      .then(function (results) {
        return loadFleetOutlines().then(function () { return { results: results, outlines: true }; },
          function () { return { results: results, outlines: false }; });
      })
      .then(function (got) {
        if (signal.aborted) return;
        // every answer, before any is held back, for a caller looking for
        // the street a shared address is on (runAddressSearch)
        if (seen) seen(got.results);
        // held to the house when the query is an address without the name
        // it was shared with (runAddressSearch)
        var results = !mustCarry ? got.results : got.results.filter(function (r) {
          return isHouse(r, mustCarry);
        });
        var away = !!handOn && !results.length;
        if (away) results = got.results;
        var perMetro = {}; // id -> rows rendered
        var labels = [];
        results.forEach(function (r) {
          var lat = parseFloat(r.lat), lng = parseFloat(r.lon);
          var metro = got.outlines ? fleetSiblingAt(lat, lng) : siblingMetroAt(lat, lng);
          if (!metro) return; // not inside any sibling metro
          perMetro[metro.id] = (perMetro[metro.id] || 0) + 1;
          if (perMetro[metro.id] > SIBLING_ROWS_PER_METRO) return;
          if (labels.indexOf(metro.label) === -1) labels.push(metro.label);
          geoResultsList.appendChild(away ? handOnSiblingRow(metro, r, handOn) : buildSiblingResultRow(metro, r));
        });
        if (!labels.length) return 0; // leave the standard "no matches" status as-is
        geoStatus.textContent = "Nothing in " + appPlaceName() + " matched — but it looks like you're searching " +
          labels.join(" and ") + ".";
        geoStatus.className = "geo-status";
        return labels.length;
      })
      .catch(function (err) {
        if (err.name === "AbortError") throw err; // the callers' handlers ignore aborts
        // otherwise swallow: the home search itself succeeded, so the honest
        // state is the standard "no matches" message already on screen — a
        // failed easter-egg lookup must never surface as a search error
      });
  }

  /* ---------- unit-fragment retry ----------
   * People paste the address off a letterhead — "233 S Wacker Dr Suite 8400,
   * Chicago" — and the search box handed that to the geocoder verbatim, so a
   * fragment the POI path has stripped since cleanPoiAddress() existed still
   * sank the search box. Measured on 12 paired Chicago queries (same address
   * with and without the unit): 7/12 resolved as typed, 9/12 after cleaning —
   * and 9/12 is also what the hand-stripped address scores, so cleaning closes
   * the entire gap the fragment opens. (The 3 that still miss lack the house
   * number in OSM and fail on the plain address too — not a unit problem.)
   *
   * Retry rather than clean up front: the raw query keeps first crack, so a
   * search that works today cannot regress, and a genuine place name that
   * merely looks like a unit ("The Room 43") is only ever cleaned once the
   * as-typed query has already come back empty. Costs one extra request on a
   * miss, exactly like the sibling-metro fallback it runs ahead of. */
  /* ---------- search by district name: "33rd ward", "State Senate District 7" ----------
   * An instance that declares DISTRICT_SEARCH_URL (the worksheet's
   * district_search key; Illinois today) ships an index of its numbered
   * districts, one point inside each plus its extent, built by
   * scripts/build_district_search.py. The query is matched against that
   * file in the browser, so a district name never goes to the geocoder.
   *
   * THE VARIANTS ARE DERIVED HERE, ONCE. The file stores each layer's
   * official district name with the number taken out ("ward", "state senate
   * district") and its sidebar label; a query matches when, after lower-
   * casing, dropping punctuation, "#", "no." and "number", it is one number
   * (plain or ordinal: 33, 33rd) plus words that are:
   *   - a phrase exactly ("ward 33", "33rd ward", "City Ward #33"), or
   *   - the start of one, for type-ahead ("33rd wa"), or
   *   - its last two or more words ("senate district 7").
   * A bare "district 7" matches nothing: every layer here has a District 7,
   * and picking one would be a guess. The same reason makes a COUNTY-
   * dispatched layer ("where_per_district", the county board) match only
   * when the query names the county: "Lake County Board District 3" or
   * "Lake Board District 3", never "county board district 3", which is
   * every county's. A lettered district ("Clay County Board District C")
   * is read from the one letter that follows "district".
   *
   * A SECOND NAME. A layer whose districts also have a name of the form
   * number-plus-letter carries it on each district as `alt` — the Chicago
   * Board of Education's seats, 1a..10b, the name the Board itself gives
   * each of the 20. "2b" or "2 b" in a query is read as that name, and
   * matches only through the layer's `alt_phrases` ("school board district
   * 2b", "school board 2b").
   * Where the layer marks its own number INTERNAL (`alt_title` in the
   * index), that number is never matched or shown — the Board's seats are
   * rows 1..20 of a shapefile to this index and nothing to a reader — and a
   * bare number after the layer's own phrases names every district whose
   * second name starts with it: "school board district 4" is the Board's
   * District 4, which is 4a and 4b.
   *
   * A complete name (the phrase, or its last two or more words) answers
   * alone and nothing is sent to the geocoder. A type-ahead prefix is shown
   * ABOVE the address results, because "33 wa" is also the start of
   * "33 Wabash Ave" and the reader may mean the street.
   *
   * Selecting a result turns its layer on (a real click, so every handler
   * runs, as the permalink restore does), selects the stored point — the
   * spot inside the district farthest from its edges — and frames the
   * district's extent. Every card then answers for that point exactly as it
   * would for a click there. */
  var districtIndexPromise = null;
  var districtSearchSeq = 0;
  function loadDistrictIndex() {
    if (typeof DISTRICT_SEARCH_URL === "undefined") return Promise.resolve(null);
    if (!districtIndexPromise) {
      districtIndexPromise = fetchJSONWithRetry(DISTRICT_SEARCH_URL, {}, 2)["catch"](function (err) {
        console.warn("district search index unavailable; searching addresses only:", err);
        districtIndexPromise = null; // try again on the next search
        return null;
      });
    }
    return districtIndexPromise;
  }
  function normalizeDistrictText(text) {
    return String(text || "").toLowerCase().replace(/#/g, " ").replace(/[^a-z0-9]+/g, " ").trim();
  }
  function parseDistrictQuery(q) {
    var num = null, words = [];
    var tokens = normalizeDistrictText(q).split(" ");
    for (var i = 0; i < tokens.length; i++) {
      var t = tokens[i];
      if (!t || t === "no" || t === "number") continue;
      var m = /^(\d+)(st|nd|rd|th)?$/.exec(t);
      if (m) {
        if (num !== null) return null; // two numbers: an address, not a district
        num = String(parseInt(m[1], 10));
        // "2 b": a number and the one letter after it are a second name
        if (!m[2] && /^[a-z]$/.test(tokens[i + 1] || "")) {
          num += tokens[i + 1].toUpperCase();
          i++;
        }
        continue;
      }
      // "2b": a second name written as one word
      var alt = /^(\d+)([a-z])$/.exec(t);
      if (alt) {
        if (num !== null) return null;
        num = String(parseInt(alt[1], 10)) + alt[2].toUpperCase();
        continue;
      }
      words.push(t);
    }
    // no number: a lettered district is the single letter after "district"
    if (num === null) {
      var at = words.indexOf("district");
      if (at >= 0 && at + 1 < words.length && /^[a-z]$/.test(words[at + 1])) {
        num = words[at + 1].toUpperCase();
        words.splice(at + 1, 1);
      }
    }
    // no number at all is still a parse: it is how "Lake County Board Dis"
    // reaches the type-a-number hint before the reader has typed one
    if (num === "0" || !words.length) return null;
    return { num: num, words: words.join(" ") };
  }
  // 2 = the phrase or its last 2+ words (a complete name: answers alone),
  // 1 = typing it, from ANY word of the phrase ("33rd wa", "house dis" of
  //     "state house district") — which may yet become a street name,
  // 0 = no match. The one partial that never counts is "district" or the
  // start of it: every phrase contains it, so "district 7" or "dist" would
  // name every layer at once.
  function districtPhraseScore(words, phrase) {
    if (words === phrase) return 2;
    if (words.indexOf(" ") > 0 && phrase.length > words.length &&
        phrase.slice(-(words.length + 1)) === " " + words) return 2;
    if (words.length < 2 || "district".indexOf(words) === 0) return 0;
    if ((" " + phrase).indexOf(" " + words) >= 0) return 1;
    return 0;
  }
  // The ids each layer (and each county of a county-dispatched layer) holds,
  // gathered once per index load for the hint's "type its number (1–19)".
  function districtIdSets(index) {
    if (index.__idSets) return index.__idSets;
    var sets = {};
    index.features.forEach(function (f) {
      var p = f.properties || {};
      var spec = index.layers[p.layer];
      if (!spec) return;
      var key = p.layer + "|" + (spec.where_per_district ? (p.where || "") : "");
      (sets[key] = sets[key] || []).push(String(p.id));
    });
    index.__idSets = sets;
    return sets;
  }
  // "1–50" when the ids run unbroken, "22 districts, 1–25" when they do not
  // (CPD has no 13th District), "A–N" for letters.
  function describeIds(ids) {
    var nums = ids.filter(function (d) { return /^\d+$/.test(d); }).map(Number).sort(function (a, b) { return a - b; });
    if (nums.length === ids.length && nums.length) {
      var lo = nums[0], hi = nums[nums.length - 1];
      return (hi - lo + 1 === nums.length ? "" : nums.length + " districts, ") + lo + "\u2013" + hi;
    }
    var letters = ids.slice().sort();
    return letters[0] + "\u2013" + letters[letters.length - 1];
  }
  function firstId(ids) {
    return ids.slice().sort(function (a, b) {
      var na = /^\d+$/.test(a), nb = /^\d+$/.test(b);
      return na && nb ? Number(a) - Number(b) : (na ? -1 : nb ? 1 : (a < b ? -1 : a > b ? 1 : 0));
    })[0];
  }
  /* ---------- NAMED districts: "Galesburg school district", "Curie HS" ----------
   * A layer marked `match: "name"` (the statewide school districts, CPS's
   * attendance zones) is matched by each district's own name, because its
   * numbers repeat — 36 Illinois unified districts are District 1. The
   * query and the name are read the same way: lower case, punctuation
   * dropped, a number split from a letter stuck to it ("365U", "D203"), and
   * the abbreviations districts use for themselves spelled out (CUSD,
   * CCSD, HSD, SD, HS...), so "Galesburg CUSD 205" and "Galesburg Community
   * Unit School District 205" are one query. Every word typed must be in
   * the name or in the layer's own name ("Galesburg unified"), and the last
   * may be the start of one, for type-ahead. It must also:
   *   - name a place: at least one word of the district's name that is not
   *     a kind of school ("Galesburg", "Curie"). A number alone finds only a
   *     district whose name has no place in it ("District 214"), since the
   *     number is the only name it has;
   *   - say it is a school, with a school word or the number, so "Evanston"
   *     alone stays an address search.
   * A name typed in full answers alone, like a numbered district; one still
   * being typed is shown above the address results. */
  var NAMED_EXPAND = {
    cusd: "community unit school district", ccsd: "community consolidated school district",
    chsd: "community high school district", thsd: "township high school district",
    hsd: "high school district", esd: "elementary school district", cud: "community unit district",
    csd: "consolidated school district",
    sd: "school district", d: "district", dist: "district", hs: "high school", es: "elementary school",
    ms: "middle school", elem: "elementary", twp: "township", comm: "community", cons: "consolidated",
    sch: "school", schools: "school", unified: "unit"
  };
  var NAMED_FILLER = { no: 1, number: 1, the: 1, of: 1, and: 1, public: 1 };
  // words that say what kind of school, never which one
  var NAMED_KIND = { school: 1, district: 1, high: 1, elementary: 1, middle: 1, unit: 1, community: 1,
    consolidated: 1, township: 1, zone: 1, cps: 1, grade: 1, junior: 1, senior: 1, area: 1, county: 1, city: 1 };
  function namedTokens(text) {
    var out = [];
    normalizeDistrictText(text).replace(/(\d)([a-z])/g, "$1 $2").replace(/([a-z])(\d)/g, "$1 $2")
      .split(" ").forEach(function (t) {
        if (!t || NAMED_FILLER[t]) return;
        if (/^\d+$/.test(t)) { out.push(String(parseInt(t, 10))); return; }
        (NAMED_EXPAND[t] || t).split(" ").forEach(function (w) { out.push(w); });
      });
    return out;
  }
  // per feature, once per index load: every word it answers to, and the
  // words of its own name that name a place
  function namedEntries(index) {
    if (index.__named) return index.__named;
    var entries = [];
    index.features.forEach(function (f) {
      var p = f.properties || {}, spec = index.layers[p.layer];
      if (!spec || spec.match !== "name") return;
      var own = namedTokens(p.name), all = {}, place = [];
      own.concat(namedTokens(spec.label), namedTokens(spec.name), namedTokens(spec.where || ""))
        .forEach(function (w) { all[w] = 1; });
      own.forEach(function (w) {
        if (w.length > 1 && !/^\d+$/.test(w) && !NAMED_KIND[w] && place.indexOf(w) < 0) place.push(w);
      });
      entries.push({ f: f, spec: spec, all: all, words: Object.keys(all), place: place });
    });
    index.__named = entries;
    return entries;
  }
  function findNamedDistricts(index, q) {
    var query = namedTokens(q), found = [];
    if (!query.length) return found;
    namedEntries(index).forEach(function (e) {
      var partial = false, place = 0, num = false, kind = false;
      for (var i = 0; i < query.length; i++) {
        var w = query[i], hit = null;
        if (e.all[w]) hit = w;
        else if (i === query.length - 1 && w.length > 1 && !/^\d+$/.test(w)) {
          for (var k = 0; k < e.words.length; k++) {
            if (e.words[k].indexOf(w) === 0 && (!hit || e.place.indexOf(e.words[k]) >= 0)) hit = e.words[k];
          }
          if (hit) partial = true;
        }
        if (!hit) return;
        if (e.place.indexOf(hit) >= 0) place++;
        else if (/^\d+$/.test(hit)) num = true;
        else if (NAMED_KIND[hit]) kind = true;
      }
      if (!kind && !num) return;
      if (!place && !(num && !e.place.length)) return;
      var p = e.f.properties, where = p.where || e.spec.where;
      found.push({ layer: p.layer, label: where ? e.spec.label + " \u00b7 " + where : e.spec.label,
        name: p.name, bbox: p.bbox, coords: e.f.geometry.coordinates, score: partial ? 1 : 2,
        missed: e.place.length - place });
    });
    found.sort(function (a, b) {
      return b.score - a.score || a.missed - b.missed || (a.name < b.name ? -1 : a.name > b.name ? 1 : 0);
    });
    return found.slice(0, 8);
  }
  function findDistricts(q) {
    var parsed = parseDistrictQuery(q);
    var none = { hits: [], exact: false, hints: [] };
    if (!parsed && !namedTokens(q).length) return Promise.resolve(none);
    return loadDistrictIndex().then(function (index) {
      if (!index || !index.layers || !index.features) return none;
      var found = parsed ? findNumberedDistricts(index, parsed) : none;
      var named = findNamedDistricts(index, q);
      if (!named.length) return found;
      return {
        hits: found.hits.concat(named),
        exact: found.exact || named.some(function (h) { return h.score === 2; }),
        // a name is being matched: "Type its name" would only repeat it
        hints: found.hints.filter(function (h) { return !h.named; })
      };
    });
  }
  function findNumberedDistricts(index, parsed) {
    var hits = [], exact = false, hints = [];
    // the query's words with one place name taken out, or null when the
    // place is not in them ("lake" out of "lake county board district")
    // A two-word name is also tried run together, since "DeWitt" is how
    // many readers type De Witt.
    var without = function (words, place) {
      var w = " " + words + " ", name = normalizeDistrictText(place);
      var forms = [name, name.replace(/ /g, "")];
      for (var k = 0; k < forms.length; k++) {
        var p = " " + forms[k] + " ";
        if (forms[k] && w.indexOf(p) >= 0) return w.replace(p, " ").trim();
      }
      return null;
    };
    var bestScore = function (words, phrases) {
      var score = 0;
      if (words) (phrases || []).forEach(function (p) { score = Math.max(score, districtPhraseScore(words, p)); });
      return score;
    };
    if (parsed.num === null) {
      // NO NUMBER YET: the words name a kind of district ("Lake County
      // Board Dis", "ward"), so say what to type next rather than guess
      // which district is meant. A hint is not a result and is not a
      // button: it cannot be picked, and #q= never auto-selects it.
      var sets = districtIdSets(index);
      Object.keys(index.layers).forEach(function (layerId) {
        var spec = index.layers[layerId];
        if (spec.match === "name") {
          if (bestScore(parsed.words, spec.phrases)) {
            hints.push({ name: spec.name, named: true,
              note: "Type its name" + (spec.where ? " \u00b7 " + spec.where : "") });
          }
          return;
        }
        if (spec.where_per_district) {
          var named = false, example = null;
          Object.keys(sets).forEach(function (key) {
            if (key.indexOf(layerId + "|") !== 0) return;
            var place = key.slice(layerId.length + 1);
            if (!example || place < example) example = place;
            var rest = without(parsed.words, place);
            if (rest === null || !bestScore(rest, spec.phrases)) return;
            named = true;
            hints.push({ name: place + " " + spec.name, note: "Type its number: " + describeIds(sets[key]) });
          });
          if (!named && bestScore(parsed.words, spec.phrases) && example) {
            hints.push({ name: spec.name, note: "Start with the county, e.g. " + example + " " + spec.name + " " +
              firstId(sets[layerId + "|" + example]) });
          }
          return;
        }
        var words = spec.where ? (without(parsed.words, spec.where) || parsed.words) : parsed.words;
        if (spec.alt_phrases && (bestScore(words, spec.alt_phrases) || (spec.alt_title && bestScore(words, spec.phrases)))) {
          var alts = index.features.filter(function (f) { return f.properties.layer === layerId && f.properties.alt; })
            .map(function (f) { return f.properties.alt; })
            .sort(function (a, b) { return parseInt(a, 10) - parseInt(b, 10) || (a < b ? -1 : a > b ? 1 : 0); });
          if (alts.length) {
            // alt_title where the number is internal ("School Board District"),
            // else the layer's name with alt_name in place of "District"
            hints.push({ name: spec.alt_title || spec.name.replace(/\s*district$/i, "") + " " + spec.alt_name,
              note: "Type its " + spec.alt_name.toLowerCase() + ": " + alts[0] + "\u2013" + alts[alts.length - 1] +
                (spec.where ? " \u00b7 " + spec.where : "") });
          }
        }
        if (spec.alt_title || !bestScore(words, spec.phrases)) return; // an internal number is never offered
        hints.push({ name: spec.name, note: "Type its number: " + describeIds(sets[layerId + "|"] || []) +
          (spec.where ? " \u00b7 " + spec.where : "") });
      });
      return { hits: hits, exact: exact, hints: hints };
    }
    Object.keys(index.layers).forEach(function (layerId) {
      var spec = index.layers[layerId];
      if (spec.match === "name") return; // matched by name, below
      for (var i = 0; i < index.features.length; i++) {
        var f = index.features[i], props = f.properties || {};
        if (props.layer !== layerId) continue;
        // the district's number, or its second name ("2B") — which matches
        // only through the layer's own words for that name
        var byAlt = !!props.alt && String(props.alt).toUpperCase() === parsed.num;
        // an internal number: "district 4" is every district whose second
        // name is 4 and a letter, and the index's own 4 matches nothing
        var byPair = !!spec.alt_title && !byAlt && !!props.alt &&
          String(props.alt).replace(/[a-z]$/i, "") === parsed.num;
        if (spec.alt_title ? !byAlt && !byPair : String(props.id) !== parsed.num && !byAlt) continue;
        var score;
        if (byAlt) {
          var altWords = spec.where ? (without(parsed.words, spec.where) || parsed.words) : parsed.words;
          score = bestScore(altWords, spec.alt_phrases);
        } else if (spec.where_per_district) {
          // the county is required: every county has this district number
          var rest = without(parsed.words, props.where || "");
          if (rest === null) continue;
          score = bestScore(rest, spec.phrases);
        } else {
          // a layer that belongs to one city ("where": "Chicago") also
          // matches with that city named in the query: "Chicago Ward 33"
          var words = spec.where ? (without(parsed.words, spec.where) || parsed.words) : parsed.words;
          score = bestScore(words, spec.phrases);
        }
        if (!score) continue;
        hits.push({ layer: layerId, label: spec.where ? spec.label + " \u00b7 " + spec.where : spec.label, name: props.name, bbox: props.bbox, coords: f.geometry.coordinates, score: score });
        if (score === 2) exact = true;
        if (!spec.where_per_district && !byPair) break; // one district per number in a single-numbering layer
      }
    });
    hits.sort(function (a, b) { return b.score - a.score; });
    return { hits: hits, exact: exact, hints: hints };
  }

  /* ---------- a ZIP code: all of it, compared ----------
   * A ZIP code typed into the box used to be sent to the geocoder like an
   * address, and its answer selected one spot, so every card named that
   * spot's districts. A ZIP code is not a spot: Chicago's 60625 crosses four
   * wards (measured 2026-10-21 against Census 2020 blocks: Ward 33 41%, Ward
   * 40 30%, Ward 39 16%, Ward 47 13%), and a reader who searched it reported
   * that the map named one. On Adam's choice of that day the search now
   * selects a spot inside the ZIP code, turns its ZIP layer on and PINS it
   * for comparison (ENGINE relationship-pinning), so the map outlines every
   * district of the other layers on that lies inside the ZIP code or crosses
   * its edge, and the reader sees the comparison without having to find it.
   * When no other layer is on, the one turned on to compare with is chosen
   * at the spot, in the order Adam gave: a city's council districts (the
   * concept city-council: Chicago's wards) where the spot has one, the
   * county board's districts where it does not ("Ward then county district
   * rep if ward isn't present"), and the U.S. House's where neither answers,
   * which is North Carolina's case outside its cities. A layer already on,
   * such as one a question page turned on, is compared as it is.
   *
   * THE SPOT IS NOT AN ADDRESS, and the message says so: the cards answer
   * for the selected point, as they always do, and the point is the place in
   * the ZIP code farthest from its edges (scripts/build_zip_index.mjs, which
   * also keeps it on the ZIP layer's own ground), nobody's house.
   *
   * NOTHING GOES TO A GEOCODER for a ZIP code a list holds: this app's own
   * (ZIP_SEARCH_URL, the worksheet's zip_search key) or, for a ZIP code
   * another app holds, the fleet's (../fleet-zips.json), whose row opens
   * that app with the ZIP code as #q=. A ZIP code no list holds (a post
   * office box's, one too narrow for a spot clear of its edges, or any in an
   * app with no ZIP layer, Kentucky's) is searched as before, and its
   * answers are marked so that #q= never selects one and a reader who picks
   * one is told the cards are for that spot alone.
   *
   * THE COST, stated to Adam before he chose it: a comparison reads the
   * whole of each layer it compares, so in Illinois a ZIP code search
   * downloads the ZIP layer's whole file from the Census Bureau (about 4 MB)
   * and, in Chicago, every county's wards (about 12 MB from about 20
   * servers). Counted as zip-search/<the ZIP layer>, never by ZIP code,
   * which says where a reader lives more precisely than the two-decimal
   * rounding every point event is held to; the pin it makes is not counted
   * as compare/<layer>, which stays the count of readers who pinned by hand. */
  var ZIP_COMPARE_CONCEPTS = ["city-council", "county-board", "us-house"];
  var zipIndexPromise = null;
  function loadZipIndex() {
    if (typeof ZIP_SEARCH_URL === "undefined") return Promise.resolve(null);
    if (!zipIndexPromise) {
      zipIndexPromise = fetchJSONWithRetry(ZIP_SEARCH_URL, {}, 2).then(function (data) {
        var byZip = {};
        ((data && data.features) || []).forEach(function (f) {
          var p = f && f.properties;
          if (p && p.zip && getLayerById(p.layer) && f.geometry && f.geometry.coordinates) byZip[p.zip] = f;
        });
        return byZip;
      })["catch"](function (err) {
        console.warn("ZIP code index unavailable; ZIP codes are searched as addresses:", err);
        zipIndexPromise = null; // try again on the next search
        return null;
      });
    }
    return zipIndexPromise;
  }

      });
    });
  }

  // A message about the ZIP code, kept on screen until another point is
  // selected or another search begins; returns a function that rewrites it
  // while it is still the one on screen
  function zipMessage(text, extra) {
    var seq = districtSearchSeq, picked = state.sequence;
    function current() { return seq === districtSearchSeq && picked === state.sequence; }
    // The message stays open while the reader is at the search bar and folds
    // away once they move off it (the pointer leaves it, a click or a tap
    // lands elsewhere, or focus moves out), as a picked address's line does
    // at once; hovering or focusing the bar shows it again. Only a mouse
    // leaving counts, because a finger "leaves" every time it lifts.
    var left = false;
    function write(t, more) {
      if (!current()) return;
      geoStatus.textContent = t;
      geoStatus.className = "geo-status";
      if (more) geoStatus.appendChild(more);
      collapseAfterSelect = left;
      refreshSearchShellExpanded();
    }
    function stopWatchingAway() {
      searchShellEl.removeEventListener("pointerleave", away);
      searchShellEl.removeEventListener("focusout", away);
      document.removeEventListener("pointerdown", away, true);
    }
    function away(e) {
      if (!current()) { stopWatchingAway(); return; }
      if (e.type === "pointerleave" && e.pointerType !== "mouse") return;
      if (e.type === "focusout" && searchShellEl.contains(e.relatedTarget)) return;
      if (e.type === "pointerdown" && searchShellEl.contains(e.target)) return;
      left = true;
      stopWatchingAway();
      collapseAfterSelect = true;
      refreshSearchShellExpanded();
    }
    write(text, extra);
    searchShellEl.addEventListener("pointerleave", away);
    searchShellEl.addEventListener("focusout", away);
    document.addEventListener("pointerdown", away, true);
    // every selection adds the selected point's marker (setSelectedPoint)
    var watch = function () {
      if (state.sequence === picked) return;
      map.off("layeradd", watch);
      stopWatchingAway();
      if (seq !== districtSearchSeq) return; // a newer search wrote its own words
      geoStatus.textContent = "";
      collapseAfterSelect = true;
      refreshSearchShellExpanded();
    };
    map.on("layeradd", watch);
    return write;
  }

  function showZip(zip, f) {

    geoResultsList.innerHTML = "";

    var picked = state.sequence, point = state.selectedPoint;
    var box = document.getElementById("toggle-" + layer);
    if (box && !box.checked) box.click(); // a real click, so the layer's handlers run
    // pinned without being counted as a reader's pin
    if (pinnedParentId !== layer) setPinnedParent(layer, { quiet: true });
    var lead = "The map has selected a spot inside ZIP code " + zip + ", not at any address, so the cards below " +
      "are for that spot. The ZIP code is outlined in bold on the map";
    var write = zipMessage(lead + ".");
    var b = f.bbox;
    // framed below the search panel, with room for the line the comparison
    // adds to the message (clearOfSearchPanel)
    if (b && b.length === 4) {
      map.fitBounds([[b[1], b[0]], [b[3], b[2]]], { paddingTopLeft: [24, clearOfSearchPanel() + 36], paddingBottomRight: [24, 24] });
    } else {
      map.setView([lat, lng], Math.max(map.getZoom(), 13));
    }
    return zipCompareWith(mod, point, picked).then(function (compared) {
      if (picked !== state.sequence || !compared) return;
      var with1 = compared.turnOn || (compared.on.length === 1 ? compared.on[0] : null);
      if (compared.turnOn) {
        var other = document.getElementById("toggle-" + compared.turnOn.id);
        if (other && !other.checked) other.click(); // a real click, as for the ZIP layer
      }
      write(lead + ", and so is every " + (with1 ? zipDistrictWord(with1) + " it crosses" :
        "district it crosses in the layers you have on") + ". ", zipStatsButton(picked));
    });
  }
  // What the ZIP code is compared with: { on: [the polygon layers already
  // on that answer here] }, or { turnOn: the layer to turn on }, or null
  function zipCompareWith(zipMod, point, picked) {
    var zipIds = layerIdsOfConcept["zip-code"] || [];
    var on = layers.filter(function (mod) {
      return state.layersOn[mod.id] && mod.id !== zipMod.id && zipIds.indexOf(mod.id) < 0 && mod.overlay && mod.overlay.style;
    });
    return Promise.all(on.map(function (mod) { return layerRelevance(mod, picked); })).then(function (relevant) {
      on = on.filter(function (mod, i) { return relevant[i]; });
      if (on.length) return { on: on };
      var candidates = [];
      ZIP_COMPARE_CONCEPTS.forEach(function (concept) {
        (layerIdsOfConcept[concept] || []).forEach(function (id) {
          var mod = getLayerById(id);
          if (mod && mod.overlay && mod.overlay.style && candidates.indexOf(mod) < 0) candidates.push(mod);
        });
      });
      // the first that has a district at the spot, asked one at a time, so a
      // layer later in the order downloads nothing when an earlier one answers
      return candidates.reduce(function (chain, mod) {
        return chain.then(function (found) {
          if (found || picked !== state.sequence) return found;
          return Promise.resolve()
            .then(function () { return mod.coverage ? mod.coverage(point) : true; })
            ["catch"](function () { return true; }) // a coverage test that throws fails open, as the cards' does
            .then(function (inside) { return inside ? mod.query(point, picked) : null; })
            .then(function (result) { return result ? { turnOn: mod } : null; }, function (err) {
              console.warn("[" + mod.id + "] no answer at the ZIP code's spot:", err);
              return null;
            });
        });
      }, Promise.resolve(null));
    });
  }

  /* ---------- counting the map links the search box reads ----------
   * One event for each Google Maps or Apple Maps link the search box reads,
   * sent once its result is known, so the report can say how often places
   * are shared or pasted here and how often the search found them:
   *
   *   map-link-read/<tag>/<source>/<form>/<via>/<outcome>
   *
   * <tag> is this app's; <source> google or apple; <form> short (a short
   * link the lookup had to read), short-retried (read only after a retry)
   * or full; <via> opened (the link came in the page's address, #q=, which
   * is how a place shared into the installed app arrives) or pasted (put
   * in the box); and <outcome> what the reader was shown:
   *   point        a position the link carried, offered as one row
   *   found        address rows for the place
   *   pin          the Census Bureau's pin for the address, with the
   *                street it is on when the search knows that street
   *                (showStreetOnly)
   *   street       the street the address is on, framed for a tap, or
   *                listed to pick (listHeldAnswers)
   *   name         places with the link's name, listed for the reader
   *   other-state  the place is another app's, offered there
   *   uncovered    the place is in a state no app covers
   *   district     a district name (DISTRICT_SEARCH_URL) answered alone
   *   none         nothing matched, or only answers on other streets
   *                without the house number (listHeldAnswers)
   *   unreadable   the link names no place (a directions link, for one)
   *   failed-<why> the lookup could not read a short link, and why
   *                (resolveShortMapLink's reasons, or "other")
   *
   * NOTHING ABOUT THE PLACE IS IN IT: no address, name, link or position,
   * only which kind of link it was and what became of it, so the count says
   * as much about a reader as the `share-in/` count already does. A typed
   * coordinate pair is not a map link and is not counted here. A link read
   * twice with the same result (pasted, then searched again with Enter) is
   * counted once. */
  var linkRead = null, linkReadSent = {}, linkVia = "pasted";
  function beginLinkRead(link) {
    linkRead = null;
    if (!link || (link.source !== "google" && link.source !== "apple")) return;
    linkRead = { key: link.url || link.text || "", source: link.source,
                 short: link.kind === "short", retried: false, via: linkVia };
  }
  function countLinkRead(outcome) {
    var r = linkRead;
    if (!r) return;
    linkRead = null;
    if (linkReadSent[r.key + "\n" + outcome]) return;
    linkReadSent[r.key + "\n" + outcome] = true;
    var form = r.short ? (r.retried ? "short-retried" : "short") : "full";
    // the counter loads after this script, and a link handed over in the
    // address can be answered before it has
    dstCountWhenReady("map-link-read/" + METRO_BRAND.instanceTag + "/" + r.source + "/" + form +
                      "/" + r.via + "/" + outcome);
  }

  /* ---------- a pasted Google Maps or Apple Maps link ----------
   * parseMapLink (ENGINE map-link) reads it in the browser. A position
   * becomes ONE row, so the reader still confirms it and #q= selects it as
   * it would a single address match; a place name the link carries is
   * searched as if it had been typed; a short link is followed by the lookup
   * when there is one; and nothing a link carries goes to the geocoder as
   * address text. */
  var mapLinkController = null;
  function showMapLink(link) {
    var seq = ++districtSearchSeq;  // a newer search owns the list now
    clearRangePin();
    if (inFlightController) inFlightController.abort();
    if (mapLinkController) mapLinkController.abort();
    clearResultRows(true);
    clearResultRows(false);
    if (link.kind === "point") {
      geoResultsList.appendChild(mapLinkRow(link));
      syncResultCount();
      countLinkRead("point");
      return Promise.resolve();
    }
    if (link.kind !== "short") {
      geoStatus.textContent = mapLinkMessage(link);
      geoStatus.className = "geo-status err";
      countLinkRead("unreadable");
      return Promise.resolve();
    }
    geoStatus.textContent = "Reading the link\u2026";
    geoStatus.className = "geo-status";
    mapLinkController = new AbortController();
    // a failed lookup is tried again (ENGINE map-link resolveShortMapLink),
    // and the reader is told so rather than left watching the same words
    return resolveShortMapLink(link, mapLinkController.signal, function () {
      if (linkRead) linkRead.retried = true;
      if (seq === districtSearchSeq) geoStatus.textContent = "Reading the link\u2026 trying again";
    }).then(function (full) {
      if (seq !== districtSearchSeq) return;
      if (full.kind === "text") {
        geoStatus.textContent = "Searching\u2026";
        return runGeocodeSearch(full.text, full);
      }
      return showMapLink(full);
    }, function (err) {
      if (seq !== districtSearchSeq || (err && err.name === "AbortError")) return;
      geoStatus.textContent = mapLinkMessage(link, err);
      geoStatus.className = "geo-status err";
      countLinkRead("failed-" + ((err && err.reason) || "other"));
    });
  }
  function mapLinkRow(link) {
    var li = document.createElement("li");
    var btn = document.createElement("button");
    btn.type = "button";
    var where = link.lat.toFixed(5) + ", " + link.lng.toFixed(5);
    btn.textContent = link.source === "coordinates" ? "The point " + where :
      "The place in the " + (link.source === "apple" ? "Apple" : "Google") + " Maps link (" + where + ")";
    btn.addEventListener("click", function () {
      clearTimeout(debounceTimer);
      if (inFlightController) inFlightController.abort();
      trackEvent("map-link/" + link.source);
      geoResultsList.innerHTML = "";
      geoStatus.textContent = "Selected: " + where;
      geoStatus.className = "geo-status";
      collapseAfterSelect = true;
      refreshSearchShellExpanded();
      // A link can name a place anywhere, so it is routed like a click: a
      // point another app answers for opens that app with the point selected.
      if (!handoffSelection(link.lat, link.lng, "map-link", function () { setSelectedPoint(link.lat, link.lng); })) {
        setSelectedPoint(link.lat, link.lng);
      }
    });
    li.appendChild(btn);
    return li;
  }

  // `submitted` is a search the reader sent (Search or Enter, or #q=), as
  // against one the box runs while they type (runAddressSearch)
  function runGeocodeSearch(q, fromLink, submitted) {
    var link = parseMapLink(q);
    // a search of the text a short link led to is still that link's read
    if (!fromLink) beginLinkRead(link);
    if (link && link.kind !== "text") return showMapLink(link);
    if (link) { q = link.text; fromLink = link; }
    var seq = ++districtSearchSeq;
    clearRangePin(); // a pin answers the search before this one
    var tag = fromLink ? addressStateTag(q) : "";
    if (tag && tag !== METRO_BRAND.instanceTag && showLinkElsewhere(fromLink, tag)) return Promise.resolve();
    // a ZIP code typed, or handed over as #q= (runZipSearch)
    var zip = fromLink ? "" : zipQuery(q);
    if (zip) return runZipSearch(zip, submitted);
    return findDistricts(q).then(function (found) {
      if (seq !== districtSearchSeq) return; // a newer search owns the list now
      renderDistrictSection(found);
      if (found.exact) {
        // a district name answers alone: nothing goes to the geocoder, the
        // stale address rows go, and an address lookup still in flight from
        // an earlier keystroke is dropped
        if (inFlightController) inFlightController.abort();
        clearResultRows(false);
        syncResultCount();
        countLinkRead("district");
        return;
      }
      return runAddressSearch(q, fromLink, submitted);
    });
  }

  /* What a reader is told when a link names a place without saying where it
   * is and nothing here matched the name it gave. */
  function mapLinkTextMessage(link) {
    var name = link.source === "apple" ? "Apple Maps" : "Google Maps";
    return "That " + name + " link names \u201c" + link.text + "\u201d but doesn't say where it is, " +
      "and no address in " + appPlaceName() + " matched it. Find the place on the map and tap it.";
  }

  /* The text as it came first, then each retry only after a miss: without
   * a unit fragment (cleanPoiAddress), then the sibling apps.
   *
   * TEXT A MAP LINK CARRIED gets two more tries, because a place shared from
   * the Google Maps app is its name and its address in one string, which a
   * geocoder often cannot match whole (ENGINE map-link addressAfterName).
   * The address alone is tried here and then in the sibling apps, and an
   * answer to it counts only if it is the house (isHouse): without the
   * house number the geocoder answers with streets that partly match, and a
   * single such answer would be selected as the place.
   * Last, the place's name alone, whose answers are listed and never
   * selected, since a name can belong to another place; the reader can tell
   * from the address on the row. Typed text gets none of these, so typing
   * costs the geocoder no more requests than it did.
   *
   * THE STREET WITHOUT THE NUMBER. An answer to the address alone that is on
   * the address's street in its town (labelOnStreet) is kept aside, and when
   * no answer carries the number the map shows that street and asks the
   * reader to tap the place (showStreetOnly), here when this app's own
   * search found it and after the sibling apps' lookup otherwise.
   *
   * AN ADDRESS WITH NO NAME IN FRONT, typed or carried by a link, is held
   * to its house number as well, from 2026-10-17, because an answer without
   * the number is a street and a street's point is a spot somewhere along
   * it. Typed into the front door, such an answer opened its app with that
   * spot selected, and a question page's address box selected it whenever
   * it was the only answer, so a reader asking about one address was shown
   * the cards for another, a ward or more away, with nothing to say so:
   * Photon holds a Chicago storefront's street and not its number. Every
   * answer is still listed, and one that is not the house is marked
   * (listHeldAnswers) so that nothing selects it for the reader. When no
   * answer is the house but one is on the address's street
   * (addressInTownOf, which also reads a town typed with no comma, or not
   * typed at all, from the answer's own label), a search the reader sent
   * shows that street and the Census Bureau's pin as for a shared link, if
   * the answers put the street in one town; while the reader is typing, the
   * street's row does that when it is picked. The geocoder is asked as often
   * as before, bar one retry without a unit ("Apt 2") when the first answers
   * are not the house.
   *
   * THE HOUSE, NOT ONLY ITS NUMBER, from 2026-10-19: until then an answer
   * carrying the house number anywhere in its label counted, and of 49
   * Cook County addresses typed into the Illinois map's search box that day
   * the one answer for one of them was that number on a different street,
   * which a question page's address box selects when it is the only answer.
   *
   * WHEN THE SEARCH KNOWS NEITHER THE HOUSE NOR ITS STREET THERE, from the
   * same day, a search the reader sent with the address's town asks the
   * Census Bureau about the address (showStreetOnly with no street), written
   * out with this app's state when it names none (addressForLookup, ENGINE
   * map-link), before the answers are listed and, when there are none,
   * after the sibling apps' lookup; its pin answers if it has the house,
   * and otherwise the search ends as it did before. A link's address is
   * asked about the same way before the place's name is searched. Of the 17 of
   * those 49 addresses the map's search could not place, 12 had no answer
   * on the street in its town, or only a stretch more than half a mile
   * from the house (STREET_NEAR_MILES, no longer applied to the Census
   * Bureau's answer), and its records put all 12 in the right district;
   * Adam chose its pin for them ("Use Census"). */
  function runAddressSearch(q, fromLink, submitted) {
    var seq = districtSearchSeq;
    var bare = fromLink ? addressAfterName(q) : null;
    var name = bare ? q.split(",")[0].trim() : "";
    var numbered = !bare && OPENS_WITH_ADDRESS.test(q) ? q : null;
    var sent = submitted || !!fromLink; // a link is sent when it is pasted or shared
    var tries = [], held = [], street = [], loose = null, looseOnStreet = false;
    function add(t, mustCarry) {
      if (t && tries.indexOf(t) < 0) { tries.push(t); held.push(mustCarry || null); }
    }
    // each answer on the address's street, with the address it is in there
    function keepStreet(results) {
      (results || []).forEach(function (r) {
        var at = numbered ? addressInTownOf(numbered, r.display_name) : labelOnStreet(r.display_name, bare) ? bare : null;
        if (at) street.push({ answer: r, at: at });
      });
    }
    add(q, numbered); add(cleanPoiAddress(q), numbered);
    if (bare) { add(bare, bare); add(cleanPoiAddress(bare), bare); }
    var i = 0;
    function next(results) {
      if (seq !== districtSearchSeq) return; // a newer search owns the list now
      var all = results, before = street.length;
      if (results && held[i]) {
        keepStreet(results);
        results = results.filter(function (r) { return isHouse(r, held[i]); });
      }
      if (results && results.length) {
        if (numbered) return listHeldAnswers(all, numbered, street, sent);
        renderGeocodeResults(results);
        if (bare) markLinkPlace(name, bare);
        countLinkRead("found");
        return;
      }
      // the answers listed if none is the house: the first try's, or a
      // later try's that found the street when the first did not
      if (numbered && all && all.length && (!loose || (!looseOnStreet && street.length > before))) {
        loose = all;
        looseOnStreet = street.length > before;
      }
      if (++i < tries.length) return geocodeAddress(tries[i]).then(next);
      var towns = streetTowns(street);
      if (towns.length === 1 && (bare || sent)) {
        return showStreetOnly(streetIn(street, towns[0]), towns[0], name, geocodeAddress);
      }
      // a sent address with its town, whose street the search does not know
      // there either: the Census Bureau is asked about it, in this app's
      // state when it names none (addressForLookup)
      var unknown = sent && numbered && !towns.length ? addressForLookup(numbered, METRO_BRAND.instanceTag) : null;
      if (loose && unknown) {
        return showStreetOnly([], unknown, "", geocodeAddress, loose).then(function (shown) {
          if (!shown && seq === districtSearchSeq) listHeldAnswers(loose, numbered, street, sent);
        });
      }
      if (loose) return listHeldAnswers(loose, numbered, street, sent);
      renderGeocodeResults(results, fromLink);
      return Promise.resolve(maybeRenderSiblingMatches(q, numbered, null, numbered)).then(function (found) {
        if (seq !== districtSearchSeq) return;
        if (found) return countLinkRead("other-state");
        if (found === 0 && unknown) {
          return showStreetOnly([], unknown, "", geocodeAddress, []).then(function (shown) {
            if (!shown && seq === districtSearchSeq) countLinkRead("none");
          });
        }
        // 0, never undefined: an aborted or sibling-less lookup tries no more
        if (found !== 0 || !bare) return countLinkRead("none");
        return Promise.resolve(maybeRenderSiblingMatches(bare, bare, keepStreet)).then(function (again) {
          if (seq !== districtSearchSeq) return;
          if (again) return countLinkRead("other-state");
          if (again !== 0) return countLinkRead("none");
          if (street.length) return showStreetOnly(streetIn(street, bare), bare, name, searchEverywhere);
          var asked = addressForLookup(bare, METRO_BRAND.instanceTag);
          if (!asked) return searchName();
          return showStreetOnly([], asked, name, searchEverywhere, []).then(function (shown) {
            if (!shown && seq === districtSearchSeq) return searchName();
          });
        });
      });
    }
    // the place's name alone, last of all for a link: its answers are listed
    // and never selected
    function searchName() {
      if (name.length < 3) return countLinkRead("none");
      return geocodeAddress(name).then(function (named) {
        if (seq !== districtSearchSeq) return;
        if (!named || !named.length) return countLinkRead("none");
        countLinkRead("name");
        listByName(named);
        geoStatus.textContent = "No address in " + appPlaceName() + " matched \u201c" + bare + "\u201d. " +
          "A search for its name, \u201c" + name + "\u201d, found " +
          (named.length === 1 ? "the place below. Pick it if it's the one you meant."
                              : "the places below. Pick the one you meant, if it's there.");
        geoStatus.className = "geo-status";
      });
    }
    return geocodeAddress(tries[0]).then(next);
  }
  // the sibling apps' lookup, with the abort slot every search shares
  function searchEverywhere(text) {
    inFlightController = new AbortController();
    return geocodeUnbounded(text, inFlightController.signal);
  }

  // The addresses the answers on a street are in (keepStreet), one per town,
  // and the answers in one of them
  function streetTowns(street) {
    var towns = [];
    street.forEach(function (s) { if (towns.indexOf(s.at) < 0) towns.push(s.at); });
    return towns;
  }
  function streetIn(street, at) {
    return street.filter(function (s) { return s.at === at; }).map(function (s) { return s.answer; });
  }

  /* Whether an answer is the house an address names. Photon's answers are
   * OpenStreetMap's data, labelled by this app (normalizePhoton), and one is
   * the house only if a part of its label is the number on the address's
   * street (labelAtHouse, ENGINE map-link). An answer from another search,
   * New York City's own address file, is held to the number alone
   * (labelCarriesNumber), as every answer was until 2026-10-19: the street
   * test was written for and tried on OpenStreetMap's labels, not that
   * file's. houseMark is the class a row that is not the house carries:
   * other-street when it has the number on another street, else no-number. */
  function isHouse(r, address) {
    return r.source === "openstreetmap" ? labelAtHouse(r.display_name, address)
                                        : labelCarriesNumber(r.display_name, address);
  }
  function houseMark(r, address) {
    return isHouse(r, address) ? "" : labelCarriesNumber(r.display_name, address) ? "other-street" : "no-number";
  }

  /* Answers to an address held to its house number (runAddressSearch).
   * Every answer is listed, as before, and one that is not the house is
   * marked (houseMark), which #q= never selects (prefillGeocodeSearch).
   * When none is the house, a row on the address's street shows that
   * street when it is picked (showStreetOnly), and the point the Census
   * Bureau gives the number, instead of selecting a spot somewhere along
   * the street, and the status says what the map's search knows. */
  function listHeldAnswers(answers, address, street, sent) {
    var none = !answers.some(function (r) { return isHouse(r, address); });
    renderGeocodeResults(answers, null, {
      mark: function (r) { return houseMark(r, address); },
      pick: function (r) {
        var at = none ? addressInTownOf(address, r.display_name) : null;
        if (!at) return false;
        showStreetOnly(streetIn(street, at), at, "", geocodeAddress);
        return true;
      }
    });
    if (!none) return countLinkRead("found");
    var towns = streetTowns(street.filter(function (s) { return answers.indexOf(s.answer) >= 0; }));
    if (towns.length) {
      var names = towns.map(function (at) { return addressParts(at).town; });
      var a = addressParts(towns[0]);
      geoStatus.textContent = "The map's search knows " + a.street + " in " +
        (names.length > 1 ? names.slice(0, -1).join(", ") + " and " + names[names.length - 1] : names[0]) +
        " but not number " + a.number + " on it. Pick " +
        (names.length > 1 ? "the street in the town you meant" : "the street") + " to see it on the map.";
    } else if (sent) {
      var typed = addressStreet(address);
      var withNumber = answers.some(function (r) { return labelCarriesNumber(r.display_name, address); });
      geoStatus.textContent = (withNumber ? "None of these is number " + typed.number + " on the street you typed"
                                          : "None of these has the house number " + typed.number) +
        ", so pick one only if it's the place you meant, or tap the place on the map.";
    }
    geoStatus.className = "geo-status";
    countLinkRead(towns.length ? "street" : "none");
  }

  /* THE PLACE A LINK NAMES, AMONG SEVERAL ANSWERS. A search for a place's
   * name and address can answer the place and others beside it: on
   * 2026-10-16 Photon, asked for a Wisconsin state park's name and address,
   * answered three places, the park first. An answer whose label opens with
   * the link's name, carries its house number and is on its street in its
   * town (labelCarriesNumber, labelOnStreet, ENGINE map-link) is the place
   * itself, so its row is marked, and #q= selects it when it is the only one
   * (prefillGeocodeSearch). A name alone is not enough: a chain's stores
   * share one. */
  function markLinkPlace(name, bare) {
    var rows = geoResultsList.querySelectorAll("li");
    for (var k = 0; k < rows.length; k++) {
      if (rows[k].className) continue; // the address rows just drawn carry no class
      var label = rows[k].textContent;
      if (samePlaceName(label.split(",")[0], name) && labelCarriesNumber(label, bare) &&
          labelOnStreet(label, bare)) rows[k].className = "link-place";
    }
  }
  // "Devil’s Lake" and "Devil's Lake", "A & B" and "A and B", "Café" and
  // "Cafe": one name
  function samePlaceName(a, b) {
    function key(t) {
      t = String(t || "");
      if (t.normalize) t = t.normalize("NFD").replace(/[\u0300-\u036f]/g, "");
      return t.toLowerCase().replace(/&/g, " and ").replace(/[^a-z0-9]+/g, " ").trim();
    }
    return !!key(a) && key(a) === key(b);
  }

  /* Places a link's name found. Marked so that #q= never selects one for the
   * reader (prefillGeocodeSearch): a name can belong to another place. */
  function listByName(results) {
    renderGeocodeResults(results);
    var rows = geoResultsList.querySelectorAll("li");
    for (var k = 0; k < rows.length; k++) {
      if (!rows[k].className) rows[k].className = "name-result"; // the address rows just drawn
    }
  }

  /* The street a shared or typed address is on, when no answer is its house
   * (isHouse, labelOnStreet, addressInTownOf, ENGINE map-link); `bare` is
   * the address in the street's town. The map frames the street, from the
   * extent the geocoder gives it, and the reader taps the place: nothing
   * is selected, because a point on a street is not the place and a
   * district's edge can run down the middle of the street. The place's name
   * is searched next, and an answer is listed (never selected) only if it is
   * within STREET_NEAR_MILES of the street, since that is where the place
   * is: on 2026-10-14 a search for the name of a small business shared
   * from the Google Maps app answered three other places with similar
   * names, 20 to 50 km from its street. At the same time the Census
   * Bureau is asked where the house is (lookupAddressRanges), and its
   * answer is a pin whose point is selected, with the message saying it
   * can be a block or more from the house, unless another state's app
   * answers there (placeRangePin).
   *
   * WITH NO STREET (`street` empty, from 2026-10-19) the search knows
   * neither the house nor its street there (runAddressSearch), and the
   * Census Bureau is asked about the address the reader sent, with its town
   * (addressForLookup).
   * Nothing is shown until it answers: its pin and the message when it has
   * the house, with the name's answers near the pin, and nothing at all
   * when it has not, so that the caller shows what the search found. The
   * promise resolves true when the pin is shown. `answers` are what the
   * search found, which decide whether the message offers OpenStreetMap's
   * note form. */
  var STREET_NEAR_MILES = 0.5;
  function showStreetOnly(street, bare, name, search, answers) {
    var seq = districtSearchSeq;
    var picked = state.sequence; // a point selected after this answers the message
    var answered = false;
    var a = addressParts(bare);
    var known = street.length > 0; // the search knows the street there
    var bounds = null, zips = [];
    street.forEach(function (r) {
      var e = r.extent; // [west, north, east, south]
      var lat = parseFloat(r.lat), lng = parseFloat(r.lon);
      var b = e && e.length === 4 ? L.latLngBounds([e[3], e[0]], [e[1], e[2]])
                                  : L.latLngBounds([lat, lng], [lat, lng]);
      bounds = bounds ? bounds.extend(b) : b;
      var zip = labelZip(r.display_name);
      if (zip && zips.indexOf(zip) < 0) zips.push(zip);
    });
    var pin = null, pinPicked = false, near = [];
    // every answer came from OpenStreetMap's data (normalizePhoton), so the
    // message can offer its note form; New York's city search reads the
    // city's own address file, which OpenStreetMap's volunteers do not keep
    var from = known ? street : answers || [];
    var osm = from.length > 0 && from.every(function (r) { return r.source === "openstreetmap"; });
    function noteAt() {
      if (pin) return { lat: pin.lat, lng: pin.lng, zoom: 19 };
      // the middle of the part of the map below the search panel, which is
      // where the street was framed
      var size = map.getSize();
      var c = map.containerPointToLatLng([size.x / 2, (clearOfSearchPanel() + size.y) / 2]);
      return { lat: c.lat, lng: c.lng, zoom: Math.min(19, Math.max(17, Math.round(map.getZoom()))) };
    }
    function current() { return !answered && seq === districtSearchSeq && picked === state.sequence; }
    function compose() {
      var text = known ? "The map's search knows " + a.street + " in " + a.town + " but not number " + a.number + " on it"
                       : "The map's search didn't find that address";
      var put = known ? "that number" : "it";
      // the pin's point is selected (pinPicked) unless another state's app
      // answers there, and then the pin waits for a tap (placeRangePin)
      var tapPin = !!pin && !pinPicked;
      text += !pin ? ", so the map is showing that street." : pinPicked ?
        ". The map has selected the pin, where the Census Bureau's street records put " + put + ". " +
        "It can be a block or more from the house, and a district line can run between the two." :
        ". The pin is where the Census Bureau's street records put " + put + ", which can be a block or more from the house.";
      if (near.length) {
        text += " A search for its name, \u201c" + name + "\u201d, " + (pin ? "also " : "") + "found " +
          (near.length === 1 ? "the place below. Pick it if it's the one you meant" : "the places below. Pick the one you meant") +
          (tapPin ? ", tap the pin, or tap the map." : ", or tap the map.");
      } else {
        text += tapPin ? " Tap the pin to use it, or tap the place you meant." :
          pinPicked ? " If the place is elsewhere, tap it on the map." : " Tap the place you meant.";
      }
      geoStatus.textContent = text;
      geoStatus.className = "geo-status";
      if (!osm) return;
      geoStatus.appendChild(document.createTextNode(" The search reads OpenStreetMap's data. If you know where number " +
        a.number + " is yourself, not from another map, you can "));
      geoStatus.appendChild(osmNoteLink(noteAt));
      geoStatus.appendChild(document.createTextNode(" so its volunteers can add it."));
    }
    // the reader's tap answers the message, which then goes
    function done() {
      if (answered) return;
      answered = true;
      if (seq !== districtSearchSeq) return;
      clearRangePin();
      clearResultRows(false);
      geoStatus.textContent = "";
      collapseAfterSelect = true;
      refreshSearchShellExpanded();
    }
    // the street, and the message, as soon as the search knows it
    function show() {
      clearResultRows(false);
      compose();
      refreshSearchShellExpanded();
      map.once("click", done);
    }
    if (known) {
      show();
      map.fitBounds(bounds, { paddingTopLeft: [24, clearOfSearchPanel()], paddingBottomRight: [24, 24], maxZoom: 17 });
    }
    // a pin needs far less room below the search panel than a street does
    function framePin() {
      map.fitBounds(L.latLngBounds([pin.lat, pin.lng], [pin.lat, pin.lng]),
        { paddingTopLeft: [24, clearOfSearchPanel(64)], paddingBottomRight: [24, 24], maxZoom: 18 });
    }
    // the name's answers near the street, or near the pin with no street
    function nearHere(r) {
      var lat = parseFloat(r.lat), lng = parseFloat(r.lon);
      var b = bounds || L.latLngBounds([pin.lat, pin.lng], [pin.lat, pin.lng]);
      return haversineMiles(lat, lng,
        Math.max(b.getSouth(), Math.min(b.getNorth(), lat)),
        Math.max(b.getWest(), Math.min(b.getEast(), lng))) <= STREET_NEAR_MILES;
    }
    function searchName() {
      return search(name).then(function (found) {
        if (!current()) return;
        near = (found || []).filter(nearHere);
        if (!near.length) return;
        listByName(near);
        compose();
        if (pin) framePin(); // the list grew the panel the pin is framed below
      });
    }
    // the Census Bureau is asked with the ZIP code of the street's answers
    // when they name one and the address none (lookupAddressRanges)
    var ranged = lookupAddressRanges(zips.length === 1 ? withZip(bare, zips[0]) : bare, zips).then(function (m) {
      // a link's read is counted once the Census answer says what the reader
      // is shown, the house pinned or the street alone (countLinkRead); with
      // no street, only a pin is this function's to count
      if (seq === districtSearchSeq && (known || (m && current()))) countLinkRead(m && current() ? "pin" : "street");
      if (!m || !current()) return false;
      // the app that answers at the pin decides whether its point is selected
      // here (ENGINE metro-portal); an owner that cannot be told leaves it
      // here, as a tap there would
      return placeOwner(m.lat, m.lng)["catch"](function () { return null; }).then(function (owner) {
        if (!current()) return false;
        pin = m;
        pinPicked = !owner || owner === "here";
        if (!known) show();
        // the reader's tap on the map answers the message (done), and so does
        // a point selected any other way, which takes the pin
        placeRangePin(m, a, done, pinPicked);
        if (pinPicked) picked = state.sequence; // the pin's own selection answers nothing
        compose();
        framePin();
        return true;
      });
    });
    if (name.length < 3) return ranged;
    if (known) return Promise.all([ranged, searchName()]);
    return ranged.then(function (shown) {
      return shown && searchName().then(function () { return true; });
    });
  }

  /* OPENSTREETMAP'S NOTE FORM, for a reader who knows where a house number
   * the map's search lacks is. Photon is built from OpenStreetMap's data, so
   * a number missing there is missing from OpenStreetMap or from Photon's copy
   * of it, and OpenStreetMap's volunteers add one when a note says where it
   * is. Added on Adam's choice of 2026-10-17. Its wiki (wiki.openstreetmap.org
   * /wiki/Notes, read that day) says a note needs no account, that it is a
   * message from one person to another ("Don't create automated notes"), and
   * that it must not be copied from another map, "even if they do not cost
   * money or are from the government"; a reader who arrived with a Google
   * Maps link is the likeliest to copy one, which is why the message says
   * "not from another map", and a note is public, which is why the link says
   * so. So the link only opens the form, with nothing filled in, and
   * districtry sends OpenStreetMap nothing. The place is in the part of the
   * address after "#", which a browser never sends to a server: the form's
   * page reads it to open its map there and puts the new note's marker in
   * the middle of that map, where the reader drags it, or taps the map, to
   * move it to the place (read 2026-10-17 in the new-note script
   * openstreetmap.org serves, which takes ?lat=&lon= instead when given
   * them; a query string is sent, so it is not used). The link opens at the
   * pin, or, when there is no pin, at the part of the street the map is
   * showing, read again each time the map stops moving while the link is on
   * the page, because the reader may have moved the map to the place.
   * Following it is counted as osm-note, never by place. */
  var OSM_NOTE_URL = "https://www.openstreetmap.org/note/new#map=";
  function osmNoteLink(where) {
    var link = document.createElement("a");
    link.target = "_blank";
    link.rel = "noopener noreferrer";
    link.textContent = "leave OpenStreetMap a public note";
    function aim() {
      // a message rewritten or cleared has dropped the link
      if (link.href && !document.body.contains(link)) { map.off("moveend", aim); return; }
      var at = where();
      link.href = OSM_NOTE_URL + at.zoom + "/" + at.lat.toFixed(5) + "/" + at.lng.toFixed(5);
    }
    aim();
    map.on("moveend", aim);
    link.addEventListener("click", function () { trackEvent("osm-note"); });
    return link;
  }

  /* ---------- where the Census Bureau puts a house number ----------
   * A place shared from the Google Maps app is its name and its address
   * with no position, and Photon often holds the street and not the number
   * on it. The Census Bureau's geocoder estimates where a number falls on a
   * street from the address ranges in its TIGER street file. Measured
   * 2026-10-17 for a Chicago storefront Photon knew only as its street: the
   * estimate was on the right side of the street and on the storefront's
   * block, about 45 m from the building. The Census geocoder sends no CORS
   * header, so it is asked through districtry's lookup (MAP_LINK_LOOKUP,
   * ENGINE map-link; services/link-lookup/main.ts), which passes on each
   * match's address and position: the Census Bureau sees the lookup and
   * not the reader, and the privacy page names both, measured from the
   * `?a=` call below (build_privacy_page.py). From the map's search, only an
   * address the reader searched for is sent: one a shared map link carried,
   * or one typed and then sent with Search or Enter, handed over as #q=, or
   * picked from the list by its street (runAddressSearch, listHeldAnswers),
   * never text while it is being typed. An app's office-pin provider may
   * also ask about the public address of an office a card shows, when its
   * own geocoder cannot place it (New York's poiGeocodeRequest, from
   * 2026-10-20). build_privacy_page.py fails on a call from anywhere else.
   *
   * An answer is used only if it is the ONE match that is the house
   * (labelAtHouse, ENGINE map-link) in the address's town or ZIP code
   * (labelOnStreet), or in a ZIP code the map's answers on its street carry
   * (`zips`, showStreetOnly), since one number on one street is one house
   * in a ZIP code: asked on 2026-10-14 for a Westmont address, the Census
   * geocoder offered that number on Oakley Ln and on Oakley Ct, two other
   * streets. The ZIP codes are what take the house when the address names
   * no town, which the map's answers then give as the Cook County township
   * a street is in while the Census Bureau names the city (ENGINE map-link
   * addressInTownOf). When those answers name one ZIP code and the address
   * none, the lookup is asked with it (showStreetOnly). Measured
   * 2026-10-19: asked about the Lincoln Park Zoo's address in the township
   * Photon names, the Census geocoder found the zoo with the zoo's ZIP code
   * and nothing with another, and asked about City Hall's in Chicago with
   * another ZIP code, it still found City Hall. So the ZIP code costs
   * nothing where the town is the city, and a house it misses in the
   * township for being in another ZIP code would have been refused here
   * anyway. Whether it finds, typed with no town, the three of 49 Cook
   * County addresses the township alone found nothing for was not
   * measured, because those addresses were not kept.
   *
   * Until 2026-10-19 the match also had to lie within STREET_NEAR_MILES of
   * the street the map found, which refused it for 2 of 49 Cook County
   * addresses tested that day, whose street the map's search knew only more
   * than half a mile from the house; the Census Bureau had the right
   * district for both. Adam chose that day to take its answer whenever it
   * names the number, the street and the town ("Use Census").
   *
   * THE PIN'S POINT IS SELECTED AT ONCE, on Adam's choice of 2026-10-18,
   * and the message says what that can cost (showStreetOnly): an estimate
   * from address ranges can be a block or more from the house (14 to 296 m
   * for the 48 of those 49 addresses it found), and in Chicago a ward's
   * edge often runs down the alley in the middle of a block, so the cards
   * can name the district next door. For its first day the pin selected
   * nothing until it was tapped, for that reason; Adam chose the cards at
   * once, with the warning, over a tap the reader had to know to make
   * (placeRangePin). Resolves to that match's position, or to null when
   * there is none, the lookup fails or it takes longer than
   * ADDRESS_RANGES_TIMEOUT_MS; it never rejects. */
  var ADDRESS_RANGES_TIMEOUT_MS = 9000;
  var RANGE_PIN_COLOR = "#1d5fd6"; // the token file's --data-500, the selection's own blue
  function lookupAddressRanges(address, zips) {
    if (!MAP_LINK_LOOKUP || typeof AbortController === "undefined") return Promise.resolve(null);
    var ctl = new AbortController();
    var timer = setTimeout(function () { ctl.abort(); }, ADDRESS_RANGES_TIMEOUT_MS);
    return fetch(MAP_LINK_LOOKUP + "?a=" + encodeURIComponent(cleanPoiAddress(address)), { signal: ctl.signal })
      .then(function (r) { return r.ok ? r.json() : null; })
      .then(function (body) {
        var on = (body && Array.isArray(body.matches) ? body.matches : []).filter(function (m) {
          if (!m || typeof m.address !== "string" || !isFinite(Number(m.lat)) || !isFinite(Number(m.lng))) return false;
          return labelAtHouse(m.address, address) && (labelOnStreet(m.address, address) ||
            (zips || []).some(function (zip) { return labelOnStreet(m.address, withZip(address, zip)); }));
        });
        return on.length === 1 ? { lat: Number(on[0].lat), lng: Number(on[0].lng) } : null;
      })["catch"](function () { return null; })
      .then(function (m) { clearTimeout(timer); return m; });
  }
  // The ZIP code an answer's label ends with, or ""
  function labelZip(label) {
    return (/(?:^|,)\s*(\d{5})(?:-\d{4})?\s*$/.exec(String(label || "")) || [])[1] || "";
  }
  // The address with `zip` written after it, unless it names a ZIP code of
  // its own
  function withZip(address, zip) {
    var a = addressParts(address);
    return a && !a.zip ? String(address).trim() + " " + zip : address;
  }
  /* The pin, with its point selected (selectNow). That pin ignores a tap,
   * which would only move the selection a few metres off the estimate and
   * take the message with it. Where another state's app answers at the pin
   * (showStreetOnly asks placeOwner), a selection would open that app with
   * the point and without the message, so the pin waits for a tap, which
   * selects its point as a tap on the map would, through the map's own
   * click (ENGINE selection-controls), and so opens that app. Either way
   * any other selection, a new search or a tap elsewhere removes the pin. */
  var rangePin = null;
  function clearRangePin() {
    if (!rangePin) return;
    map.off("layeradd", rangePin.watch);
    map.removeLayer(rangePin.marker);
    rangePin = null;
  }
  function placeRangePin(m, a, onGone, selectNow) {
    clearRangePin();
    var at = L.latLng(m.lat, m.lng);
    var where = "Where the Census Bureau puts " + a.number + " " + a.street;
    var label = selectNow ? where : where + ". Select this point.";
    var marker = L.marker(at, { icon: poiPinIcon(RANGE_PIN_COLOR), title: label, zIndexOffset: 1000,
                                keyboard: !selectNow });
    function select() {
      clearRangePin();
      map.fire("click", { latlng: at });
    }
    if (!selectNow) {
      marker.on("click", select);
      // Leaflet makes a marker focusable and a button, and leaves Enter and
      // Space to the page
      marker.on("keydown", function (e) {
        var key = e.originalEvent && e.originalEvent.key;
        if (key !== "Enter" && key !== " ") return;
        e.originalEvent.preventDefault();
        select();
      });
    }
    marker.addTo(map);
    var el = marker.getElement();
    if (el) {
      el.setAttribute("aria-label", label);
      // a picture of where the selected point came from, which a tap leaves
      // as it is (a marker's click does not reach the map)
      if (selectNow) { el.setAttribute("role", "img"); el.style.cursor = "default"; }
    }
    if (selectNow) setSelectedPoint(at.lat, at.lng, { skipPan: true });
    var picked = state.sequence;
    // Every selection adds the selected point's marker (setSelectedPoint), so
    // a point selected any other way (a result row, the keyboard's select
    // button, a located point) is seen here, and the pin goes with it, as
    // does the message about it (onGone).
    var watch = function () {
      if (state.sequence === picked) return;
      clearRangePin();
      if (onGone) onGone();
    };
    map.on("layeradd", watch);
    rangePin = { marker: marker, watch: watch };
  }
  /* How far below the map's top edge the street is framed. The search panel
   * floats over the top of the map while it holds the message, and on a phone
   * it covers nearly half of it (measured at 412x915: the panel ends 168 px
   * into a 384 px map), so the street is framed in the part below the panel.
   * Not when that part is too short to frame a street in (`room`, 160 px
   * unless the caller frames something smaller); the street then sits at
   * the middle of the map, as any search result does. */
  function clearOfSearchPanel(room) {
    var top = 24;
    var panel = document.getElementById("search-extra");
    var r = panel ? panel.getBoundingClientRect() : null;
    var m = map.getContainer().getBoundingClientRect();
    if (r && r.height > 0 && r.left < m.right && r.right > m.left && r.bottom > m.top && m.bottom - r.bottom >= (room || 160)) {
      top = Math.max(top, Math.round(r.bottom - m.top) + 12);
    }
    return top;
  }

  /* ---------- an address handed over in the URL (#q=) ----------
   * A question page (ward.html, county-board.html and the rest) carries its
   * own address box. It does not geocode: it builds `./#q=<address>` and
   * hands the typed text to this app, which runs the same bounded search a
   * reader typing in the box runs.
   *
   * THE HASH RATHER THAN THE QUERY STRING, and that is a privacy decision.
   * GoatCounter's count.js sends location.search with every hit, so a `?q=`
   * would put a reader's typed street address into the analytics record on
   * a site whose whole standard is that a selected point is rounded to two
   * decimals before anything is sent. A hash never leaves the browser.
   *
   * It SELECTS a result only when the geocoder returns exactly one. Several
   * matches is the case where taking the first would name the wrong
   * officeholder, so the list stays up and the reader picks. A sibling-metro
   * row is never auto-clicked — that button navigates to another instance —
   * and neither is a place found by a shared link's name alone
   * (listByName), which may be another place of the same name. Two
   * exceptions: among several answers to a link, the one that is the place
   * the link names (markLinkPlace) is selected when it is the only such
   * one; and the row a link to a place in another state draws
   * (showLinkElsewhere) is followed, because the address itself says which
   * state's app it belongs to, so a place shared into the installed app
   * opens in its own state. So is the row a ZIP code another app holds
   * draws (renderZipRow), for the same reason, and a ZIP code this app holds
   * is one row, which is selected (showZip); an answer to a ZIP code no list
   * holds is never selected (zipAnswerRows), because it is one spot in it.
   *
   * NOR IS AN ANSWER THAT IS NOT THE HOUSE the address names
   * (listHeldAnswers), though it be the only one: an answer without the
   * house number is a street, and its point a spot somewhere along it, and
   * one with the number on another street is another house. Until
   * 2026-10-17 a question page's address box selected the first, and the
   * cards named the districts of another stretch of the street, and until
   * 2026-10-19 the second. */
  function prefillGeocodeSearch(q) {
    linkVia = "opened";
    geocodeInputEl.value = q;
    geoStatus.textContent = "Searching\u2026";
    geoStatus.className = "geo-status";
    reengageSearchShell();
    return runGeocodeSearch(q, null, true).then(function () {
      var own = geoResultsList.querySelectorAll("li:not(.sibling-result):not(.name-result) > button");
      var place = geoResultsList.querySelectorAll("li.link-place > button");
      var away = geoResultsList.querySelectorAll("li.map-link-away > button, li.zip-away > button");
      if (own.length === 1) { if (!/\b(?:no-number|other-street|zip-answer)\b/.test(own[0].parentNode.className)) own[0].click(); }
      else if (place.length === 1) place[0].click();
      else if (!own.length && away.length === 1) away[0].click();
    })["catch"](function (err) {
      if (err.name === "AbortError") return;
      geoStatus.textContent = "Search failed \u2014 the geocoder may be busy. Try again in a moment.";
      geoStatus.className = "geo-status err";
    });
  }

  document.getElementById("geocode-form").addEventListener("submit", function (e) {
    e.preventDefault();
    var q = document.getElementById("geocode-input").value.trim();
    if (!q) return;
    // cancel the pending input-debounce so it can't re-fire after this submit
    // and abort/duplicate the very request we're about to make (the same guard
    // the result-click handler already uses)
    clearTimeout(debounceTimer);
    linkVia = "pasted";
    geoStatus.textContent = "Searching…";
    geoStatus.className = "geo-status";
    runGeocodeSearch(q, null, true)
      .catch(function (err) {
        if (err.name === "AbortError") return;
        geoStatus.textContent = "Search failed — the geocoder may be busy. Try again in a moment.";
        geoStatus.className = "geo-status err";
      });
  });

  document.getElementById("geocode-input").addEventListener("input", function () {
    var q = this.value.trim();
    clearTimeout(debounceTimer);
    if (q.length < 3) {
      // abort any in-flight search too — its late response would otherwise
      // repopulate the just-cleared list for text no longer in the box
      if (inFlightController) inFlightController.abort();
      geoResultsList.innerHTML = "";
      geoStatus.textContent = "";
      clearRangePin();
      return;
    }
    showDistrictsNow(q)["catch"](function () { /* the debounced search below still runs */ });
    debounceTimer = setTimeout(function () {
      linkVia = "pasted";
      geoStatus.textContent = "Searching…";
      runGeocodeSearch(q).catch(function (err) {
        if (err.name === "AbortError") return;
        geoStatus.textContent = "Search failed — the geocoder may be busy. Try again in a moment.";
        geoStatus.className = "geo-status err";
      });
    }, 400); // debounced to stay a courteous client of the fork's geocoder while feeling type-ahead-responsive
  });
